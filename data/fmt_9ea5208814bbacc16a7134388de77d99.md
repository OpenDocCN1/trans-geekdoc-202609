import numpy as np
import pandas as pd
from scipy.optimize import minimize

def portfolio_optimization(returns):
    n_assets = returns.shape[1]
    args = (returns.mean(), returns.cov())
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - 1}
    bounds = tuple((0, 1) for asset in range(n_assets))
    result = minimize(portfolio_variance, n_assets*[1./n_assets], args=args, method='SLSQP', bounds=bounds, constraints=constraints)
    return result

def portfolio_variance(weights, mean_returns, cov_matrix):
    return np.dot(weights.T, np.dot(cov_matrix, weights))

### 定义问题

经典计算机基于确定性序列生成伪随机数。然而，量子计算机由于量子比特测量的概率性可以生成真随机数。通过对零态量子比特应用 `Hadamard` 门，使其进入叠加态。

### 随机数生成器

1. 定义`max`为要生成的数值。
2. 确定表示`max`所需的随机位数（`nBits`）。
3. 生成长度为`nBits`的位字符串。
4. 如果位字符串表示的数字大于`max`，则重复该过程。
5. 否则，返回生成的数字。

### 写入随机位

第一步是编写一个生成随机位的 `Q#` 操作：

```qsharp
operation GenerateRandomBit() : Result {
    // 分配一个量子比特
    use q = Qubit();
    // 使用 Hadamard 门将量子比特设置为 0 和 1 的叠加态
    H(q);
    // 测量量子比特值
    let result = M(q);
    // 将量子比特重置为 |0> 
    Reset(q);
    // 返回测量结果
    return result;
}
```

### 使用布洛球可视化 Q# 代码

在布洛球上，北极代表经典值 0，南极代表经典值 1。任何叠加态都可以由球上的一点表示。对零态量子比特应用 `Hadamard` 门会使其进入等叠加态，导致测量为 0 或 1 的概率均为 50%。

### 随机数生成器的完整代码

```qsharp
namespace QuantumNumberGenerator {
    open Microsoft.Quantum.Convert;
    open Microsoft.Quantum.Intrinsic;
    open Microsoft.Quantum.Math;

    @EntryPoint()
    operation Main() : Int {
        let max = 100;
        Message($"采样 0 到{max}之间的随机数:");
        return GenerateRandomNumberInRange(max);
    }

    operation GenerateRandomNumberInRange(max : Int) : Int {
        mutable bits = [];
        let nBits = BitSizeI(max);
        for idxBit in 1..nBits {
            set bits += [GenerateRandomBit()];
        }
        let sample = ResultArrayAsInt(bits);
        return sample > max ? GenerateRandomNumberInRange(max) | sample;
    }

    operation GenerateRandomBit() : Result {
        use q = Qubit();
        H(q);
        let result = M(q);
        Reset(q);
        return result;
    }
}
```
