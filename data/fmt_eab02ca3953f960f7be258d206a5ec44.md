            layer = self._layers[i] 
            loss_grad = layer.backward(loss_grad)
            layer.reg_grad(reg) 
        return loss_grad    
    
    def backpropagation(self, X, y,loss_function,reg=0):

        #The gradient of the loss function about the output f
        loss,loss_grad = loss_function(f,y)

        self.zero_grad()
        self.backward(loss_grad)  
        reg_loss = self.reg_loss_grad(reg)

        for i in range(len(self._layers)):
            reg_loss+=self._layers[i].reg_loss(reg)
        return reg_loss
    
    def parameters(self): 
        return self._params
    
    def zero_grad(self):

此类中 `add_layer()` 用于向神经网络添加各种层，`forward()` 接受输入数据并生成相应输出，`__call__()` 是函数调用方法，对于 `NeuralNetwork` 对象 `nn` 和输入 `X`，`nn(X)` 等价于 `nn.forward(X)`，`backward()` 接受损失函数关于网络输出的梯度，并执行反向推导以求得损失函数关于模型参数和中间变量的梯度。

### 4.3.8 独立参数优化器

为了便于使用不同的梯度下降优化策略更新模型参数，可以将它们编写为单独的类，例如：

#### SGD 优化器实现

```python
class SGD():
    def __init__(self, model_params, learning_rate=0.01, momentum=0.9):
        self.params, self.lr, self.momentum = model_params, learning_rate, momentum
        self.vs = []
        for p, grad in self.params:
            v = np.zeros_like(p)
            self.vs.append(v)
        
    def zero_grad(self):        
        for i, _ in enumerate(self.params):           
            self.params[i][1].fill(0) 
                
    def step(self):   
        for i, _ in enumerate(self.params):     
            p, grad = self.params[i]           
            self.vs[i] = self.momentum * self.vs[i] + self.lr * grad             
            self.params[i][0] -= self.vs[i] 
      
    def scale_learning_rate(self, scale):
        self.lr *= scale
```

优化器类 `SGD` 的构造函数参数 `model_params` 是一个 Python 列表对象，每个元素都是模型参数及其梯度的列表对象。如果模型有 2 个参数 `W`、`b`，其对应的梯度参数分别为 `dW`、`db`，则 `model_params` 参数形式如下：

```
[[W, dW], [b, db]]
```

构造函数的另外两个参数分别是梯度下降算法的学习率 `learning_rate` 和动量优化策略的参数 `momentum`。若将 `momentum` 设为 0，则等价于不带动量的最基本梯度更新策略。

`SGD` 的 `zero_grad()` 方法用于将所有参数对应的梯度重置为 0，`step()` 方法用于根据梯度和优化策略更新模型参数。梯度下降法在迭代过程中有时需要调整学习率，`scale_learning_rate()` 即为调整学习率的方法。

通过定义 `SGD` 类的优化器对象 `optimizer` 即可使用梯度下降法更新模型参数：

```python
learning_rate = 1e-1
momentum = 0.9
optimizer = SGD(nn.parameters(), learning_rate, momentum)
```

同理，也可定义其他优化器类，例如以下 `Adam` 优化器：

```python
class Adam():
    def __init__(self, model_params, learning_rate=0.01, beta_1=0.9, beta_2=0.999, epsilon=1e-8):
        self.params, self.lr = model_params, learning_rate
        self.beta_1, self.beta_2, self.epsilon = beta_1, beta_2, epsilon
        self.ms = []
        self.vs = []
        self.t = 0
        for p, grad in self.params:
            m = np.zeros_like(p)
            v = np.zeros_like(p)
            self.ms.append(m)
            self.vs.append(v)
        
    def zero_grad(self):        
        for i, _ in enumerate(self.params):
            self.params[i][1].fill(0) 
                
    def step(self):   
        beta_1, beta_2, lr = self.beta_1, self.beta_2, self.lr
        self.t += 1
        t = self.t
        for i, _ in enumerate(self.params):     
            p, grad = self.params[i]       
            
            self.ms[i] = beta_1 * self.ms[i] + (1 - beta_1) * grad
            self.vs[i] = beta_2 * self.vs[i] + (1 - beta_2) * grad**2 
            
            m_1 = self.ms[i] / (1 - np.power(beta_1, t))
            v_1 = self.vs[i] / (1 - np.power(beta_2, t))  
            self.params[i][0] -= lr * m_1 / (np.sqrt(v_1) + self.epsilon)
      
    def scale_learning_rate(self, scale):
        self.lr *= scale
```

#### 训练函数 `train_nn`

训练函数 `train_nn()` 接受一个数据迭代器，每次从中取出一批训练样本（输入 `input`、目标 `target`）。对每批样本，先执行 `forward()` 计算输出，再用损失函数计算其损失及关于输出 `output` 的梯度 `loss_grad`，再通过 `backward()` 函数将梯度 `loss_grad` 反向传播，得到模型参数和中间变量的梯度。最后使用优化器的 `step()` 函数更新模型参数。

```python
def train_nn(nn, X, y, optimizer, loss_fn, epochs=100, batch_size=50, reg=1e-3, print_n=10):
    iter = 0
    losses = [] 
    for epoch in range(epochs):
        for X_batch, y_batch in data_iter(X, y, batch_size):     
            optimizer.zero_grad()      
            
            f = nn(X_batch)  # nn.forward(X_batch)      
            loss, loss_grad = loss_fn(f, y_batch)       
            nn.backward(loss_grad, reg)               
            loss += nn.reg_loss(reg)
          
            optimizer.step()
            
            losses.append(loss)            
            if iter % print_n == 0:
                print(iter, "iter:", loss)
            iter += 1          

    return losses
```
