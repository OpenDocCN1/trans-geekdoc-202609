
```solidity
pragma solidity ^0.8.0;

contract AMLCompliance {

    struct Transaction {
        address sender;
        address receiver;
        uint256 amount;
        uint256 timestamp;
    }

    Transaction[] public transactions;

    function addTransaction(address receiver, uint256 amount) public {
        transactions.push(Transaction(msg.sender, receiver, amount, block.timestamp));
    }

    function monitorTransactions(uint256 threshold) public view returns (Transaction[] memory) {
        uint count = 0;
        for (uint i = 0; i < transactions.length; i++) {
            if (transactions[i].amount > threshold) {
                count++;
            }
        }

        Transaction[] memory flagged = new Transaction[](count);
        uint index = 0;
        for (uint i = 0; i < transactions.length; i++) {
            if (transactions[i].amount > threshold) {
                flagged[index] = transactions[i];
                index++;
            }
        }
        return flagged;
    }
}
```

上述合约演示了监控可疑交易的基本机制，可用于 `AML` 合规。

#### 理解 Solidity 合约结构

智能合约是存储在区块链特定地址上的代码（函数）和数据（状态）的集合。

```solidity
pragma solidity ^0.8.0;

contract HelloWorld {
    string public message;

    constructor(string memory initialMessage) {
        message = initialMessage;
    }

    function updateMessage(string memory newMessage) public {
        message = newMessage;
    }
}
```

1. **Pragma**：定义合约使用的 `Solidity` 版本。
2. **Contract**：`Solidity` 程序的基本单元。
3. **State**：如 `message` 等变量存储在区块链上。
4. **Functions**：执行操作并修改状态。示例中，`updateMessage` 允许修改消息。

#### 数据类型

`Solidity` 支持多种数据类型，包括：

1. **基本类型**：
   - `uint`：无符号整数。
   - `bool`：布尔值（true 或 false）。
   - `address`：代表以太坊地址。
2. **复杂类型**：
   - `struct`：用于创建自定义类型。
   - `mapping`：将键映射到值的数据结构。

```solidity
pragma solidity ^0.8.0;

contract DataTypes {
    struct Product {
        string name;
        uint price;
    }

    mapping(uint => Product) public products;

    function addProduct(uint id, string memory name, uint price) public {
        products[id] = Product(name, price);
    }
}
```

在此合约中，`mapping` 使用 ID 作为键来存储产品。

#### 支付合约

该合约演示了如何发送和接收以太币。

```solidity
pragma solidity ^0.8.0;

contract Payment {
    address public owner;

    constructor() {
        owner = msg.sender;
    }

    function sendPayment() public payable {
        require(msg.value > 0, "Payment must be greater than 0");
    }

    function withdraw() public {
        require(msg.sender == owner, "Only owner can withdraw");
        payable(owner).transfer(address(this).balance);
    }

    function getBalance() public view returns (uint) {
        return address(this).balance;
    }
}
```

1. **函数 `sendPayment`**：允许用户向合约发送以太币。
2. **函数 `withdraw`**：只有所有者可以提取累积的余额。
3. **函数 `getBalance`**：返回合约的当前余额。
