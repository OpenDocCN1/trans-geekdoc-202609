- 当传递变量索引时，输出该类的系数。

### 列表 5.18. 返回系数的 ML 内部类

```java
class MultipleLinearRegression {
    private final int N; // 记录数
    private final int p; // 因变量数量
    private final Matrix beta; // 回归系数
    private double SSE; // 平方和
    private double SST; // 平方和

    public MultipleLinearRegression(double[][] x, double[] y) {
        // 如果记录数不一致则抛出异常
        if (x.length != y.length) {
            throw new RuntimeException("dimensions don't agree");
        }
        N = y.length; // 总记录数
        p = x[0].length; // 因变量数量

        Matrix X = new Matrix(x);

        // 从向量创建矩阵
        Matrix Y = new Matrix(y, N);

        // 使用 JAMA 库中的 QRDecomposition 类寻找最小二乘解
        QRDecomposition qr = new QRDecomposition(X);
        // 求解 A*X = B
        beta = qr.solve(Y);
        // y[] 值的平均值，即因变量
        double sum = 0.0;
        for (int i = 0; i < N; i++) {
            sum += y[i];
        }
        // 计算平均值
        double mean = sum / N;
        // 需要解释的总变异
        for (int i = 0; i < N; i++) {
            double dev = y[i] - mean;
            SST += dev * dev;
        }
        // 未解释的变异
        Matrix residuals = X.times(beta).minus(Y);
        SSE = residuals.norm() * residuals.norm();

        // 当传递变量索引时输出类的系数
        public double beta(int j) {
            return beta.get(j, 0);
        }

        public double R2() {
            return 1.0 - SSE / SST;
        }
```
