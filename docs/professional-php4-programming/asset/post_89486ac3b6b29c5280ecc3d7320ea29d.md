    // Returns an array of all privileges

validate() checks the submitted form for valid data and creates a new Privilege object depending on the results. If an ID is supplied and no Privilege with that ID exists, a new Privilege with that ID is created. If no ID is supplied, a blank Privilege is used:

function validate()
{

    if ($priv_id) {
        // Make sure it's a valid PrivID

        if (!$oPriv->privilegeExists) return 0;

    return 1;
}

function save()

    global $oPriv, $description;

    $oPriv->setDescription(addslashes($description));
    if ($oPriv->privilegeExists) {

}
