```
if (!$db->open()) {
    die ($db->error());
}

if (!$db->query('SELECT * FROM author')) {
    die ($db->error());
}

echo("Number of rows: " . $db->numRows() . "<br />");

while ($row = $db->fetchAssoc()) {
    echo($row['auth_id'] . " " . $row['auth_name'] . "<br />");
}

$db->freeResult();
$db->close();
```
