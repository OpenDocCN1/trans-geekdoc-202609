# ONE-PERIOD BINOMIAL OPTION PRICING MODEL - Calculating the Call Option Premium

# INPUT

+   - Current Bitcoin (S)= $ 47,000
+   - Up (u) = 1.20x
+   - Down (d) = 0.90x
+   - Exercise Price (X) = $ 50,000
+   - Risk Free Rate (I) = 5.00%
+   - Time in Years (t) = 1
+   - Periods = 1

# OUTPUT

PERIOD 0 PERIOD 1

Su= 56,400 -> Cu= 6,400.00

S = 47,000.0

Sd = 42,300 -> Cd= -

C = 3,047.62

p = 0.50
1-p= 0.50

C= 3,047.62 European Option Premium

# FORMULAS

+   Su = S . u
+   Sd = S . d
+   Cu = Max (0, Su - X)
+   Cd = Max (0, Sd - X)
+   p = [(1+i) - d ] / (u - d)
+   C= [ (p . Cu) + [(1-p) Cd] ] / [(1+i)^t]
