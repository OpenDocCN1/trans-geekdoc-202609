    $pdo = $this->db;

    $insert = $pdo->prepare('INSERT INTO twitterLog (tdate, thash, tweet)
        VALUES (:date, :code, :text)');
    $params = array(':date' => date('Y-m-d'), ':code' => $this->lastTweet, ':
        text' => $message);
    try {

        throw $e;
    }

    return FALSE;
}
