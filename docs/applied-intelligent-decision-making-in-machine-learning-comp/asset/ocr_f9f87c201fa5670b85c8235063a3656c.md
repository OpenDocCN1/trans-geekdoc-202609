# # Creating the predicted classes

```
```
L1 <- (1/2)*t(mu_1)*%inv(Sp_1)*%mu_1 + log(n/1)
L2 <- (1/2)*t(mu_2)*%inv(Sp_2)*%mu_2 + log(n/2)
L3 <- (1/2)*t(mu_3)*%inv(Sp_3)*%mu_3 + log(n/3)
ans_max <- max(L1, L2, L3)
for (i in 1:n) {
  if (ans_max == L1) {
    ans[i,1] <- 1
  } else if (ans_max == L2) {
    ans[i,1] <- 2
  } else if (ans_max == L3) {
    ans[i,1] <- 3
  }
}
