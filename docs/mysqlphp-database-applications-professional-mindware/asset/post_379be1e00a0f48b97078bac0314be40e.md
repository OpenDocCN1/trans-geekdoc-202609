    // Innocent until proven guilty

    switch ($Name)
    {
case "mcd" :
    $GoodCard = ereg("^5[1-5].{14}$", $Num);
    break;

case "vis" :
    $GoodCard = ereg("^4.{15}$|^4.{12}$", $Num);
    break;

case "amx" :

    break;

$Total = 0;

for ($x=0; $x<strlen($Num); $x++)
{
$digit = substr($Num,$x,1);

//    If it's an odd digit, double it
    if ($x/2 != floor($x/2))
    {
        $digit *= 2;

    if (strlen($digit) == 2)
