require 'vendor/autoload.php';

use Elasticsearch\ClientBuilder;

$client = ClientBuilder::create()->setHosts(['localhost:9200'])->build();

// Document indexing
$params = [
    'index' => 'my_index',
    'body' => ['key' => 'value']
];

$response = $client->index($params);

// Search queries
$searchParams = [
    'index' => 'my_index',
    'body' => [
        'query' => [
            'match' => ['key' => 'value']
        ]
    ]
];

$searchResponse = $client->search($searchParams);
