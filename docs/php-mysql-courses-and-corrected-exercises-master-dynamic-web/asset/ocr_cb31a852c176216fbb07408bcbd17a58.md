Write a function that takes an array of numbers as a parameter and returns the sum of all even numbers in the array.

<?php
function sum_even_numbers($tab) {
    $sum = 0;
    foreach ($tab as $number) {
        if ($number % 2 == 0) {
            $sum += $number;
        }
    }
