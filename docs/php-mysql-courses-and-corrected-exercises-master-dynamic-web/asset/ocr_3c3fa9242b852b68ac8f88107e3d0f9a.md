function greaterNumbers($number1, $number2, $number3) {
    $bigNumber = $number1;

    if ($number2 > $bigNumber) {
        $bigNumber = $number2;
    }

    if ($number3 > $bigNumber) {
        $bigNumber = $number3;
    }

    return $bigNumber;
}
