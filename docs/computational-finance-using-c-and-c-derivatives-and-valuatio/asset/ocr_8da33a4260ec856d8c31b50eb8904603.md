public class TPair : IComparable
{
    public double t;
    public double val;

    public TPair(double t1, double val1 {
        t = t1;
        val = val1;
    }

    int CompareTo(object obj) {
        TPair temp = (TPair)obj;
        if (this.t < temp.t)
            return -1;
        if (this.t > temp.t)
            return 1;
        else
            return 0;
    }
}
