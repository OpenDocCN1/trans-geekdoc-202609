using System;
namespace Computational_lib
{
    public class GenericityBasketOptionDealAverageCall : GenericityBasketOptionDeal
    {
        public override string Name(string Name)
        {
            string temp_string = "";
            temp_string = "Generic: Call on the average of " + n.ToString() + " assets";
            return temp_string;
        }

        public override double Payoff()
        {
            // Implement Call on the average of n, assets
            double the_average = 0.0;
            double pay_val = 0.0;
            the_average = ST[0];
            for (int j = 1; j < n; ++j)
            {
                the_average += ST[j];
            }
            the_average /= n;
            pay_val = Math.Max(the_average - Strike_, zero);
            return pay_val;
        }
    }
}
