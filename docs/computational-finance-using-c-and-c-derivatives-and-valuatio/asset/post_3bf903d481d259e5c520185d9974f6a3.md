        {
            // Implement Call on the average of n, assets
            double the_average = 0.0;
            double pay_val = 0.0;
            the_average = ST[0];
            for (int j = 1; j < n; ++j)
            {
                the_average += ST[j];
            }
            the_average /= n;
            pay_val = Math.Max(the_average - Strike_, 0);
            return pay_val;
        }

double time_step = Time_To_Expiry / TimeSteps;
double sqrt_time_step = System.Math.Sqrt(time_step);

double opt_val = 0.0;
bool not_out = true;
int k = 0;
double STN = 0.0;
double mean = (RiskFreeRate - DividendYield - Signal * Signal * 0.5) * time_step;
double z = 0;

for (int i = 0; i < NumberScenarios; i++) {
    // generate the asset path
    double ST1 = STN;
    not_out = true;
    k = 0;

    while (not_out && k < TimeSteps) {
        z = PricingUtils.NextNormal(std);
        ST1 = ST1 * System.Math.Exp(z);
        if (ST1 < BarrierLevel) not_out = false;
        k++;
    }
    asset_path[i] = ST1;

    if (is_put) {
        opt_val = System.Math.Max(Strike - STN, 0.0);
    } else {
        opt_val = System.Math.Max(STN - Strike, 0.0);
    }

    if (not_out) {
        // only if asset value is above the barrier_
        // compute the probability that the asset remained above the barrier
        if (UseBrownbridge) {
            double total_probability = 1.0, pr;
            double sigma2 = Signal * Signal;
            double barrier_level = System.Math.Log(BarrierLevel);
            for (int j = 0; j < TimeSteps; j++) {
                double log_s_i = System.Math.Log(asset_path[j]);
                double log_s_i1 = System.Math.Log(asset_path[j+1]);
                double fac = -2.0 * (barrier_level - log_s_i) * (barrier_level - log_s_i1) / (sigma2 * time_step);
                pr = System.Math.Exp(fac); // probability of staying above the barrier between i and i+1
                total_probability *= pr;
            }
            sum_opts = total_probability * opt_val * disc;
        } else {
            // don't use the Brownian Bridge
            sum_opts = opt_val * disc;
        }
    }
}
double temp = sum_opts / (double)NumberScenarios;
