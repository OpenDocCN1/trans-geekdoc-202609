seq

std[] /* the standard deviations of the independent normal variates, idim must be less than 49 */

quasi[] /* the elements quasi[0],... quasi[idim-1] the independent Normal variates */
 i;
double v1, v2;

    return 1;
}
 for (i = 0; i < idim; i += 2) {

    }

} else if (seq == 2) {
    Generate_Sobol_Sequence(call, idim, QUASI);
}

 indl[1] = i+1;
