            double barrier_level = System.Math.Log(BarrierLevel);
            for (int j = 0; j < TimeSteps; j++) {
                double log_s_i = System.Math.Log(asset_path[j]);
                double log_s_i1 = System.Math.Log(asset_path[j+1]);
                double fac = -2.0 * (barrier_level - log_s_i) * (barrier_level - log_s_i1) / (sigma2 * time_step);
                pr = System.Math.Exp(fac); // probability of staying above the barrier between i and i+1
                total_probability *= pr;
