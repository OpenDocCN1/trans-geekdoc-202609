void bs_lattice(double value, double greeks[], double S0, double X, double sigma, double T, double r, int put, long *flag) {
    // Input parameters:
    S0 // the current price of the underlying asset
nX // the strike price
    sigma // the volatility
T // the time to maturity
    r // the interest rate
    q // the continuous dividend yield
    put // if put is 0 a call option, otherwise a put option
M // the number of time steps
    ===========================================
    value // the value of option, greeks[] // the hedge statistics as follows: greeks[0] is gamma,
           greeks[1] is delta, greeks[2] is theta,
           an error index.

    // allocate the arrays s[((M+1)/2)], and v[M+1] */
    dt = T/(double)M;
    u = exp(sigma*sqrt(dt));
    d = 1/u;
    p = (exp(r*dt)-d)/(u-d);
    if (p < 0 || (p > 1.0)) return; // Invalid probability */
    discount = exp(-r*dt);
    n = p*discount;
    j = 0;
    s[0] = S0;
    /* The "price" of j, at a given instant, the value of the asset price */
    for (int i=1; i<=M; i++) { /* Calculate values at (M-i)th time step */
        for (int j=i; j>=1; j--) {
            s[j] = u*s[j-1];
        }
        s[0] = d*s[0];
    }

    for (int i=0; i<=M-n; i++) { /* Use Black-Scholes for the final step */
        black_scholes(TNL,NULL,s[i], X, sigma, r, put, &flag[i]);
    }

    for (int i=M-1; i>=0; i--) { /* work backwards through the lattice */
        for (int j=0; j<=i; j++) {
            if (v[j] == 0 && v[j+1] == 0) {
                hold = 0;
            } else {
                if (put) {
                    v[j] = MAX(hold, s[j]-X);
                } else {
                    v[j] = MAX(hold, X-s[j]);
                }
            }
            hold = v[j];
        }
    }
}
