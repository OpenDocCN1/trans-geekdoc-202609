// readSecret reads a file from /var/openfaas/secrets/<name> or
// returns an error
func readSecret(name string) (string, error) {
    data, err := os.ReadFile("/var/openfaas/secrets/" + name)
    if err != nil {
        return "", err
    }
    return strings.TrimSpace(string(data)), nil
}
