```
version: 1.0
provider:
  name: openfaas
  gateway: http://127.0.0.1:8080
functions:
  bcrypt-fn:
    lang: golang-middleware
    handler: ./bcrypt-fn
    image: docker.io/alexellis2/bcrypt-fn:latest
```

You can change the latest tag to a specific version such as 0.1.0 and then increment this before each build of your function.

Update the bcrypt-fn/handler.go file so that it computes a bcrypt hash of whatever input you send it:

```
package function

import (
  "fmt"
  "io"
  "net/http"

  "golang.org/x/crypto/bcrypt"
)

func Handle(w http.ResponseWriter, r *http.Request) {
  var input []byte

  if r.Body != nil {
    defer r.Body.Close()
    body, _ := io.ReadAll(r.Body)
    input = body
  }

  res, err := bcrypt.GenerateFromPassword(input, bcrypt.DefaultCost)
  if err != nil {
    http.Error(w, fmt.Sprintf("Error: %s", err), http.StatusInternalServerError)
    return
  }

  w.WriteHeader(http.StatusOK)
  w.Write(res)
}
```
