    people, err := getAstros(apiURL)
    if err != nil {
        log.Fatal(err)
    }
    fmt.Printf("%d people found in space.\n", people.Number)

    for _, p := range people.Person {
        fmt.Printf("Let's wave to: %s\n", p.Name)
    }
}
