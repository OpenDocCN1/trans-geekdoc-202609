package function

import (
    "database/sql"
    "fmt"
    _ "github.com/lib/pq"
)

func connect(user, password, host, dbName, sslMode string, port int) (*sql.DB, error) {
    connStr := "postgres://" + user + ":" + password + "@" + host + ":" + fmt.Sprintf("%d", port) + "/" + dbName + "?sslmode=" + sslMode

    db, err := sql.Open("postgres", connStr)
    if err != nil {
        return nil, err
    }

    err = db.Ping()
    if err != nil {
        return nil, err
    }

    return db, nil
}

func create(db *sql.DB, todo Todo) error {
    if len(todo.Description) == 0 {
        return fmt.Errorf("description is required")
    }

    res, err := db.Query(`INSERT INTO
    todo
    (id, description, created_date)
    VALUES
    (DEFAULT, $1, now());`,
        todo.Description)

    if err != nil {
        return err
    }

    defer res.Close()
    return nil
}

func list(db *sql.DB) ([]Todo, error) {
    rows, err := db.Query(`SELECT
        id,
        description,
        created_date,
        completed_date
        FROM todo;`)

    if err != nil {
        return []Todo{},
            fmt.Errorf("unable to get from todo table: %w", err)
    }

    todos := []Todo{}
    defer rows.Close()
    for rows.Next() {
        result := Todo{}
        err := rows.Scan(&result.ID,
            &result.Description,
            &result.CreatedDate,
            &result.CompletedDate)
        if err != nil {
            return []Todo{}, fmt.Errorf("row scan error: %w", err)
        }
        todos = append(todos, result)
    }
    return todos, nil
}

## Managing config between multiple environments

Whenever you move this application between a staging and production environment, you need to alter the database URL and the secrets for it.

The secrets are relatively simple to set up, you just run faas-cli secrets against both environments and set the values as needed.

For the environment variables, you can create override files and specify these during deployment.

Example contents of: staging.yaml:

```
environment:
  db_host: "todo-pg11-do-user-2197152-0.b.db.ondigitalocean.com"
  db_port: "25060"
  db_name: "defaultdb"
  db_ssl_mode: "require"
```

Example contents of todo-fn.yaml:

```
version: 1.0
provider:
  name: openfaas
  gateway: http://127.0.0.1:8080
functions:
  todo-fn:
    lang: golang-middleware
    handler: ./todo-fn
    image: docker.io/alexellis2/todo-fn:latest
    secrets:
      - db-user
      - db-pass
    environment_file:
      - ${OF_ENV:-staging}.yaml
```

This configuration picks staging.yaml as a default, but if you configure the OF_ENV variable with prod, then the values from prod.yaml will be used instead.

func makeCreate() func(http.ResponseWriter, *http.Request) {
    return func(w http.ResponseWriter, r *http.Request) {
        if r.Body != nil {
            defer r.Body.Close()
            c := Customer{}
            data, _ := io.ReadAll(r.Body)
            if err := json.Unmarshal(data, &c); err != nil {
            http.Error(w, "unable to parse input as a Customer", http.StatusInternalServerError)
                return
            }

            log.Printf("Create for %s", c.ID)
            w.WriteHeader(http.StatusCreated)
        }
    }
