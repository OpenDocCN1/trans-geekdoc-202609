void main(void)
gost_ctx gc;
u4 k[8],data[10];
int i;

/* Initialize context. */
gost_init(&gc);

/* Prepare --a simple key should be OK, with this many rounds! */
for(i=0;i<8;i++) k[i] = i;
gost_key(&gc,k);

/* Try some test vectors. */
data[0]=0; data[1]=0;
gost_crypt(&gc,data);
printf("Enc of zero vector: %%08x%%08x\n\n",data[0],data[1]);
gost_decrypt(&gc,data);
printf("Enc of above: %%08x%%08x\n\n",data[0],data[1]);
data[0] = 0xffffff;
gost_crypt(&gc,data);
printf("Enc of ones vector: %%08x%%08x\n\n",data[0],data[1]);
gost_decrypt(&gc,data);
printf("Enc of above: %%08x%%08x\n\n",data[0],data[1]);

/* Does gost_dec() properly reverse gost_enc()? Do
we deal OK with single-block lengths passed in gost_dec())
we do OK with different lengths passed in? * */

/* Init data * */
for(i=0;i<10;i++) data[i]=i;

/* Encrypt data as 5 blocks. */
gost_enc(&gc,data,5);

/* Display encrypted data. */
for(i=0;i<10;i+=2) printf("Block %%02d = %%08x%%08x\n\n",i/2,data[i],data[i+1]);

/* Decrypt in different sized chunks. */
gost_dec(&gc,data,1);
gost_dec(&gc,data+2,4);
printf("%s\n");
