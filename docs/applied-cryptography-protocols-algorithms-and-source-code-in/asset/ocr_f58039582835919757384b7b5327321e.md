unsigned long r3;               /* 23 bit shift register *

r2 = ((key[2] >> 3) | (key[3] << 5) | (key[4] << 13) | (key[5] << 21)) & 0x3ffffff;
r3 = ((key[5] >> 5) | (key[6] << 7) | (key[7] << 15)) & 0x7ffff;
