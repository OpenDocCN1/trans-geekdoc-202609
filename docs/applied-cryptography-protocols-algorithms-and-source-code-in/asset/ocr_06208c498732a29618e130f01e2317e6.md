unsigned char *key;
short edf;
{
register int i, j, l, m, n;
    unsigned char pclim[56], pcr[56];
    unsigned long kn[32];

    for ( j = 0; j < 56; j++) {
        l = pclim[j];
        m = 1 & 07;
        pclim[j] = (key[l] >> 3) & bytebit[m] ? 1 : 0;
        for(i = 0; i < 16; i++) {
            if(edf & 1) m = (15 - i) << 1;
            else m = i << 1;
            n = m + 1;
            kn[m] = kn[n];
            for j = 0; j < 28; j++) {
                l = j + totrot[i];
                if (l < 28) pcr[j] = pclim[l];
                else pcr[j] = pclim[l - 28];
            }
        }

for ( j = 28; j < 56; j++) {
        l = j + totrot[i];
        if (l < 56) pcr[j] = pclim[l];
        else pcr[j] = pclim[l - 28];
    }

for ( j = 0; j < 24; j++) {
    if(pc2[pc2[j]]) kn[m] = bigbyte[j];
    if(pc2[pc2[j+24]]) kn[n] = bigbyte[j];
}
}
