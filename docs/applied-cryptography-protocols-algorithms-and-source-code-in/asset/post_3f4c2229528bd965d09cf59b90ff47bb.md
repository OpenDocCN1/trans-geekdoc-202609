short exp8(short a, short b, short gen) { /* base of exponentiation for multiply */
    gen;                           /* irreducible polynomial generating Galois Field */
    short product = 0;             /* result of multiplication */
    while(b != 0) {                /* multiplier is non-zero */
        if (b & 1) {
            product *= a;              /* add multiplicand LSB of b set */
            a <<= 1;                   /* shift multiplicand one place */
