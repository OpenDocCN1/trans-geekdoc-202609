void gost_enc(gost_ctx *c, u4 *d, int blocks)
    int i;

    for(i=0;i<blocks;i++) {
        gostcrypt(c,d);
        d+=2;
    }

void gost_dec(gost_ctx *c, u4 *d, int blocks)
    int i;

    for(i=0;i<blocks;i++) {
        gostdecrypt(c,d);
        d+=2;
    }

void gost_key(gost_ctx *c, u4 *k)
    int i;

    for(i=0;i<8;i++) c->k[i]=k[i];
