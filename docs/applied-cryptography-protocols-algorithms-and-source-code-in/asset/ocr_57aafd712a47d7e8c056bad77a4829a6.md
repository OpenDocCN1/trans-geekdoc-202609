/* Encrypt several blocks in ECB mode. Caller is responsible for
short blocks. */
void des_enc(des_ctx *dc, unsigned char *data, int blocks[]) {
unsigned long work[2];
int i;
unsigned char *cp;

cp = data;
for(i=0;i<blocks;i++) {
scrunch(work,cp);
desfunc(work,dc->ek);
unscrun(work,cp);
ccp=d;
}
