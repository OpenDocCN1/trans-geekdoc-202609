\# 输入：权重和偏置
\# ：梯度
\# ：学习率
\# 输出：1 次迭代后更新的权重和偏置
function \[weights, biases\] = gradientDescent(weights, biases, gradsW, gradsB, learningRate)
    L = size(weights)(2); \# 神经网络中的层数
    \# 每个参数的更新规则。
    for l = 1:L
        weights{l} = weights{l} - learningRate * gradsW{l};
        biases{l} = biases{l} - learningRate * gradsB{l};
    endfor
end
