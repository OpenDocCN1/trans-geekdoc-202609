## 定义 Relu 函数

```matlab
function [A,cache] = relu(Z)
A = max(0,Z);
cache=Z;
end
```

## 定义 Softmax 函数

```matlab
function [A,cache] = softmax(Z)
# 获取未归一化的概率
exp_scores = exp(Z');
# 对每个样本进行归一化
A = exp_scores ./ sum(exp_scores,2);
cache=Z;
end
```

## 定义 Relu 导数

```matlab
function [dZ] = reluDerivative(dA,cache)
Z = cache;
dZ = dA;
# 获取大于 0 的元素
a = (Z > 0);
# 仅选择 Z > 0 的那些元素
dZ = dZ .* a;
end
```

## 定义 Softmax 导数

```matlab
function [dZ] = softmaxDerivative(dA,cache,Y)
Z = cache;
# 获取未归一化的概率
exp_scores = exp(Z');
# 对每个样本进行归一化
probs = exp_scores ./ sum(exp_scores,2);
# dZ = pi - yi
a=sum(Y==0);
b=sum(Y==1);
c=sum(Y==2);
m= repmat([1 0 0],a,1);
n= repmat([0 1 0],b,1);
o= repmat([0 0 1],c,1);
yi=[m;n;o];
dZ=probs-yi;
end
```

## 计算深度网络中第 'l' 层的前向传播激活值

**输入**：
- `A_prev` - 上一层的激活值
- `W,b` - 权重和偏置矩阵及向量
- `activationFunc` - 激活函数 - sigmoid、tanh、relu 等

**返回**：`A`、`forward_cache`、`activation_cache`

**公式**：
- `Z = W * X + b`

## 计算 1 个周期的反向传播

**输入**：
- 神经网络参数 - `dA`
- `cache` - `forward_cache` 和 `activation_cache`
- 输入特征
- 输出值 `Y`

**返回**：`dA_prev`、`dW`、`db`

**公式**：
- `dL/dWi= dL/dZi*Al-1`
- `dl/dbl = dL/dZl`
- `dL/dZ_prev=dL/dZl*W`

```matlab
function [dA_prev dW db] = layerActivationBackward(dA, forward_cache, activation_cache, Y, activationFunc)
if (strcmp(activationFunc,"relu"))
dZ = reluDerivative(dA, activation_cache);
elseif (strcmp(activationFunc,"sigmoid"))
dZ = sigmoidDerivative(dA, activation_cache);
elseif(strcmp(activationFunc, "tanh"))
dZ = tanhDerivative(dA, activation_cache);
elseif(strcmp(activationFunc, "softmax"))
dZ = softmaxDerivative(dA, activation_cache,Y);
endif

A_prev = forward_cache{1};
numTraining = size(A_prev)(2);

# 如果激活函数是 softmax
if(strcmp(activationFunc, "softmax"))
W =forward_cache{2};
b = forward_cache{3};
dW = 1/numTraining * A_prev * dZ;
db = 1/numTraining * sum(dZ,1);
dA_prev = dZ*W;
else
W =forward_cache{2};
b = forward_cache{3};
dW = 1/numTraining * dZ * A_prev';
db = 1/numTraining * sum(dZ,2);
dA_prev = W'*dZ;
endif
end
```

import numpy as np
import matplotlib.pyplot as plt
import matplotlib
import matplotlib.pyplot as plt
from matplotlib import cm
import math

# 计算向量的 sigmoid
def sigmoid(Z):
    A=1/(1+np.exp(-Z))
    cache=Z
    return A,cache

# 计算向量的 Relu
def relu(Z):
    A = np.maximum(0,Z)
    cache=Z
    return A,cache

# 计算向量的 tanh
def tanh(Z):
    A = np.tanh(Z)
    cache=Z
    return A,cache

# 计算向量的 softmax
def softmax(Z):
    # 获取未归一化的概率
    exp_scores = np.exp(Z.T)
    # 对每个样本进行归一化
    A = exp_scores / np.sum(exp_scores, axis=1, keepdims=True)
    cache=Z
    return A,cache

# 计算向量的稳定 Softmax
def stableSoftmax(Z):
    # 以数值稳定的方式计算向量 x 的 softmax。
    shiftZ = Z.T - np.max(Z.T,axis=1).reshape(-1,1)
    exp_scores = np.exp(shiftZ)
    # 对每个样本进行归一化
    A = exp_scores / np.sum(exp_scores, axis=1, keepdims=True)
    cache=Z
    return A,cache

# 计算 Relu 的导数
def reluDerivative(dA, cache):
    Z = cache
    dZ = np.array(dA, copy=True) # 只是将 dz 转换为正确的对象。
    # 当 z <= 0 时，也应将 dz 设为 0。
    dZ[Z <= 0] = 0
    return dZ

# 计算 Sigmoid 的导数
def sigmoidDerivative(dA, cache):
    Z = cache
    s = 1/(1+np.exp(-Z))
    dZ = dA * s * (1-s)
    return dZ

# 计算 tanh 的导数
def tanhDerivative(dA, cache):
    Z = cache
    a = np.tanh(Z)
    dZ = dA * (1 - np.power(a, 2))
    return dZ

# 计算 Softmax 的导数
def softmaxDerivative(dA, cache,y,numTraining):
    # 注意：未使用 dA。dL/dZ = dL/dA * dA/dZ = pi-yi
    Z = cache
    # 计算 softmax
    exp_scores = np.exp(Z.T)
    # 对每个样本进行归一化
    probs = exp_scores / np.sum(exp_scores, axis=1, keepdims=True)
    # 计算分数上的梯度
    dZ = probs
    # dZ = pi - yi
    dZ[range(int(numTraining)),y[:,0]] -= 1
    return(dZ)

# 计算稳定 Softmax 的导数
def stableSoftmaxDerivative(dA, cache,y,numTraining):
    # 注意：未使用 dA。dL/dZ = dL/dA * dA/dZ = pi-yi
    Z = cache
    # 计算稳定 softmax
    shiftZ = Z.T - np.max(Z.T,axis=1).reshape(-1,1)
    exp_scores = np.exp(shiftZ)
    # 对每个样本进行归一化
    probs = exp_scores / np.sum(exp_scores, axis=1, keepdims=True)
    #print(probs)
    # 计算分数上的梯度
    dZ = probs
    # dZ = pi - yi
    dZ[range(int(numTraining)),y[:,0]] -= 1
    return(dZ)
