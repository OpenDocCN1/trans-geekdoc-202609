import numpy as np

def convolution(A, B):
    l_A = np.size(A)
    l_B = np.size(B)
    C = np.zeros(l_A + l_B - 1)

    for m in np.arange(l_A):
        for n in np.arange(l_B):
            C[m + n] = C[m + n] + A[m] * B[n]

    return C

def crosscorrelation(A, B):
    return convolution(np.conj(A), B[::-1])
