    gen = Activation('relu')(gen)

    gen = Reshape((7, 7, 512))(gen)

    gen = Conv2D(128, (4,4), padding='same', kernel_initializer=init)(gen)
    gen = Activation('relu')(gen)
    gen = BatchNormalization()(gen)
