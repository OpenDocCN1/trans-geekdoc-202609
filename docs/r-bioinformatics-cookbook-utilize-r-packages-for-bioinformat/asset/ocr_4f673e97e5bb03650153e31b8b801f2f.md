1. Import the required libraries:

```
library(GenomicRanges)
library(gmapR)
library(rtracklayer)
library(VariantAnnotation)
library(VariantTools)
```

2. Then, load the datasets:

```
bam_file <- fs::package("extdata",
  "hg17_snps.bam",
  package="rbioinfocookbook"
)

fasta_file <- fs::package("extdata",
  "chr17.83k.fa",
  package="rbioinfocookbook"
)
```

3. Set up the genome object and the parameter objects:

```
fa <- rtracklayer::FastFile(fast_file)

gnome <- gmapR::GmapGenome(fa, create=TRUE)

qual_params <- TallyVariantsParam(
  genome = genome,
  minimum_mapq = 20)

var_params <- VariantCallingFilters(read.count = 19,
  p.lower = 0.01)
```

4. Call the variants:

```
called_variants <- callVariants(bam_file,
  qual_params,
  calling.filters = var_params
)

head(called_variants)
```
