```r
# Convert named columns to factors
#
# code{factorise_cols} returns a copy of the passed data frame
in which all
# explicitly named columns are converted to factors with \ncode{as.factor}
# All columns with only NA values are omitted.
#
# @param df data frame
# @param col_list vector of quoted column names
# @keywords internal
#
factorise_cols <- function(df, col_list) {
  not_all_na <- function(x) any(!is.na(x))

  for ( i in col_list) {
    df <- dplyr::mutate(!!(i)) := as.factor(!!(i))
  }
  df <- dplyr::select_if(not_all_na)
}
```
