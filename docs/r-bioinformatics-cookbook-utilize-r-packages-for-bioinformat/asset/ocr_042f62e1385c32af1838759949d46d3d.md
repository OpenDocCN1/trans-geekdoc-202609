Get counts in windows across the genome:

```
library (csaw)
library (rbioinfocookbook)
```

```
bam_path <- fs::path_package("extdata",
    "windows.bam",
    package="rbioinfocookbook"
)
```

```
gff_path <- fs::path_package("extdata",
    "genes.gff",
    package="rbioinfocookbook"
)
```

```
whole_genome <- csaw::windowCounts(
    bam_path,
    bin = TRUE,
    filter = 500,
    param = csaw::readParam(minq = 20,
        dedup = TRUE,
        pe = "both"
)
```

```
colnames(whole_genome) <- c("small_data")
annotated_regions <--
get_annotated_regions_from_gff(gff_path)
```

Find overlaps between annotations and our windows, and subset the windows:

```
library (IRanges)
library (SummarizedExperiment)
```

```
windows_in <- IRanges::overlapsAny(
SummarizedExperiment::rowRanges(whole_genome),
annotated_regions)
```

Subset the windows into those annotated and non-annotated regions:

annotated_window_counts <- whole_genome[windows_in_genes, ]
