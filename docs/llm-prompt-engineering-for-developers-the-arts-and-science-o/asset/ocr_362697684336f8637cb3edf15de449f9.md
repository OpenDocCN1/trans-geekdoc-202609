```
import os
import openai
openai.api_key = os.getenv("OPENAI_API_KEY")

print("=== Knowledge Generation 1 ===")
prompt1 = ""
What are the different types of chatbots? AI chatbots, rule-based chat\b
bots, live chat agents..etc
"""
response1 = openai.Completion.create(
    model="text-davinci-003",
    prompt=prompt1,
    temperature=1,
    max_tokens=500,
    top_p=1,
    frequency_penalty=0,
    presence_penalty=0
)
text1 = response1['choices'][0]['text']
print(text1)
```
