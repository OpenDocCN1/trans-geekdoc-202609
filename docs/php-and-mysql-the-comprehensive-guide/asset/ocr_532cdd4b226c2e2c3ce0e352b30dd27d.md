function createDB() {
    $db = new SQLite3($GLOBALS["sessions_db"]);

}

function timestamp() {
    return date("YmdHis");
}

    if (!file_exists($GLOBALS["sessions_db"])) {
        createDB();

}

    $GLOBALS["sessions_id"]->close();
    unset($GLOBALS["sessions_id"]);
    return true;
}
