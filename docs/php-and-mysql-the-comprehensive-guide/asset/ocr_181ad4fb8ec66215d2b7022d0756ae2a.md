$white = imagecolorallocate($image, 255, 255, 255);

imagefilledarc($image, 100, 100, 150, 150, 0, 180, $red, IMG_ARC_PIE);

imagefilledarc($image, 100, 100, 150, 150, 180, 260, $blue, IMG_ARC_PIE);
