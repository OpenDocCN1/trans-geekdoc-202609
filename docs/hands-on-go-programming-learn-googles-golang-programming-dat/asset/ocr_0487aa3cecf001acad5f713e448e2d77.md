```
package main
import "fmt"
func main() {
rect := rectangle{12.3, 21.45}
a := rect.area()
fmt.Println(a)
}
type rectangle struct {

length float32
width float32
}
func (r rectangle) area() float32 {
return r.length * r.width
}
func (r rectangle) perimeter() float32 {
return (2 * r.length) + (2 * r.width)
}
```
