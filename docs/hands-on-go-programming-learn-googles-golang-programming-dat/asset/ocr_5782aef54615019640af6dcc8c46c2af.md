```go
ch := make(chan int, 5)
done := make(chan bool)
go producer(ch, done)
go consumer(ch)
<-done
}
```
```

```
```go
func producer(ch chan int, done chan bool) {
for index := 0; index < 10; index++ {
fmt.Printf("PRODUCER: sending %v\n", index)
ch <- index
}

close(ch)
done <- true
}

func consumer(ch chan int) {
for val := range ch {
fmt.Printf("CONSUMER: read %v\n", val)
}
}
```
```

The following will be the output for the above program:

```
```text
PRODUCER: sending 0
PRODUCER: sending 1
PRODUCER: sending 2
PRODUCER: sending 3
