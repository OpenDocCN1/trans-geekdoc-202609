```solidity
pragma solidity ^0.5.3;
contract Wallet {
    string name = "RatterCoin";
    uint256 public balance;
    mapping(address => uint256) public balances;
    constructor() {
        name = "RatterCoin"; // Please feel free to change it
        balances[msg.sender] = 1000; // Please feel free to change the value
    }
    function transfer(address _to, uint256 _amount) public returns (bool) {
        // key-value pair check if sender has enough balance
        if (balances[msg.sender] >= _amount) {
            balances[msg.sender] -= _amount;
            balances[_to] += _amount;
            return true;
        } else {
            return false;
        }
    }
    function deposit(uint256 _amount) public returns (bool) {
        if (_amount > 0) {
            balances[msg.sender] += _amount;
            return true;
        }
        return false;
    }
    function balance(address _addr) public view returns (uint256) {
        return balances[_addr];
    }
}
```
