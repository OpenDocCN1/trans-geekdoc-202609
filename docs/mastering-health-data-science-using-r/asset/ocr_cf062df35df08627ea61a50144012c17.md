#' Find unique elements of a vector
#'
#' @param x vector
#' @return new vector with duplicates removed
own_unique <- function(x){
  # check for empty vector
  if (length(x) == 0){
    return(NULL)
  }

  # otherwise y will be unique values
  y <- c()

  for(i in 1:length(x)){
    # if value in x is not in y, we add it
    if(!(x[i] %in% y)){
      y <- c(y, x[i])
    }
  }

  return(y)
}
