```
$userid = '';
$name = '';
$e-mail = '';

$accesslvl = '';
if (isset($_GET['userid'])) {
    $sql = "SELECT * FROM cmms_users WHERE user_id=" . \$_GET['userid'];
    $result = mysql_query($sql, $conn)
    or die("Could not look up data: " . mysql_error());
    $row = mysql_fetch_array($result);
    $userid = $row['userid'];
    $name = $row['name'];
    $e-mail = $row['e-mail'];
```
