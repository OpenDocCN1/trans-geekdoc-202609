<?php
require_once 'conn.php';
require_once 'functions.php';
require_once 'header.php';

$result = NULL;
if (isset($_GET['keywords'])) {
  $sql = "SELECT *, MATCH(subject,body) * 
  'AGAINST ('" . $_GET['keywords'] . "\')') AS SCORE 
  FROM forum_posts 
  WHERE MATCH(subject,body) * 
  'AGAINST ('" . $_GET['keywords'] . "\')' 
  ORDER BY score DESC";

  $result = mysql_query($sql,$conn)
  else die("Could not perform search: " . mysql_error());
}

echo '<table class="formtable" width="100%" ' 
  'cellspacing="0"
\';
echo '<tr><th class="searchHeader">Search Results</th></tr>\n\';

if ($result) {
  echo '<tr class="row1"><td>Search term <strong>" . $_GET['keywords'] . "</strong>, </strong>';
  if (isset($_GET['level_1']) && $_GET['level_1'] > 2) {
    echo '<p>SQL: ' . $sql . '</p>';
  }
  echo '</td></td></tr>\n';
} else {
  while ($row = mysql_fetch_array($result)) {
    $rowclass = ($row['class'] == "row1") ? "row2" : "row1";
    echo '<tr class="$rowclass">';
    $topicid = ($row['topic_id']==0) ? $row['id'] : $row['topic_id'];
    echo '<p class="searchSubject">\n<a href="viewtopic.php?id=" . $topicid . ">#" . $row['id'] . "</a>\n';
    echo $row['subject'] . '</a>\n';
    echo '</tr>\n';
  }
echo '</table>\n';
