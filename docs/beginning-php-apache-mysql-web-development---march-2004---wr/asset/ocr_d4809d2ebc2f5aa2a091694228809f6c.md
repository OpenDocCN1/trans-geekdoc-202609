function getFormID($id) {
$sql = "SELECT forum_name as name, forum_desc as description, " .
"forum_moderator as mod " .
"FROM forum_forum " .
"WHERE id = " . $id . ";"
$result = mysql_query($sql)
or die (mysql_error()) . "<br>" . $sql);
$row = mysql_fetch_array($result);
return $row;
}
