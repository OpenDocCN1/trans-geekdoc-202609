"ORDER BY date_published DESC";
$result = mysql_query($conn,$sql)
or die("Could not get list of pending articles: " . mysql_error($conn));

if (mysql_num_rows($result) == 0) {
    echo "<li>没有可用的发布文章</li>";
} else {
    while ($row = mysql_fetch_array($result)) {
        echo "<tr>\n";
