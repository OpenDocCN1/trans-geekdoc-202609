<?php
$conn = mysqli_connect(SQL_HOST, SQL_USER, SQL_PASS)
or die('Could not connect to MySQL database. ' . mysql_error());

mysqli_select_db(SQL_DB, $conn);

if (isset($_GET['u'])) {
    $uid = $_GET['u'];
    $sql = "SELECT * FROM ml_users WHERE user_id = '$uid'";
    $result = mysqli_query($sql)
    or die('Invalid query: ' . mysql_error());
    if (mysqli_num_rows($result)) {
        $row = mysqli_fetch_array($result);
        $msg = "<h2>" . $row['firstname'] . "</h2>" . "<br /></p><br />\*";
        $e-mail = $row['e-mail'];
    } else {
        die('No match for user id ' . $uid);
    }
}

if (isset($_GET['ml'])) {
    $ml_id = $_GET['ml'];
    $sql = "SELECT * FROM ml_lists WHERE ml_id = '$ml_id'";
    $result = mysqli_query($sql)
    or die('Invalid query: ' . mysql_error());
    if (mysqli_num_rows($result)) {
        $row = mysqli_fetch_array($result);
        $msg = "<li>Thank you for subscribing to the <i>" .
              $row['listname'] . "</li> mailing list. <br />";
        echo $msg;
    } else {
        die('Could not find Mailing List ' . $ml_id);
    }
} else {
    die('Mailing List is missing.');
}

if (isset($_GET['t'])) {
    switch ($_GET['t']) {
        case 'c':
            $msg = "A confirmation request has been sent " .
                  "to <b>" . $e-mail . "</b>.<br /></br />\*";
            break;
        case 's':
            $msg = "A subscription notification has been " .
                  "sent to <b>" . $e-mail . "</b>.<br /></br />\*";
    }
    $msg = "<a href='user.php?id='" . $uid . \'>" .
          "Return to Mailing List Signup page</a>";
}
?>
