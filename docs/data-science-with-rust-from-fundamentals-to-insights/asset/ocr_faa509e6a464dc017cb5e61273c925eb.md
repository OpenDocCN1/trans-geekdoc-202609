fn pca(data: &Array2<f64>, num_components: usize) -> Array2<f64>
{
    let mean = data.mean_axis(Axis(0)).unwrap();
    let centered_data = data - &mean;

    let svd = centered_data.svd(true, true).unwrap();
    let u = svd.0.unwrap();

    u.slice(s![.., ..num_components]).to_owned()
}
