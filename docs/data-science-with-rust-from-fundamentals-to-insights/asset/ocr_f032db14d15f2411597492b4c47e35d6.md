# 1. Aggregating Data

```
```rust
use polars::prelude::*;

fn main() -> Result<()> {
    let df = df! {
        "Name" => &["Alice", "Bob", "Charlie", "Alice"],
        "Age" => &[25, 30, 35, 25],
        "Salary" => &[50000, 60000, 70000, 55000]
    }?;

    let grouped_df = df.groupby("Name")?
        .select("Salary")
        .mean()?;
    println!("{:?}", grouped_df);
    Ok(())
}
