        .caption("Patient Age Distribution", ("sans-serif", 50).into_font())

        .x_label_area_size(30)
        .y_label_area_size(30)
        .build_cartesian_2d(0..90, 0..10)?;

    chart.configure_mesh().draw()?;

    chart.draw_series(

            .style(GREEN.filled())
            .data(age_data.iter().map(|&x| (x, 1)))
    )?;

    Ok(())
}
