# Example: Setting Up a Simple Distributed System with Rust

```
```rust
extern crate tokio; extern crate hyper;

use tokio::net::TcpListener;
use hyper::service::{make_service_fn, service_fn};
use hyper::{Body, Request, Response, Server};
use std::sync::{Arc, Mutex};
use std::collections::HashMap;
```
```

```
```rust
#[tokio::main]
async fn main() {
    let data = Arc::new(Mutex::new(HashMap::new()));

    let make_svc = make_service_fn(|_conn| {
        let data = Arc::clone(&data);
        async move {
            Ok::<_, hyper::Error>(service_fn(move |req: Request<Body>| {
                handle_request(req, Arc::clone(&data))
            }))
        }
    });
```
```

```
```rust
let addrr = ([127, 0, 0, 1], 3000).into();
let server = Server::bind(&addrr).serve(make_svc);

println!("Running the server at http://{{}}", addrr);
if let Err(e) = server.await {
    eprintln!("Server error: {{}}", e);
}
