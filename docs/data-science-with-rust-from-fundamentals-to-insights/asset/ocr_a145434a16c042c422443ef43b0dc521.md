fn main() -> Result<(), Box<dyn std::error::Error>> {
    let drawing_area = BitMapBackend::new("line_chart.png", (800,

    drawing_area.fill(&WHITE)?;

        .y_label_area_size(30)

        (0.0..10.0).map(|x| (x, x * x)),
        &RED,
    ))?;

    Ok(())
}
'''
