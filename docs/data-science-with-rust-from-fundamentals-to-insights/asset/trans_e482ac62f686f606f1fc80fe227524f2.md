async fn load_data(pool: &sqlx::PgPool, record: Record) -> Result<(), Box<dyn Error>> {
    sqlx::query("INSERT INTO records (id, name, age, email) VALUES ($1, $2, $3, $4)")
        .bind(record.id)
        .bind(record.name)
        .bind(record.age)
        .bind(record.email)
        .execute(pool)
        .await?;

    Ok(())
}
