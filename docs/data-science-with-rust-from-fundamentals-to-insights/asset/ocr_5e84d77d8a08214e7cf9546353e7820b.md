fn main() {

        [1., 2.],

        [1., 4.],
        [1., 5.]];
    let y = array![[3.], [6.], [7.], [8.], [11.]];
    let result = x.least_squares(&y).unwrap();
    let coefficients = result.solution;
    println!("Coefficients: {:?}", coefficients);
}
