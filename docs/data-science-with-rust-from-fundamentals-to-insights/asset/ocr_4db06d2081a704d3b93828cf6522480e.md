```
\#[derive(Debug, Deserialize)]
struct Record {
    id: u32,
    name: String,
    value: f64,
}
```

```
fn main() -> Result<(), Box<dyn Error>> {
    let file_path = "data/input.csv";
    let mut rdr = csv::Reader::from_path(file_path)?;
    for result in rdr.deserialize() {
        let record: Record = result?;
        println!("{:?}", record);
    }
    Ok(())
}
```
