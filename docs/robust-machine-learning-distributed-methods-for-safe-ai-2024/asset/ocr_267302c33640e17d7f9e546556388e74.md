Next, from Lemma 6.4, we obtain that

$$
\mathbb{E}\big[\mathcal{L}_H(\boldsymbol{\theta}_t)-\mathcal{L}_H(\boldsymbol{\theta}_{t-1})\big] \le -\gamma\Big(\frac12-2\gamma L\Big)\mathbb{E}\big[\|\nabla\mathcal{L}_H(\boldsymbol{\theta}_{t-1})\|^2\big]
\quad + \gamma\,(1+2\gamma L)\mathbb{E}\big[\|\boldsymbol{\xi}_{t-1}\|^2\big] + \gamma\,(1+\gamma L)\mathbb{E}\big[\|\boldsymbol{\delta}_{t-1}\|^2\big].\tag{6.28}
$$

Thus, substituting from (6.28) and (6.27) in (6.26), we obtain that

$$
\mathbb{E}[V_t-V_{t-1}] = 2\mathbb{E}[\mathcal{L}_H(\boldsymbol{\theta}_t)-\mathcal{L}_H(\boldsymbol{\theta}_{t-1})] + z\mathbb{E}[\|\boldsymbol{\xi}_t\|^2 - \|\boldsymbol{\xi}_{t-1}\|^2]
\le \Big(-2\gamma\Big(\frac12-2\gamma L\Big)+4zL\gamma(1+\gamma L)\beta^2\Big)\mathbb{E}[\|\nabla\mathcal{L}_H(\boldsymbol{\theta}_{t-1})\|^2]
\quad +\Big(2\gamma(1+2\gamma L)+z(1+\gamma L)(1+4\gamma L)\beta^2 -z\Big)\mathbb{E}[\|\boldsymbol{\xi}_{t-1}\|^2]
\quad +\Big(2\gamma(1+\gamma L)+z2L\gamma(1+\gamma L)\beta^2\Big)\mathbb{E}[\|\boldsymbol{\delta}_{t-1}\|^2] + z\frac{(1-\beta)^2\sigma^2}{b(n-f)}.
$$

Let us denote

$$
A := 2\Big(\frac12-2\gamma L\Big)-4zL(1+\gamma L)\beta^2,\tag{6.29}
\nB := 2\gamma(1+2\gamma L)+z(1+\gamma L)(1+4\gamma L)\beta^2 -z,\text{ and}\tag{6.30}
\nC := 2\gamma(1+\gamma L)+z2L\gamma(1+\gamma L)\beta^2.\tag{6.31}
$$

Thus, we can rewrite the above inequality as

$$
\mathbb{E}[V_t-V_{t-1}] \le -\gamma A\,\mathbb{E}[\|\nabla\mathcal{L}_H(\boldsymbol{\theta}_{t-1})\|^2]
\quad + B\,\mathbb{E}[\|\boldsymbol{\xi}_{t-1}\|^2] + C\,\mathbb{E}[\|\boldsymbol{\delta}_{t-1}\|^2]
\quad + z(1-\beta)^2\frac{\sigma^2}{b(n-f)}.\tag{6.32}
\n\text{Next, we separately analyze }A,B,\text{ and }C.\n\text{Term }A.\text{ Upon substituting }z=\frac1{8L}\text{ in }(6.29),\text{ and recalling that }\gamma\le\frac1{24L}\text{ and }\beta^2<1,\text{ we obtain that}
\nA \ge 1-4\gamma L-\frac{4L}{8L}(1+\gamma L)\ge\frac12-\frac{9\gamma L}{2}\ge\frac12-\frac3{16}\ge\frac14.\tag{6.33}
$$
