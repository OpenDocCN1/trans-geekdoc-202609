```

$posts = array();
foreach ($rows as $row) {
    // the post entity ID on this row
    $post_id = $row['posts__post_id'];
    
    // does the post entity already exist in the $posts collection?
    if (!isset($posts[$post_id])) {
        // create a new post entity
        $post = new Post();
        $post->post_id = $row['posts__post_id'];
        $post->author_id = $row['posts__author_id'];
        $post->datetime = $row['posts__datetime'];
```
