func New(less func(interface{}, interface{})) bool) *Map
New returns an empty Map that uses the given less function to compare keys.
For example:

type Point { X, Y int }
pointMap := omap.New(func(a, b interface{}) bool {
	a, b := a.(Point), b.(Point)
	if a.X < b.X {
		return a.X < b.X
		return a.Y < b.Y
	})
))
