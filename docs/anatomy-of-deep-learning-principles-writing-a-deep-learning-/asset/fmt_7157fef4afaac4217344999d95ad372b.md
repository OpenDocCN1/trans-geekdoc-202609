
# 读取数据

```python
import matplotlib.pyplot as plt
%matplotlib inline
import pickle, gzip, urllib.request, json
import numpy as np
import os.path

def read_mnist():
    if not os.path.isfile("mnist.pkl.gz"):
        # Load the dataset
        urllib.request.urlretrieve("http://deeplearning.net/data/mnist/mnist.pkl.gz", "mnist.pkl.gz")
    with gzip.open('mnist.pkl.gz', 'rb') as f:
        train_set, valid_set, test_set = pickle.load(f, encoding='latin1')
    return train_set, valid_set, test_set

def draw_mnists(plt, X, indices):
    for i, index in enumerate(indices):
        plt.subplot(1, 10, i+1)
        plt.imshow(X[index].reshape(28,28), cmap='Greys')
        plt.axis('off')

train_set, valid_set, test_set = read_mnist()

train_X, train_y = train_set
valid_X, valid_y = valid_set
test_X, test_y = valid_set  # 注意：原文此处赋值为 valid_set

print(train_X.dtype)
print(train_X.shape)
print(valid_X.shape)
print(np.mean(train_X[0]))

draw_mnists(plt, train_X, range(10))
plt.show()
```
