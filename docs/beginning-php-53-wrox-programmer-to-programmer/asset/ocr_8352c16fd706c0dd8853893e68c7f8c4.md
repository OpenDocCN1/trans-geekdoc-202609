```
function deleteMember() {
    $member = new Member(array(
        "id" => isset($_POST['memberId']) ? (int) $_POST['memberId'] : "",
    ));
    LogEntry::deleteAllMember($member->getValue("id"));
    $member->displaySuccess();
}
```

```
function clearLog() {
    $id = isset($_POST['memberId']) ? (int) $_POST['memberId'] : "";
    LogEntry::deleteAllMember($id);
    displaySuccess();
}
```
