; Default password for mysqli_connect() (doesn't apply in safe mode);

```
; Maximum time (in seconds) for connection timeout. -1 means no limit
; http://php.net/mysql.connect-timeout
ysql.connect_timeout = 60
```

```
; Trace mode. When trace mode is active (=On), warnings for table/index scans and
; SQL-Errors will be displayed.
; http://php.net/mysql.trace-mode
ysql.trace_mode = Off
```

```
; Maximum number of persistent links. -1 means no limit.
; http://php.net/mysql.max-persistent

```
; Maximum number of links. -1 means no limit.
; http://php.net/mysql.max-links
ysql.max_links = -1

```
; If mysqli is used: Number of cache slots for the internal result set cache
; http://php.net/mysqli.cache_size
ysql.cache_size = 2000

; http://php.net/mysql.default-port
ysql.default_port = 3360
```
