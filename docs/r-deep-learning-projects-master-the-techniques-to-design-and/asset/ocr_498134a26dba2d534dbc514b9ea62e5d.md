```
library(readr)
library(stringr)

library(tokenizers)

set.seed(1234)

zeros_like <- function(M) {
  return(matrix(0,dim(as.matrix(M))[1],dim(as.matrix(M))[2]))
}

softmax <- function(x) {
  xt <- exp(x-max(x))
  return(xt/sum(xt))
}
```
