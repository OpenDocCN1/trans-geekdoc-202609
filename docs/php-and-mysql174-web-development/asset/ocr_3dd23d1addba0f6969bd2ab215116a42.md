public function DisplayMenu($buttons)
{
  echo "<!-- menu -->";
  <nav>
    while (list($name, $url) = each($buttons)) {
      $this->DisplayButton($name, $url,
        $this->IsUrlCurrentPage($url));
    }
echo "</nav>";
}
```

```
public function IsUrlCurrentPage($url)
{
  if (strpos($_SERVER['PHP_SELF'],$url)==false) {
    return false;
  } else {
    return true;
  }
}
