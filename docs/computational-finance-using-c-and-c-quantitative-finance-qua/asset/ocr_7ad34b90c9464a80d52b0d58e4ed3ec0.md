```
using System;
namespace ComputationalLib
{
    public class GenericEquityBasketOptionMaxPut : GenericEquityBasketOption
    {
        public override string Name()
        {
            string temp_string = "**";
            temp_string = "Generic option: Put on maximum maximum * + n_.ToString() + " *assets";
            return temp_string;
        }

        public override double Payoff(// // implement max, put)
            double the_max = 0.0;
            double pay_val = 0.0;
            double zero = 0.0;
            for (int j = 0; j < n_; j++)
            {
                if (ST_[j] > the_max) the_max = ST_[j];
            }
            pay_val = Math.Max(Strike - the_max, zero);

            return pay_val;
        }
    }
```
