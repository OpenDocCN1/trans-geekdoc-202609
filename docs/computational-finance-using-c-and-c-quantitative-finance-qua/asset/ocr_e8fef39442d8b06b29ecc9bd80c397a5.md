private double MonteCarlo(bool is_put)
{
    int seed = 11;
    double[] asset_path = new double[fSteps];
    double time_step = fTimeToExpiry / fSteps;
    double sqrt_time_step = System.Math.Sqrt(time_step);
    double disc = System.Math.Exp(-fRiskRate * fTimeToExpiry);

    set_seed(seed);
    double opt_val = 0.0;
    bool not_out = true;
    int k = 0;
    double STN = 0.0;
    double mean = (fRiskRate + fDividendYield * fSignal * fSignal * 0.5) * time_step;
    double std = System.Math.Sqrt(fSignal * fSignal * time_step);
    double z2 = 0.0;

    for (int i = 0; i < fnumScenarios; i++)
    {
        // generate the asset path
        double STN = ST1;
        not_out = true;
        k = 0;

        while (not_out && k < fSteps)
        {
            z = RandNorm(mean, std);
            STN = STN * System.Math.Exp(z);
            if (STN < fBarrierLevel) not_out = false;
            asset_path[k] = STN;
            k++;
        }

        if (is_put)
        {
            opt_val = System.Math.Max(fStrike, STN, 0.0);
        }
        else
        {
            opt_val = System.Math.Max(fStrike, STN, 0.0);
        }
        if (not_out)
        {
            // only if asset value is above the barrier level
            // compute the probability that the asset remained above the barrier
            // (UseBrownianBridge)
            double total_probability_above = 1.0;
            double sigma_2 = fSignal * fSignal;
            double log_barrier_level = System.Math.Log(fBarrierLevel);
            for (int j = 0; j < fSteps - 1; j++)
            {
                double log_S_i = System.Math.Log(asset_path[j]);
                double log_S_iplus1 = System.Math.Log(asset_path[j + 1]);
                fac = 2.0 * (log_barrier_level - log_S_i) / (sigma_2 * time_step);
                pr = (1.0 - System.Math.Exp(-fac)); // probability of staying above the barrier between i and i+1
                total_probability_above *= pr;
            }
        }
    }}
