sumit_max_put = zero;
sumit_max_call = zero;
sumit_min_put = zero;
sumit_min_call = zero;

tol = 1.0e-8;

srand(seed);
if_call = 1; /* intisation call to the random number generator */
multivariate_normal(1,fMEANS(1),3,C3(1),3,tol,&rvec,&Z(1),iflag);

num_simulations = 6000;
if_call = 0;
for (int i = 0; i < num_simulations; i++) {

    /* continuation call to the random number generator */
    multivariate_normal(1,fMEANS(1),3,C3(1),3,tol,&rvec,&Z(1),iflag);

ST1 = exp(Z(1));
ST2 = exp(Z(2));
ST3 = exp(Z(3));
}
// options on the maximum
tmp2 = MAX(ST1,ST2);
tmp_max = MAX(tmp2,ST3);
tmpl = max-X;
opt_val = MAX(tmpl, zero);
sumit_max_call += opt_val*disc;

tmpl = X-min;
opt_val = MAX(tmpl, zero);
sumit_min_put += opt_val*disc;
//// options on the minimum
tmp2 = MIN(ST1,ST2);
tmp_min = MIN(tmp2,ST3);
tmpl = min-X;
opt_val = MAX(tmpl, zero);
sumit_min_call += opt_val*disc;

}

# Code example 6.1 (Continued).

If $S_t$ is the price of the asset at time $t$, and $S_{t+\Delta t}$ is the price at instant $t+\Delta t$, then we define the following equations:

$log(S_{t+\Delta t}) = log(S_t) + \epsilon_t, \quad \epsilon_t \sim N(\mu\Delta t, \sigma^2\Delta t)$

or equivalently

$$\log(\left(\frac{S_{t+\Delta t}}{S_t}\right) \sim N(\mu\Delta t, \sigma^2\Delta t)$

where $\epsilon_t$ represents a random variable and as usual $N(\mu\Delta t, \sigma^2\Delta t)$ denotes a Gaussian with mean $\mu\Delta t$ and variance $\sigma^2\Delta t$.
