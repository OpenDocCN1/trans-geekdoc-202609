h_{*,i} := argmin_{h_i} E [ (θ_i - θ̂_i)^2 ] = E [ (θ_i - h_i^T y)^2 ].

Σ[y] h[*,i] = p[i], i = 1, 2, …, l.

Σ[y] = E[y y^(T)] 和 p[i] = E[y θ[i]], i = 1, 2, .., l.
