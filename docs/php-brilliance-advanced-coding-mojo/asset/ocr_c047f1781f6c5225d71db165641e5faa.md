# Encapsulating our ideas.

```
class User {
    /**
     * @var UserDataTransport
     */
    private $userData;

    public function __construct(UserDataTransport $udt) {
        $this->userData = $udt;
    }

    private function getDataTransport() {
        return $this->userData;
    }

    public function getUsername() {
        return $this->getDataTransport()->getUsername();
    }

    public function getEmail() {
        return $this->getDataTransport()->getEmail();
    }

    public function setEmail($email) {
        try {
            $this->getDataTransport()->setEmail($email);
            return $this;
        } catch (DataTransportValidationException $e) {
            $msg = $e->getMessage();
            $code = $e->getCode();
            throw new ModelException($msg, $code);
        }
    }
}
```
