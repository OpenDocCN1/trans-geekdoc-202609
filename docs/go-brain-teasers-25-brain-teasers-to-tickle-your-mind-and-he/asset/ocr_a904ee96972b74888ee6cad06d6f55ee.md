iota values are numbers, in some cases you’d like your constants to have a human-readable representation. This is done by giving these constants a type and implementing fmt.Stringer interface for this type.

Listing 47. iota_str.go

```
package main

import (
	"fmt"
)

type FilePerm uint16 // 16 flags are enough

const (
	Read FilePerm = 1 << iota
	Write
	Execute
)

// String implements fmt.Stringer interface
func (p FilePerm) String() string {
	switch p {
	case Read:
		return "read"
	case Write:
		return "write"
	case Execute:
		return "execute"
	}

	return fmt.Sprintf("unknown FilePerm: %d", p) // don't
	use %s here :)
}
```
