To solve this problem you can use a sync/Mutex to ensure only one goroutine updates count at time.

# Listing 40. count_mu.go

```
package main

import (
	"fmt"
	"sync"
)

func main() {
	var count int
	var wg sync.WaitGroup
	var m sync.Mutex

	for i := 0; i < 1_000_000; i++ {
		wg.Add(1)
		go func() {
			m.Lock()
			defer m.Unlock()
			defer wg.Done()
			count++
		}()
	}
}
```
