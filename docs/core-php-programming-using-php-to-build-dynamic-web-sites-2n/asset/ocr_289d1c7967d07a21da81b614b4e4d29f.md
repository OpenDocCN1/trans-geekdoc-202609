<?
/*
 ** get ten random numbers,
 ** each greater than the next
 */
//seed random number generator
srand(time());

//init variables
$count = 0;
$max = 0;

//get ten random numbers
while($count < 10)
{
    $value = rand(1,100);

    //try again if $value is too small
    if($value < $max)
{
        continue;
    }
    $count++;
