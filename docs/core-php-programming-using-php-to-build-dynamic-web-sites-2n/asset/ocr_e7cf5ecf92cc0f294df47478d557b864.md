// set URL to fetch
$URL = 'http://www.php.net/';

// open file
$page = fopen($URL, "r");

print("Links at \$URL<BR>\n");
print("<UL>\n");
while(!feof($page)) {
    // get a line
    $line = fgets($page, 1024);

    // loop while there are still ULs present
    while(preg(\/HREF="[[^"]*\]*\*/, $line, $match)) {
        // print out URL
        print("<LI>\n");
        print($match[0]);
        print("<BR>\n");
    }
    // remove URL from line
    $replace = ereg_replace(".*", "*", $line);
    $line = ereg_replace(\$replace, "-", $line);

}
print("</UL>\n");

fclose($page); *
