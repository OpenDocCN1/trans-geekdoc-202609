# Listing 18.6 发送附件

```php
<?php
/**
** 函数: makeAttachment
** 输入：ARRAY 附件
** 输出：STRING
** 描述：返回一个附件的标头
** 和数据。它期望接收一个包含
** type、name 和 content 元素的数组。附件被简单地
** 进行 base64 编码，即使在不需要时也是如此。
*/
function makeAttachment($attachment) 
{
    //发送内容类型
    $headers = "Content-Type: " . $attachment["type"];

    if($attachment["name"] != "") {
        $headers = "; name=\"" . $attachment["name"] . "\"";
    }

    $headers .= "\r\n";

    $headers .= "Content-Transfer-Encoding: base64\r\n";

    $headers .= "\r\n";

    $headers .= chunk_split(base64_encode($attachment["content"]));
    $headers .= "\r\n";

    return($headers);
}

/**
** 函数: mailAttachment
** 输入：STRING 目标，STRING 主题，ARRAY 附件
** 输出：无
** 描述：通过电子邮件发送附件。附件
** 数组是一个二维数组。每个元素都是一个
** 关联数组，包含 type、name 和 content 元素。
```

# 验证地址

```php
/**
** 函数: verifyEmail
** 输入：address, error
** 输出：BOOLEAN
** 描述：尝试通过联系
** 邮件交换器来验证地址，
** 然后是精确域名本身。错误参数
** 在无法验证时包含相关文本。
 */

function verifyEmail($address, &$error)
{
    global $SERVER_NAME;

    list($user, $domain) = explode("@", $address, 2);

    //确保域名有邮件交换器
    if(!checkdnsrr($domain, "MX")) {
        $error = "未找到域名的 MX 记录";
        return FALSE;
    }

    $fp = fsockopen($domain, 25, $errno, $errstr, 10);
    if(!$fp) {
        $error = "无法连接到邮件服务器";
        return FALSE;
    }

    $stopTime = time() + 10;
    $gotResponse = FALSE;

    while (TRUE)
    {
        //尝试从邮件服务器获取行
        $line = fgets($fp, 1024);

        if(substr($line, 0, 3) == "220")
        {
            //重置计时
            $stopTime = time() + 10;
            $gotResponse = TRUE;
        } elseif(($line == "") AND ($gotResponse))
        {
            break;
        } elseif(time() > $stopTime)
        {
            break;
        }
    }

    if(!$gotResponse)
    {
        $error = "服务器没有响应";
        fclose($fp);
        return FALSE;
    }

    set_socket_blocking($fp, TRUE);

    //登录
    fputs($fp, "HELO $SERVER_NAME\r\n");
    gets($fp, 1024);

    //设置发
    fputs($fp, "MAIL FROM: <info@domain>\r\n");
    gets($fp, 1024);

    //尝试地址
    fputs($fp, "RCPT TO: <$address>\r\n");
    $line = fgets($fp, 1024);

    if(substr($line, 0, 3) == "250") {
        fclose($fp);
        return TRUE;
    } else {
        $error = "收件者拒绝";
        fclose($fp);
        return FALSE;
    }
}
```
