// create test data
$paragraph = "This is a <B>short</B> paragraph. Some ";
$paragraph = "<B>words</B> and <B>some phrases</B> ";
$paragraph .= " are surrounded by <b>bold</B> tags.";

/*
** use PREG_MATCH_ORDER to find bold words
*/
preg_match_all('/<([^>]+)>(.*?)<\/\1>|.*/', $paragraph, $match, PREG_SET_ORDER);
// print all results
print("<B>Subpattern matches</B>:<BR\n");
for($i=0; $i < count($match[0]); $i++) {
    print(htmlspecialchars($match[0][$i]) . " <BR>\n");
}

print("<B>Subpattern matches</B>:<BR\n");
for($i=0; $i < count($match[1]); $i++) {
    print(htmlspecialchars($match[0][$i]) . " <BR>\n");
}

/*
** use PREG_SET_ORDER to find bold words
*/
preg_match_all('/<([^>]+)>(.*?)<\/\1>|.*/', $paragraph, $match, PREG_SET_ORDER);
foreach($match as $m) {
    print(htmlspecialchars($m[0]));
    for($i=0; $i < count($m); $i++) {
        print(".htmlspecialchars($m[$i])" . ".");
    }
}
print("<BR>\n");
