minmax.normalize<-function(ds, scaler=NULL) {
if(is.null(scaler)) {
for(f in ds$xFeatures) {
scaler[[f]]$minval<-min(ds$data[, f])
scaler[[f]]$maxval<-max(ds$data[, f])
ds$data[, f] <- (ds$data[, f] -
scaler[[f]]$minval) / (scaler[[f]]$maxval-scaler[[f]]$minval)
}
ds$scaler <- scaler
} else {
for (f in ds$xFeatures) {
ds$data[, f] <- (ds$data[, f] -
scaler[[f]]$minval) / (scaler[[f]]$maxval-scaler[[f]]$minval)
}
}
return(ds)
}
