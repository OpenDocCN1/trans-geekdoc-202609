$result = mysql_query("SELECT * FROM flashStore_products
WHERE active=1", $link);

$xml = "<products>\n";

while($row = mysql_fetch_array($result))
{
	$xml .= "<product id=\"" . $row['id'] .
		"\">\n" . "<name=\"" . $row['name'] .
		"\">\n" . "<desc=\"" . $row['description'] .
		"\">\n" . "<thumbnail=\"" . $row['thumbnail'] . "\" />\n";
		"\" price=\"" . $row['price'] . "\"\n" />\n";
}

$xml .= "</products>";

return $xml;
