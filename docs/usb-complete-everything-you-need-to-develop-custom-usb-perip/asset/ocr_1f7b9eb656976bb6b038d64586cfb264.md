    case WAIT_OBJECT_0: {
        // Success;

        break;
    }
    case WAIT_TIMEOUT: {
        // Timeout error;
        //Cancel the read operation.
        CancelIo(ReadHandle);
        break;
    }
    default: {
        // Undefined error;
        //Cancel the read operation.
        CancelIo(ReadHandle);
        break;

}
