```
<pre>
<?php
function sortNestedArrayAssoc($a) {
    ksort($a);
    foreach ($a as $key => $value) {
        if (is_array($value)) {
            sortNestedArrayAssoc($value);
        }
    }
}

$arr = array(
    'Roman' =>
        array('one' => 'I', 'two' => 'II', 'three' =>
        'III', 'four' => 'IV'),
    'Arabic' =>
        array('one' => '1', 'two' => '2', 'three' =>
        '3', 'four' => '4')
);
sortNestedArrayAssoc(&$arr);
print_r($arr);
?>
