    echo '<ul>';
    foreach ($xml->quote as $quote) {
        $year = htmlspecialchars($quote['year']);
        $phrase = htmlspecialchars($quote->phrase);
        $author = htmlspecialchars($quote->author);
        echo "<li>$author: \"$phrase\" ($year)</li>";
