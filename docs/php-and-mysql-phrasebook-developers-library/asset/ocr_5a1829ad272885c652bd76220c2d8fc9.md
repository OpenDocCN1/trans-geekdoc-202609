function sess_destroy($id) {
    mysqli_query(
        $GLOBALS['sess_mysqli'],
        sprintf('DELETE FROM sessiondata WHERE
$id=\'%s\'',
            mysqli_real_escape_string($GLOBALS
['sess_mysqli'], $id))
    );
    return true;
}

function sess_gc($timeout) {
    $timestamp = date('YmdHis', time() - $timeout);
    mysqli_query(
        $GLOBALS['sess_mysqli'],
        sprintf('DELETE FROM sessiondata WHERE access
< \'%s\'',
            $timestamp)
    );
}
