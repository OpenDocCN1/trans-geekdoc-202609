```python
def sigmoid(a):
    return 1 / (1 + np.exp(-a))

# derivative of activation function
def sigmoid_deriv(y):
    return y * (1 - y)

class NeuralNetwork:
    def __init__(self, nin, nout, children, epochs, learning_rate):
        self.nin = nin # number of input neurons
        self.nout = nout # number of output neurons
        self.children = children # number of hidden neurons
        self.epochs = epochs # number of epochs
        self.learning_rate = learning_rate
        self.weights = []
        self.biases = []
        for i in range(self.children):
            # We et al. (2015) initialization
            self.weights.append(np.random.randn(self.nin, self.children)) # i is the bias neuron
            self.biases.append(np.zeros(self.children))
        for i in range(self.nout):
            self.weights.append(np.random.randn(self.children, self.nout))
            self.biases.append(np.zeros(self.nout))

    def forward(self, x):
        # Blend bias with weights
        h1 = np.dot(x, self.weights[0]) + self.biases[0]
        a1 = sigmoid(h1)
        # Blend bias with weights
        h2 = np.dot(a1, self.weights[1]) + self.biases[1]
        a2 = sigmoid(h2)
        return a2

    def train(self, x, y):
        # train the data.
        perm = np.random.permutation(x.shape[0])
        for epoch in range(self.epochs):
            for i in perm:
                # Blend bias with weights, apply activation function.
                h1 = np.dot(x[i], self.weights[0]) + self.biases[0]
                a1 = sigmoid(h1)
                # Blend bias with weights
                h2 = np.dot(a1, self.weights[1]) + self.biases[1]
                a2 = sigmoid(h2)
                # Errors for all training samples.
                err = a2 - y[i]
                # Errors for all samples.
                cost = 0.5 * np.sum(err**2)
                # First multiplication component, second matrix
                d2 = err * sigmoid_deriv(a2)
                d1 = np.dot(d2, self.weights[1].T)
                # Update weights in second layer
                self.weights[1] -= self.learning_rate * np.dot(a1.T, d2)
                self.biases[1] -= self.learning_rate * np.sum(d2, axis=0)
                # Update weights in first layer
                self.weights[0] -= self.learning_rate * np.dot(x.T, d1)
                self.biases[0] -= self.learning_rate * np.sum(d1, axis=0)
        return epoch
```

The elements of $m^{th}$ hidden column are the weights of synapses connecting the $m^{th}$ hidden neuron to the output. Since the activations are clearly a linear combination of the columns and the bias, the columns can be seen as the basis functions of the model from which dataset is generated. The bias is a shift in the data space, such a shift is unnecessary and the bias is zero. The application of the transfer function ensures that the output lies again in the same range as the input.
