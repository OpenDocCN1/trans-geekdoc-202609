function [idx,MU,SIGMA] = InferBaysianMixture(V, K,alpha,m, ...
    lambda,Psi,nu,iter)
  # Input:
  V: data,
  K: number of clusters
  alpha: prior concentration parameter,
  m: prior location vector,
  lambda: prior mean fraction,
  Psi: prior inverse scale matrix,
  prior degrees of freedom,
  iter: number of iterations
  idx: index vector of cluster assignments,
  MU: array of the means sampled from the posterior normal-inverse-Wishart distributions for each cluster,
  SIGMA: array of the covariances sampled from the posterior normal-inverse-Wishart distributions for each cluster.

N = size(V,1);
# Start with random cluster assignments.
idx = randi(1,N);
# For each cluster its size, prior location vector,
# prior mean fraction, posterior inverse scale matrix,
# posterior degrees of freedom.
nk = randi(1,K);
LAMBDA = randi(2,K);
PSI = randi(2,K);
# For each cluster initialize a draw from the posterior normal-
# Inverse-Wishart distribution.
STOMA = randi(2,K);
MU = randi(2,K);
# For each cluster initialize these for the initial random cluster
# assignments and draw MU and SIGMA from that posterior-
# distribution.
for k=1:K
  Vk = V(idx==k,:);
  nk = size(Vk,1);
  sampleM = mean(Vk);
  sampleC = inv(Vk) + cov(Vk);
  M(k) = (lambda + nk*(sampleM(:)-m(:))) / (lambda + nk);
  LAMBDA(k) = lambda + nk;
  PSI(k) = sampleC + lambda + nk*(sampleM(:)-m(:))'/(lambda + nk);
  # Cholesky factorization to maintain positive definiteness.
  PSI(k) = chol(PSI(k));
  NU(k) = nk;
  SIGMA(:,k) = iwishrnd(PSI(:,k)',k*PSI(:,(:,k),NU(k));
  MU(:,k) = mvnrnd(M(k),SIGMA(:,k)/LAMBDA(k));
end
