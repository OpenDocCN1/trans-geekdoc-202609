target_mean = [2 3];
target_variance = [1 1.5; 1.5 3];
# Create function to unnormalized probability density.
target = @(x) exp(-(x-target_mean)*target_variance*(x-target_mean)');

# The variances of the components are given by the eigenvalues
eig = eig(target_variance);
figure
hold on
# Set variance of the proposal distribution to the identity matrix
times the smallest variance of the target distribution.
proposal_variance = eye(2)*min(eig);

N = 300;
# Set first element of the Markov chain.
start = target_mean;
sample = target_mean;
# Evaluate the target distribution for the last element.
 = feval(target,start);
counter = 1;
accepted = 0;
while total < N
    # Generate candidate from normal distribution with mean
candidate = normrnd(sample,proposal_variance);
    # Evaluate the target distribution for the candidate.
    new = feval(target,candidate);
    # Evaluate acceptance probability.
    acc = min(1,new);
    if acc > rand
        # Draw random walk.
        line([sample(accepted,1) candidate(1)],[sample(accepted,2) candidate(2)],'o');
        # Update Markov chain.
        sample = candidate;
        counter = counter + 1;
        accepted = accepted + 1;
        # Update evaluation of target distribution for the last element.
        last = new;
else
    # Reject candidate.
    plot(candidate(1), candidate(2),''*');
    # Update counter for this state.
    counter(accepted) = counter(accepted)+1;
end
total = total +1;
# Calculate acceptance rate.
rate = accepted/total
axis
