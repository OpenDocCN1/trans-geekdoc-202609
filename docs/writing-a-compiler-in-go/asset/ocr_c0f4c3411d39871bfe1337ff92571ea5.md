var Builtins = []struct {

&Builtin{Fn: func(args ...Object) Object {
if len(args) != 1 {

}
if args[0].Type() != ARRAY_OBJ {

arr := args[0].(*Array)
length := len(arr.Elements)
if length > 0 {
newElements := make([]Object, length-1, length-1)
copy(newElements, arr.Elements[1:length])
return &Array{Elements: newElements}
}

return nil
},
},
