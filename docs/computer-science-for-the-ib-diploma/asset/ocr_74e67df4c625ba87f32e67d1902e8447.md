public static void content(String text) {
   String value = "";
   int pos = 0;
   while (pos < text.length()) {
      String letter = text.substring(pos, pos + 1);
      if (letter.equals(".")) {
         System.out.println(value);
         value = "";
      } else {
         value += letter;
      }
      pos++;
   }
   if (!value.isEmpty()) {
      System.out.println(value);
   }
}
