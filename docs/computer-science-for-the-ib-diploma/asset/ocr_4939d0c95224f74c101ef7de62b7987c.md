```
Java
int numbers[] = {9,7,2,3};
int min, minIndex;
for(int i = 0; i<numbers.length; i++) {
    min = numbers[i];
    minIndex = i;
    for(int j = i+1; j<numbers.length; j++) {
        if(numbers[j]<min) {
            min = numbers[j];
            minIndex=j;
        }
    }
    numbers[minIndex] = numbers[i];
    numbers[i] = min;
}
for(int i = 0; i<numbers.length; i++) {
    System.out.print(numbers[i] + " ");
}
```
