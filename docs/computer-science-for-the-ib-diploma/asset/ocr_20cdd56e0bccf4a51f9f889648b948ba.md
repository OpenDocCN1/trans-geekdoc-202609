# Java

```java
int numbers[] = {3,4,5,7,9,11,13};
int key = 11;
boolean found = false;
int lower = 0;
int mid = 0;
int upper = numbers.length;
while((lower<=upper) && !found ) {
    mid = (lower+upper)/2;
    if (numbers[mid]==key) {
        found = true;
    } else if (numbers[mid]<key) {
        lower = mid+1;
    } else {
        upper = mid-1;
    }
}
if (found) {
    System.out.println("the value was found on position " + mid);
} else {
    System.out.println("the value was not found" );
}
```
