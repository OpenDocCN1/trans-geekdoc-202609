    }

    public void borrow() {
        if (!isBorrowed) {
            isBorrowed = true;
            System.out.println(title + " has been borrowed.");
        } else {
            System.out.println(title + " is already borrowed.");
        }
    }

    }

    }

        return title + " - " + (isBorrowed ? "Borrowed" : "Available");
    }
}
