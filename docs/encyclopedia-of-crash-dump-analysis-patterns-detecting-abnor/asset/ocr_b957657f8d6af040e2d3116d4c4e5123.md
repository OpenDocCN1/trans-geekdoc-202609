The modeling application source code:

```
class Exception
{
     int code;
     std::string description;
public:
     Exception(int _code, std::string _desc) : code(_code), description(_desc) {}
};

void bar()
{
     throw new Exception(5, "Access Denied");
}

void foo()
{
     bar();
}

int main(int argc, const char * argv[])
{
     foo();
     return 0;
}
```
