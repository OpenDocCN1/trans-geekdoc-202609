```
__try
{
  __try
  {
    *(int *)NULL = 0; // Exception 1
    // Dump1 1st chance
  }
  __except (EXCEPTION_EXECUTE_HANDLER)
  {
    std::cout << "Inner" << std::endl;
    *(int *)NULL = 0; // Exception 2
    // Dump2 1st chance
  }
}
__except (EXCEPTION_EXECUTE_HANDLER)
{
    std::cout << "Outer" << std::endl;
    *(int *)NULL = 0;   // Exception 3
                // Dump3 1st chance
                // Dump4 2nd chance
}
```
