\(\begin{aligned} P \left( |\sqrt{n}(\hat{\boldsymbol{\beta}}_{\mathcal{S}_1^c}^{\text{ridgeR}}(y_n) - \boldsymbol{\beta}_{\mathcal{S}_1^c}^*)| \ge a_n(\log(\log(n))) \right) \\ \le P \left\{ |Z| > \frac{a_n(\log(\log(n)))}{\sqrt{\text{Var}(\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n))}} - \frac{|\text{ADB}(\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n))|}{\sqrt{\text{Var}(\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n))}} \right\} \\ = 2\Phi \left\{ \frac{|\text{ADB}(\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n))| - a_n(\log(\log(n)))}{\sqrt{\text{Var}(\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n))}} \right\} \\ \le 2\Phi \left( \frac{-c_0\sqrt{\gamma_n}a_n^2}{\log(\log(n))} \right) \\ \le \exp \left( - \frac{c_0^2 \gamma_n a_n^2}{\log(\log(n))} \right) \end{aligned}\)

is the tail probability of the normal distribution. Thus, we have

\(\begin{aligned} P \left( j \notin S_1 : |\boldsymbol{\beta}_j^*| > a_n(\log(\log(n))) \subset \{ j \notin S_1 : |\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n)| > a_n \} \right) \\ \ge 1 - P \left( \cup \{ |\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n)| \le a_n \} \right) \\ \ge 1 - P \left( \cup \{ |\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n) - \boldsymbol{\beta}_j^*| \le a_n(\log(\log(n))) \} \right) \\ \ge 1 - p_n \exp \left\{ - \frac{c_0^2 \gamma_n a_n^2}{(\log(\log(n)))^2} \right\} \\ \ge 1 - \exp \left\{ - \frac{c_0^2 \gamma_n a_n^2}{(\log(\log(n)))^2} - \log(p_n) \right\} \\ \ge 1 - \exp \{ - (c_0 \log(\log(n-1))) \log(n \vee p_n) \} . \end{aligned}\)

For a large enough \(n\), there exists \(c_0 \log(\log(n-1)) > t > 0\) for some \(t\). Thus,

\(\begin{aligned} \lim_{n \to \infty} P \left( j \notin S_1 : |\boldsymbol{\beta}_j^*| > a_n(\log(\log(n))) \subset \{ j \notin S_1 : |\hat{\boldsymbol{\beta}}_j^{\text{ridgeR}}(y_n)| > a_n \} \right) \\ \ge 1 - (n \vee p_n)^{-t} \to 1. \end{aligned}\)
