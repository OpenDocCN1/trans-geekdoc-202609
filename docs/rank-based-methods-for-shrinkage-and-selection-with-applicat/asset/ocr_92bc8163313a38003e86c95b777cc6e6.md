\(\text{ADB}(\hat{\theta}_n^{SSR+}(\lambda)) = \begin{pmatrix} \mathbf{0}_{p_1} \\ -\delta_2^\top \mathbb{E} \left[ (1-(p_2-2)\chi_{p_2+2}^{-2}(\Delta^2))I(\chi_{p_2+2}^2(\Delta^2) < p_2-2) \right] \end{pmatrix}\)

\(\text{ADL}_2\text{-risk}(\hat{\theta}_n^{SSR+}(\lambda)) = \text{ADL}_2\text{-risk}(\hat{\theta}_n^{SSR}(\lambda)) - \eta^2 \text{tr}(\mathcal{K}_2) \mathbb{E} \left[ \left( 1-(p_2-2)\chi_{p_2+2}^{-2}(\Delta^2) \right)^2 I(\chi_{p_2+2}^2(\Delta^2) < p_2-2) \right] + \delta_2^\top \delta_2 \left\{ 2 \mathbb{E} \left[ (1-(p_2-2)\chi_{p_2+2}^{-2}(\Delta^2)) I(\chi_{p_2+2}^2(\Delta^2) < p_2-2) \right] - \mathbb{E} \left[ \left( 1-(p_2-2)\chi_{p_2+4}^{-2}(\Delta^2) \right)^2 I(\chi_{p_2+4}^2(\Delta^2) < p_2-2) \right] \right\}.\)
