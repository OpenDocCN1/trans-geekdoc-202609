Tn <- function(theta, y ) {
  n <- length(y)
  diff <- y-theta
  abs_diff <- abs(diff)
  sgn_diff <- sign(diff)
  rank_abs_diff <- rank(abs_diff,
                        ties.method = "first")
  an_plus <- sqrt(3.)*(rank_abs_diff)/(n+1.)
  Tnsum <- sum(sgn_diff*an_plus)/sqrt(n)
  return(Tnsum)
}
