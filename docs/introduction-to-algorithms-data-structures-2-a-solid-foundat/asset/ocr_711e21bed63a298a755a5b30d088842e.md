def remove(self, key):
    """
    Removes Node containing data that matches the key
    Returns the node or `None` if key doesn't exist
    Takes O(n) time
    """

    current = self.head
    previous = None
    found = False

    while current and not found:
        if current.data == key and current is self.head:
            found = True
            self.head = current.next_node
            self.__count -= 1
            return current
