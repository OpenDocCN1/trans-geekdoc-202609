where

$\Sigma_{t-1, t-2|T} = \Sigma_{t-1|t-1|T} \mathbf{J}_{t-2}^{\mathbf{T}} + \mathbf{J}_{t-1} (\Sigma_{t, t-1|T} - \mathbf{F} \Sigma_{t-1|t-1}) \mathbf{J}_{t-1}^{\mathbf{T}} \tag{29.94}$

where $\mathbf{J}_t$ is the backwards Kalman gain matrix defined in Equation (8.67). We initialize this using $\Sigma_{T, T-1|T} = (\mathbf{I} - \mathbf{K}_T \mathbf{H}) \mathbf{F} \Sigma_{T-1|T-1}$, where $\mathbf{K}_T$ is the forwards Kalman gain matrix defined in Equation (8.28).

We can derive the M step as follows, using standard matrix calculus. We denote $\mathbf{A}_{\text{out}} = [\mathbf{H}, \mathbf{D}]$, $\hat{\mathbf{x}}_{\text{out}, t} = [\hat{\mathbf{z}}_t^{\mathbf{T}}, \mathbf{u}_t^{\mathbf{T}}]^{\mathbf{T}}$, $\mathbf{A}_{\text{dyn}} = [\mathbf{F}, \mathbf{B}]$, $\hat{\mathbf{x}}_{\text{dyn}, t} = [\hat{\mathbf{z}}_{t-1}^{\mathbf{T}}, \mathbf{u}_t^{\mathbf{T}}]^{\mathbf{T}}$, and

$\mathbf{P}_{\text{out}, t} = \begin{pmatrix} \mathbf{P}_t & \hat{\mathbf{z}}_t \mathbf{u}_t^{\mathbf{T}} \\ \mathbf{u}_t \hat{\mathbf{z}}_t^{\mathbf{T}} & \mathbf{u}_t \mathbf{u}_t^{\mathbf{T}} \end{pmatrix}, \qquad \mathbf{P}_{\text{dyn}, t} = \begin{pmatrix} \mathbf{P}_{t-1} & \hat{\mathbf{z}}_{t-1} \mathbf{u}_t^{\mathbf{T}} \\ \mathbf{u}_t \hat{\mathbf{z}}_{t-1}^{\mathbf{T}} & \mathbf{u}_t \mathbf{u}_t^{\mathbf{T}} \end{pmatrix}. \tag{29.95}$

+   * Output matrices:

$\frac{\partial \mathcal{Q}}{\partial \mathbf{A}_{\text{out}}} = \sum_{t=1}^T \mathbf{R}^{-1} \mathbf{y}_t \hat{\mathbf{x}}_{\text{out}, t}^{\mathbf{T}} - \sum_{t=1}^T \mathbf{R}^{-1} \mathbf{A}_{\text{out}} \mathbf{P}_{\text{out}, t} = \mathbf{0} \tag{29.96}$

$\mathbf{A}_{\text{out}}^{\text{new}} = \left( \sum_{t=1}^T \mathbf{y}_t \hat{\mathbf{x}}_{\text{out}, t}^{\mathbf{T}} \right) \left( \sum_{t=1}^T \mathbf{P}_{\text{out}, t} \right)^{-1} \tag{29.97}$

+   * Output noise covariance:

$\frac{\partial \mathcal{Q}(\mathbf{A}_{\text{out}} = \mathbf{A}_{\text{out}}^{\text{new}})}{\partial \mathbf{R}^{-1}} = \frac{T}{2} \mathbf{R} - \frac{1}{2} \sum_{t=1}^T \left( \mathbf{y}_t \mathbf{y}_t^{\mathbf{T}} - 2 \mathbf{A}_{\text{out}}^{\text{new}} \hat{\mathbf{x}}_{\text{out}, t} \mathbf{y}_t^{\mathbf{T}} + \mathbf{A}_{\text{out}}^{\text{new}} \mathbf{P}_{\text{out}, t} \mathbf{A}_{\text{out}}^{\text{new}, \mathbf{T}} \right) = \mathbf{0} \tag{29.98}$

$\mathbf{R}^{\text{new}} = \frac{1}{T} \sum_{t=1}^T (\mathbf{y}_t \mathbf{y}_t^{\mathbf{T}} - \mathbf{A}_{\text{out}}^{\text{new}} \hat{\mathbf{x}}_{\text{out}, t} \mathbf{y}_t^{\mathbf{T}}) \tag{29.99}$

+   * System dynamics matrices:

$\frac{\partial \mathcal{Q}}{\partial \mathbf{A}_{\text{dyn}}} = - \sum_{t=2}^T \mathbf{Q}^{-1} \left[ \mathbf{P}_{t, t-1}, \hat{\mathbf{z}}_t \mathbf{u}_t^{\mathbf{T}} \right] + \sum_{t=2}^T \mathbf{Q}^{-1} \mathbf{A}_{\text{dyn}} \mathbf{P}_{\text{dyn}, t} = \mathbf{0} \tag{29.100}$

$\mathbf{A}_{\text{dyn}}^{\text{new}} = \left( \sum_{t=2}^T [\mathbf{P}_{t, t-1}, \hat{\mathbf{z}}_t \mathbf{u}_t^{\mathbf{T}}] \right) \left( \sum_{t=2}^T \mathbf{P}_{\text{dyn}, t} \right)^{-1} \tag{29.101}$

+   * State noise covariance:

$\frac{\partial \mathcal{Q}(\mathbf{A}_{\text{dyn}} = \mathbf{A}_{\text{dyn}}^{\text{new}})}{\partial \mathbf{Q}^{-1}} = \frac{T-1}{2} \mathbf{Q} - \frac{1}{2} \sum_{t=2}^T (\mathbf{P}_t - 2 \mathbf{A}_{\text{dyn}}^{\text{new}} [\mathbf{P}_{t-1, t}, \mathbf{u}_t \hat{\mathbf{z}}_t^{\mathbf{T}}] + \mathbf{A}_{\text{dyn}}^{\text{new}} \mathbf{P}_{\text{dyn}, t} \mathbf{A}_{\text{dyn}}^{\text{new}, \mathbf{T}}) \tag{29.102}$

$= \frac{T-1}{2} \mathbf{Q} - \frac{1}{2} \sum_{t=2}^T (\mathbf{P}_t - \mathbf{A}_{\text{dyn}}^{\text{new}} [\mathbf{P}_{t-1, t}, \mathbf{u}_t \hat{\mathbf{z}}_t^{\mathbf{T}}]) = \mathbf{0} \tag{29.103}$

$\mathbf{Q}^{\text{new}} = \frac{1}{T-1} \sum_{t=2}^T (\mathbf{P}_t - \mathbf{A}_{\text{dyn}}^{\text{new}} [\mathbf{P}_{t-1, t}, \mathbf{u}_t \hat{\mathbf{z}}_t^{\mathbf{T}}]) \tag{29.104}$
