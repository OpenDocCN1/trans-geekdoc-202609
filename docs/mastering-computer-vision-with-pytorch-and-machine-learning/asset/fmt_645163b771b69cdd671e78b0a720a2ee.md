
### C Gradients of MSE loss function to weights in a linear regression

$$
\begin{bmatrix} 1 & x_{0,1} & {.\ ..} & x_{0,m} \\ 1 & x_{1,1} & {.\ ..} & x_{1,m} \\ {.\ ..} & {.\ ..} & {.\ ..} & {.\ ..} \\ 1 & x_{n,1} & {.\ ..} & x_{n,m} \\ \end{bmatrix} \cdot \begin{bmatrix} w_{0} \\ w_{1} \\ {.\ ..} \\ w_{m} \\ \end{bmatrix} = \begin{bmatrix} {\widehat{y}}_{0} \\ {\widehat{y}}_{1} \\ {.\ ..} \\ {\widehat{y}}_{n} \\ \end{bmatrix},\mathit{Y} = \begin{bmatrix} y_{0} \\ y_{1} \\ {.\ ..} \\ y_{n} \\ \end{bmatrix},(i = 0,1...n,j = 0,1...m)
$$

Model: $X \cdot w = \hat{Y}$

Model prediction: ${\widehat{y}}_{i} = \sum\limits_{j = 0}^{m}{x_{i,j}w_{j}}$

Loss function: $L\left( X,w \right) = \frac{1}{\left( n + 1 \right)}\sum\limits_{i = 0}^{n}\left( {\widehat{y}}_{i} - y_{i} \right)^{2}$

$$
J_{i,j} = \frac{\partial{\widehat{y}}_{i}}{\partial w_{j}} = \frac{\partial}{\partial w_{j}}\sum\limits_{k = 0}^{m}x_{i,k}w_{k} = x_{i,j}
$$

$$
\nabla L(X,w) = \left( {\frac{\partial L}{\partial w_{0}},\frac{\partial L}{\partial w_{1}},...,\frac{\partial L}{\partial w_{m}}} \right)^{T}
$$

$$
\frac{\partial L}{\partial v_{1}} = \frac{- 1}{2v_{1}^{2}}\sum\limits_{j = 1}^{n}{p\left( z_{1}\mid x_{j} \right)\left\lbrack {v_{1} - \left( x_{j} - \mu_{1} \right)^{2}} \right\rbrack} = 0
$$

$$
v_{1} = \sigma_{1}^{2} = \frac{\sum\limits_{j = 1}^{n}{\left( x_{j} - \mu_{1} \right)^{2}p\left( z_{1}\mid x_{j} \right)}}{\sum\limits_{j = 1}^{n}{p\left( z_{1}\mid x_{j} \right)}}
$$

$$
v_{i} = \sigma_{i}^{2} = \frac{\sum\limits_{j = 1}^{n}{\left( x_{j} - \mu_{i} \right)^{2}p\left( z_{i}\mid x_{j} \right)}}{\sum\limits_{j = 1}^{n}{p\left( z_{i}\mid x_{j} \right)}}
$$

$$
\frac{\partial L}{\partial p\left( z_{1} \right)} = - \frac{\partial}{\partial p\left( z_{1} \right)}\sum\limits_{j = 1}^{n}{\ \ln\ \left( { p\left( x_{j} \right)} \right)} = - \sum\limits_{j = 1}^{n}{\frac{1}{p\left( x_{j} \right)}\frac{\partial p\left( x_{j} \right)}{\partial p\left( z_{1} \right)}}
$$

$p(z_{3}) = 1 - p(z_{1}) - p(z_{2})$

$$
\frac{\partial L}{\partial p\left( z_{1} \right)} = - \sum\limits_{j = 1}^{n}{\frac{1}{p\left( x_{j} \right)}\frac{\partial}{\partial p\left( z_{1} \right)}}\sum\limits_{i = 1}^{3}{p\left( x_{j}\mid z_{i} \right)}p\left( z_{i} \right) = - \sum\limits_{j = 1}^{n}{\frac{1}{p\left( x_{j} \right)}\left\lbrack { p\left( x_{j}\mid z_{1} \right) - p\left( x_{j}\mid z_{3} \right)} \right\rbrack} = 0
$$

$$
\frac{\partial L}{\partial p\left( z_{1} \right)} = - \sum\limits_{j = 1}^{n}\left\lbrack {\frac{ p\left( z_{1} \right)}{p\left( x_{j} \right)}\frac{p\left( {x_{j} \mid z_{1}} \right)}{p\left( z_{1} \right)} - \frac{p\left( z_{3} \right)}{p\left( x_{j} \right)}\frac{p\left( {x_{j} \mid z_{3}} \right)}{p\left( z_{3} \right)}} \right\rbrack
$$

$$
\frac{\partial L}{\partial p\left( z_{1} \right)} = - \sum\limits_{j = 1}^{n}\left\lbrack {\frac{ p\left( z_{1}\mid x_{j} \right)}{p\left( z_{1} \right)} - \frac{p\left( z_{3}\mid x_{j} \right)}{p\left( z_{3} \right)}} \right\rbrack
$$

$$
\frac{\partial L}{\partial p\left( z_{1} \right)} = - \frac{1}{p\left( z_{1} \right)}\sum\limits_{j = 1}^{n}{p\left( z_{1}\mid x_{j} \right)\ + \ }\frac{1}{p\left( z_{3} \right)}\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)} = 0
$$

$$
p\left( z_{1} \right) = p\left( z_{3} \right)\frac{\sum\limits_{j = 1}^{n}{p\left( z_{1}\mid x_{j} \right)}}{\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}}
$$

$$
p\left( z_{2} \right) = p\left( z_{3} \right)\frac{\sum\limits_{j = 1}^{n}{p\left( z_{2}\mid x_{j} \right)}}{\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}}
$$

$$
p\left( z_{3} \right)\frac{p\left( z_{1} \right)}{\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}} + p\left( z_{3} \right)\frac{p\left( z_{2} \right)}{\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}} + p\left( z_{3} \right) = 1
$$

$$
p\left( z_{3} \right)\frac{1}{\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}}\left\lbrack {\sum\limits_{j = 1}^{n}{p\left( z_{1}\mid x_{j} \right)} + \sum\limits_{j = 1}^{n}{p\left( z_{2}\mid x_{j} \right)} + \sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}} \right\rbrack = 1
$$

$$
p\left( z_{3} \right)\frac{1}{\sum\limits_{j = 1}^{n}{p\left( {z_{3} \mid x_{j}} \right)}}\sum\limits_{j = 1}^{n}{\sum\limits_{i = 1}^{3}{p\left( {z_{i} \mid x_{j}} \right)}} = p\left( z_{3} \right)\frac{1}{\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}}\sum\limits_{j = 1}^{n}1 = 1
$$

$$
p\left( z_{3} \right) = \frac{1}{n}\sum\limits_{j = 1}^{n}{p\left( z_{3}\mid x_{j} \right)}
$$

$$
p\left( z_{2} \right) = \frac{1}{n}\sum\limits_{j = 1}^{n}{p\left( z_{2}\mid x_{j} \right)}
$$

$$
p\left( z_{1} \right) = \frac{1}{n}\sum\limits_{j = 1}^{n}{p\left( z_{1}\mid x_{j} \right)}
$$
