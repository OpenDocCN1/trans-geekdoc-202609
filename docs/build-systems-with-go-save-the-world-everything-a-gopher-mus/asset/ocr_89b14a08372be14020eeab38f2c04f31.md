if len(r.data) <= 0 || r.from == len(r.data) {
    return 0, io.EOF
}
n := len(r.data) - r.from
if len(p) < n {
    n = len(p)
}
for i := 0; i < n; i++ {
    b := byte(r.data[r.from])
    p[i] = b
    r.from++
}
if r.from == len(r.data) {
    return n, io.EOF
}
return n, nil
