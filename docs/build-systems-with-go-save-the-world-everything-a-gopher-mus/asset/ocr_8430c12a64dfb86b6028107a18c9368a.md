package main

import (
    "fmt"
    "github.com/confluentinc/confluent-kafka-go/kafka"
)

func main() {
    cfg := kafka.ConfigMap("bootstrap.servers": "localhost:9092")
    p, err := kafka.NewProducer(cfg)
    if err != nil {
        panic(err)
    }
    defer p.Close()

    deliveryChan := make(chan kafka.Event, 10)
    defer close(deliveryChan)
    topic := "HelloTopic"
    msgs := []string{"Save", "the", "world", "with", "Go!!!"}
    for _, word := range msgs {
        err = p.Produce(&kafka.Message{
            TopicPartition: kafka.TopicPartition{Topic: &topic, Partition: kafka.PartitionAny},
            Value:          []byte(word),
        }, deliveryChan)
        if err != nil {
            panic(err)
        }
    }
    e := <-deliveryChan
    fmt.Printf("Sent %v\n", e.TopicPartition)
    // p.Flush(10000)

The Poll function blocks the execution for a given timeout and returns an Event or nil.
