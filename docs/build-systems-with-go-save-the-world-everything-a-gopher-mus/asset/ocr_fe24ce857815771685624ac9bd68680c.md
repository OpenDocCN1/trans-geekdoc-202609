```
func func (u *UserServer) ServeGrpc() {
    lis, err := net.Listen("tcp", u.grpcAddr)
    if err != nil {
        panic(err)
    }
    s := grpc.NewServer()
    pb.RegisterUserServiceServer(s, u)
    fmt.Println("Server listening GRPC:!")

    if err := s.Serve(lis); err != nil {
        panic(err)
    }
}
```

When all the elements needed to run our gRPC server are ready, we can include the HTTP transcoding. Example 15.16 contains the code to serve the remote procedures defined in the example using HTTP. We have to register every service from the proto file using the corresponding RegisterXxxHandlerFromEndPoint function. This function binds the gRPC server address to a mux server^41, for simplicity, the server has two addresses: one for gRPC and the other for HTTP. Remember that the HTTP server is a gateway that forwards requests to our gRPC server.

# Example 15.16: gRPC gateway using HTTP (part II).

```
func (u *UserServer) ServeHttp() {
    mux := runtime.NewServeMux()
    opts := []grpc.DialOption{grpc.WithInsecure()}
    endpoint := u.grpcAddr

    err := pb.RegisterUserServiceHandlerFromEndPoint(context.Background(), mux, endpoint, opts)
    if err != nil {
        panic(err)
    }

    httpServer := &http.Server{
        Addr: u.httpAddr,
        Handler: mux,
    }

    fmt.Println("Server listing HTTP:")
    if err := httpServer.ListenAndServeServe(); err != nil {
        panic(err)
    }
}

func main() {
    us := NewServer(httpAddr:":8080", grpcAddr:":50051")
    go us.ServeGrpc()
    us.ServeHttp()
}
```
