Example 20.7 uses a writer instance to send messages to Kafka. The code is very similar to the one shown in Example 20.5. However, note that all the operations are performed using the Writer instance of a Connection instance.

# Example 20.7: Segmentio Kafka high level API producer.

```
package main

import (
    "context"
    "fmt"
    "github.com/segmentio/kafka-go"
    "time"
)

func main() {
    topic := "helloTopic"
    partition := 0
    conn, err := kafka.DialLeader(context.Background(), "localhost:9092", topic, partition)
    if err != nil {
        panic(err)
    }

    defer conn.Close()

    conn.WriteDeadline(time.Now().Add(3*time.Second))
    msgs := []string{"Save", "world", "with", "Go!!!"}
    for _, msg := range msgs {
        i, err := conn.WriteMessages(kafka.Message{Value: []byte(msg)})
        if err != nil {
            panic(err)
        }
    }
    fmt.Printf("Sent %d bytes: %s\n", i, msg)
}
```
