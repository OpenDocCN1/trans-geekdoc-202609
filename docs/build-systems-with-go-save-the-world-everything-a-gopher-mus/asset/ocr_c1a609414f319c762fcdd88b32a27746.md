# Example 5.11: Access to field tags using reflect.

```
package main

import (
    "fmt"
    "reflect"
)

type User struct {
    ID       string `tagA:"valueA1" tagB:"valueA2`
    Email    string `tagB:"value"
    Password string `tagC:"v12""
}

func main() {
    T := reflect.TypeOf(User{})

    fieldID := T.Field("ID")
    t := fieldID.Tag
    fmt.Printf("StructTag: %s\n", t)

    V, _ := t.Lookup("tagA")
    fmt.Printf("Tag: %v\n", V)

    fieldEmail := T.FieldByName("Email")
    vEmail := fieldEmail.Tag.Get("tagB")
    fmt.Printf("Email: %s\n", vEmail)

    fieldPassword := T.FieldByName("Password")
    fmt.Printf("Password tags: [%s]\n", fieldPassword.Tag)
    fmt.Println(fieldPassword.Tag.Get("tagC"))
}
```

tagB:"valueA2"
tagB:"value"
mail tag: value
Password tags: [tagC:"v1 v2"]
v12

> By convention tags must be declared like key:"value" strings. That blank spaces in the string do not follow the convention. E.G.: key:"value" is not a valid declaration.
