utilization of the plural form. The Example 18.17 defines a belongs-to-relationship, where a user belongs to a group. Notice that in this case, we only expect to have one group for the user. This is mentioned in the `db.Preload("Group")` statement.

# Example 18.17: GORM preload with single record.

```
package main

import (
	"fmt"
	"gorm.io/driver/sqlite"
	"gorm.io/gorm"
)

type User struct {
	ID uint
	Name string
	GroupID uint
	Group Group
}

type Group struct {
	ID uint
	Name string
}

func main() {
	db, err := gorm.Open("/tmp/example84.db", &gorm.Config{})
	if err != nil {
		panic("Failed to connect database")
	}

	err = db.AutoMigrate(&Group{}, &User{})
	if err != nil {
		panic(err)
	}

	g := Group(Name: "TheCoolOnes")
	u := User(Name: "John", Group: g)
	db.Create(&u)

	var recovered User
	db.Preload("Group").First(&recovered, 1)
	fmt.Println("Recovered", recovered)
}
```
