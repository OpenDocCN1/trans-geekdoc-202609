library(rpart)
Dataset1 <- maml.mapInputPort(1)
irData <- rpart(Species~Sepal.Length + Sepal.Width + Petal.Length + Petal.Width, method="class",
data=Dataset1)

printcp(irisData)
plot(irisData)

text(irisData, use.n=TRUE, all=TRUE, cex=0.8, pretty=1)
data.set = Dataset1

maml.mapOutputPort("data.set")
