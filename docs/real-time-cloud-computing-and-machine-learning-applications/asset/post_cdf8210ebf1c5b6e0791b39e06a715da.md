package NeuralNetwork;
import deepennets.data.TabularDataSet;
import deepennets.util.Tensor;
import java.util.Arrays;

/* Extends the TabularDataSet Class to implement our own DataSet Generator since
t library only supports importing data from CSV files. */
public class DataSetIterator extends TabularDataSet<TabularDataSet.Item>{
//Normalizer to normalize the dataset
private final MinMax normalizer;
//array to store the normalized data
private final float[][] normalized;
/* tf - number of the previous sample to use for prediction
 * tn - number of step to predict */
private final int tn, tf;
/**
 * Constructor that creates a 2D array of whatever observations and normalizes it.
 * @param data_o normalized 2D array used to generate the training dataset
 * @param tf number of the previous sample to use for prediction
 * @param tn number of step to predict
 */
public DataSetIterator(float[] data_o, int tf, int tn) {
    super();
    this.tf = tf;
    this.tn = tn;
    //create a new minmax normalizer
    normalizer = new MinMaxNorm(data_o);
    //normalizes the data array first
    this.normalized = normalizer.normalize();
}
/**
 * Generates a training dataset for a CNN network.
 * */
public void generateMultiVarDataSet() {
    int col = normalized[0].length;
    int row = normalized.length;
//implement rolling window to generate a dataset
for (int i = 0; i < row - tn + 1; i++) {
    float[] data_in = new float[tf][col];
    float[] data_out = new float[tn * col];
//fetch last tf row as input
for (int j = 0; j < tf; j++) {
    data_in[j] = normalized[i + j];
}
//uses the next tn row as output
int u = 0;
for (int j = 0; j < tn; j++) {
    float[] outcome = normalized[tf + i + j];
    //flattens the output and store in an array
    for (float v : outcome) {
        data_out[u] = v;
        u++;
    }
}
}

/* Create an input & output tensor to store the sample since the dimension is not the same. Use the Tensor class from the DeepNetts Library */
Tensor input = new Tensor(data_in);
Tensor output = new Tensor(data_out);
//create a new sample that contains both input & output
Item frame = new Item(input, output);
//adds the new item to the List
items.add(frame);
}

/** Generate an input tensor based on the last tf row from the observations. *
* @return an input tensor {@code model_input} for a CNN Result */ */
public Tensor getLastTimestampCNN() {
    int col = normalized[0].length;
    int row = normalized.length;
    float[][] input = new float[tf][col];
    int index = row - tf - 1;
    for (int i = 0; i < tf; i++) {
        input[i] = normalized[index + i];
    }
    return new Tensor(input);
}

public class MLPRegression {
    public DataSetIterator dataset;
    public FeedForwardNetwork network;
    public MLPRegression(float[] data, int tf, int tn) {
        //Create a new DataSetIterator object using the weather data
        dataset = new DataSetIterator(data, tf, tn);
        //Create a new FeedForwardNetwork using the DeepNets Library
        network = new FeedForwardNetwork.builder()
            .addInputLayer(tf)
            .addOutputLayer(tn, ActivationType.TANH)
            .lossFunction(LossType.MEAN_SQUARED_ERROR)
            .build();
    }

    protected void trainModel(int maxEpoch) {
        //Generate Backpropagation trainer for training network
        BackpropagationTrainer trainer = network.getTrainer();
        //maximum error for the network
        trainer.setError(0.0001f);
        //maximum iterations for the network
        trainer.setMaxEpochs(maxEpoch);
        //sets the learning rate of the network
        trainer.setLearningRate(0.01f);
        //sets the momentum of the network
        trainer.setMomentum(0.1f);
        //Train Network over generated dataset over a maximum epoch
        network.train(dataset);
    }

    public float predictNext(int epoch, int epoch, int var) {
        //reshape the data for the CNN Network
        dataset.generateSingleVarDataset(var);
        trainModel(epoch);
        //get the last item in the dataset using the Tensor from the DeepNets library
        Tensor input = dataset.getLastTimestampMLP(var);
        //activate the network with an input sensor
        network.setInput(input);
        //get the output from the network
        float[] output = network.getOutput();
    }
}
