```
// HTML page and store them as variables:

var StdId = document.getElementById('ID');
var StdName = document.getElementById('Name');
var StdCourse = document.getElementById('Course');
var StdGrade = document.getElementById('Grade');
```

// Obtain the values stored in the variables that were extracted:

```
var v1= StdId.value;
var v2= StdName.value;
var v3 = StdCourse.value;
var v4= StdGrade.value;
```

// Create a database reference variable which specifies the database\identifier in Firebase as Student_Records

```
var dataRef = firebase.database().ref('Student_Records');
```

// Call the child to make the Student ID which is stored in
//variable v1, become the key variable for a given student record and use
// set function to assign values to the fields Name, Course and Grade in the
// database ref as entered by the user.

```
dataRef.child(v1).set({
Name: v2,
Course: v3,
Grade: v4
});
```

// Send a message when the record has been saved to the database
//successfully.

```
alert("Student's Record Saved Successfully");
```

// Clear the fields in the Webpage so that the user can enter the records of
// a new student.

```
document.getElementById('ID').value = '';
document.getElementById('Name').value = '';
document.getElementById('Course').value = '';
document.getElementById('Grade').value = '';
```
