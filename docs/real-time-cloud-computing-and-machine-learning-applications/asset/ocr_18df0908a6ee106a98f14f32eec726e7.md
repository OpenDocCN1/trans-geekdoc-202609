# Listing 5.45. doGet() method for livegraph.java

```
protected void doGet(HttpServletRequest request, HttpServletResponse response)
    throws ServletException, IOException {
    // Allow cross-domain requests (Allow CORS response)
    response.setHeader("Access-Control-Allow-Origin", "*");
    
    // Indicating the source of data specified by the user, in the HTTP request
    Double source = Double.parseDouble(request.getParameter("source"));
    if (source == 1) {
        dataSource = "cloudant";
    }
    if (source == 2) {
        dataSource = "localhost";
    }

    // Fetching the last set of values from the PC database
    fetchlastvalPC();

    // Write back to JSON format after obtaining results
    JSONObject jonnn = new JSONObject();
    jonnn.put("downloadSpeed", downspeed);
    jonnn.put("uploadSpeed", upspeed);
    jonnn.put("packetsReceived", pacRec);
    jonnn.put("packetsSent", pacSent);

    // Using a PrintWriter object to write onto the page
    PrintWriter out = response.getWriter();
    // Setting the type of content to JSON object
    response.setContentType("application/json");
    response.setCharacterEncoding("UTF-8");
    // Writing values on the page
    out.print(jonnn.toString());
    out.flush();
}
```
