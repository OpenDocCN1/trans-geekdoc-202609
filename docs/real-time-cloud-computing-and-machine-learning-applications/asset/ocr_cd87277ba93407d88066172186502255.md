# Listing 6.3. OnCreate method

```
@Override
protected void onCreate(Bundle savedInstanceState) {
    super.onCreate(savedInstanceState);
    setContentView(R.layout.activity_main);

    //Retrieve Widgets from the UI
    Button analyse = findViewById(R.id.analyse_btn);
    ip_address = findViewById(R.id.ip_address);
    monitor = findViewById(R.id.monitor_btn);
    spinner = findViewById(R.id.var_spinner);
    mChart = findViewById(R.id.livedata);
    cloud_save = findViewById(R.id.cloud_save);
    download_data = findViewById(R.id.download_data);

    monitor.setOnClickListener(v -> {
        if (!monitoring) {
            // not monitoring, it starts monitoring process and changes button text
            monitoring = true;
            cloud_save.setEnabled(false);
            monitor.setText(R.string.stop_button);
            //Causes the Runnable to be added to the queue and runs after the time expires
            mHandler.postDelayed(Runnable, 1000);
        } else {
            // if monitoring, it stops monitoring and changes button text
            monitoring = false;
            cloud_save.setEnabled(true);
            monitor.setText(R.string.monitor_button);
            //Remove any pending posts of Runnable that are in the message queue.
            mHandler.removeCallbacks(Runnable);
        }
    });

analyse.setOnClickListener(v -> {
    //starts analyse activity on button click
    Intent intent = new Intent(MainActivity.this, Analysis.class);
    startActivity(intent);
});

download_data.setOnClickListener(new View.OnClickListener() {
    @Override
    public void onClick(View view) {
        //starts download activity on button click
        Intent intent = new Intent(MainActivity.this, DownloadData.class);
        startActivity(intent);
    }
});
```
