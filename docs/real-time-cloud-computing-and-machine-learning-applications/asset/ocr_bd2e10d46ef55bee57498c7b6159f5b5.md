function Retrieve_Record(){

// Get the value of the ID entered in the Web page as a variable.

  var enterID= document.getElementById('getID').value;

  // Create a database reference variable which specifies the database
  //identifier in Firebase as Student_Records
  // Call the child function to identify the Student's record from the enterID
  // variable. A snapshot object which holds the data stored in the database
  // is also thrown each time the database is requested.

  firebase.database().ref('Student_Records').child(enterID).on('value',
function(snapshot) {

  // Extract the student Name, Course and Grade from the snapshot variable and
  // store them in three different variables:

  var S_Name = snapshot.val().Name;
  var S_Course = snapshot.val().Course;
  var S_Grade = snapshot.val().Grade;

  // Output the Name, Course and Grade of the retrieved record to the
  //respective text fields on the web page:

  document.getElementById('Name2').value=S_Name;
  document.getElementById('Course2').value=S_Course;
  document.getElementById('Grade2').value=S_Grade;

});

alert("Student's Record Retrieved Successfully");
}
