# Listing 5.59. Downloading parameters by date

```
function downloadDate() {
  // Retrieving name of samples from the webpage
  var downloadDate = document.getElementById(`downloadDate`);value;
  // Fetching the selected source from either Cloudant or Localhost
  var selectedSource =
  sourceSelection.options[sourceSelection.selectedIndex].innerHTML;
  // Fetching the IP address of the localhost if it is selected
  var ipAddress = " + document.getElementById(`serverAddress`).value;
  var query; // to be passed in fetch call
  if(selectedSource === `Cloudant database`) {
    query = `https://networkcloud.eu.gb.blumemix.net/downloaderDate?` +
    `date=` + downloadDate + `&source=1`;
  }
  else if(selectedSource === `Local host`) {
    query = `http://` + ipAddress + `:.8080/NetworkCloud/downloaderDate?` +
    `date=` + downloadDate + `&source=2`;
  }
  // Making the query to the selected source
  fetch(query)
    .then(res => res.json())
    .then((out) => {
      // Obtaining the String object by servlet
      var csv_dat = out.csv;
      console.log(csv_dat);
      // Creating an element to output the downloadable .csv file
      var hiddenElement = document.createElement('a');
      // Setting required meta
      hiddenElement.href = 'data:text/csv;charset=utf-8,' + +
      encodeURI(csv_dat);
      hiddenElement.target = '_blank';
      // Setting name of file
      hiddenElement.download = 'networkPCDate.csv';
      // Outputting the .csv file as a downloadable file from the browser
      hiddenElement.click();
    }).catch(err => console.error(err));
```
