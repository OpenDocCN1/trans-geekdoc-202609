# Listing 5.58. Downloading parameters by last N samples

```
function downloadLastN() {
// Retrieving number of samples to download from the webpage
  var downloadSamples =
  document.getElementById("downloadSamples").value;
// Fetching the source from either Cloud or Localhost
  var selectedSource =
  sourceSelection.options[sourceSelection.selectedIndex].innerHTML;
// Fetching the IP address of the localhost if the toggle is selected
  var ipAddress = "" + document.getElementById("serverAddress").value;
  var query = "Query to be passed in Fetch call";
if(selectedSource === "Cloudant database") {
    query = "https://network.cloud-eu-gb.mybluemix.net/downloader?" + "number=" + downloadSamples + "&source=1";
}
if(selectedSource === "Local host") {
    query = "http://" + ipAddress + ":8080/NetworkCloud/downloader?" + "number=" + downloadSamples + "&source=2";
}
// Making the query to the selected source
fetch(query);
  .then(res => res.json())
  .then((out) => {
// Obtaining the String object returned by servlet
  var csv_dat = out.csv;
  console.log(csv_dat);
// Creating an element to output the downloadable .csv file
  var hiddenElement = document.createElement('a');
// Setting required meta
  hiddenElement.href = 'data:text/csv;charset=utf-8,' + 
  encodeURI(csv_dat);
  hiddenElement.target = '\_blank';
// Setting name of file
  hiddenElement.download = 'networkPCSamples.csv';
// Outputting the .csv file as a downloadable file from the browser
  hiddenElement.click();
}).catch(err => console.error(err));
}
```
