# Listing 5.55. Analytics methods for index1.js

function classification() {
  // Setting a on the display box
  document.getElementById(withKNC).value = "Please wait...";
  document.getElementById(withMLPC).value = "Please wait...";

  // Retrieval values from the page
  var downloadSpeed = document.getElementById(downloadSpeed).value;
  var uploadSpeed = document.getElementById(uploadSpeed).value;
  var packetsReceived = document.getElementById(packetsReceived).value;
  var packetsSent = document.getElementById(packetsSent).value;
  // Fetching the selection from either Cloud or Local
  var selectedSource =
  sourceSelection.options(sourceSelection.selectedIndex).innerHTML;
  // Fetching the IP address of the localhost is selected
  var ipAddress = "" + document.getElementById("serverAddress").value;

  var query;
  // Selecting the source as "1"
  if(selectedSource == "Cloud database") {
    query = "http://networkcloud.eu-gb.myblueix.net/" + +
    "downloadSpeed=" + downloadSpeed + "&uploadSpeed=" + uploadSpeed +
    "&packetsReceived=" + packetsReceived + "&packetsSent=" +
    "&predictionTime=10&windowSize=20&device=1&action=1&source=2";
  }
  // Selecting the source as "2"
  if(selectedSource == "Local host") {
    query = "http://" + ipAddress + ":8080/NetworkCloud/" + +
    "downloadSpeed=" + downloadSpeed + "&uploadSpeed=" + uploadSpeed +
    "&packetsReceived=" + packetsReceived + "&packetsSent=" +
    "&predictionTime=10&windowSize=20&device=1&action=1&source=2";
  }
}
