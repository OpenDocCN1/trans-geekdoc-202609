# Listing 2.2. Codes for the MainActivity.java file of the Student_Records Application

```java
package com.example.student_records;

// The following imports are required for the Firebase Database:

import com.google.firebase.database.DatabaseReference;
import com.google.firebase.database.FirebaseDatabase;
import com.google.firebase.database.*;

// Other imports for the Activity:

import androidx.annotation.NonNull;
import androidx.app.AppCompatActivity;
import android.content.Intent;
import android.os.Bundle;
import android.view.View;
import android.widget.*;

public class MainActivity extends AppCompatActivity {
    // Declare a database reference object:
    DatabaseReference mDatabase;

    // Declare the TextView, EditText and Button variables that will be required:
    TextView TV2;
    EditText ED1,ED2,ED3,ED4,ED5;
    Button B1,B2;

    // Declare the variables pertaining to the Student records:
    String id,id1,name,course,grade;

    @Override
    protected void onCreate(savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // Create an instance of the mdatabase object inside the onCreate method:
        mDatabase = FirebaseDatabase.getInstance().getReference();

        // Link all GUI variables in the MainActivity to their counterparts in the
        // activity_main.xml file

        TV2 = (TextView)findViewById(R.id.TV2);
        ED1 = (EditText)findViewById(R.id.ED1);
        ED2 = (EditText)findViewById(R.id.ED2);
        ED3 = (EditText)findViewById(R.id.ED3);
        ED4 = (EditText)findViewById(R.id.ED4);
        ED5 = (EditText)findViewById(R.id.ED5);
        B1 = (Button)findViewById(R.id.B1);
        B2 = (Button)findViewById(R.id.B2);

        B1.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                id = ED1.getText().toString();
                name = ED2.getText().toString();
                course = ED3.getText().toString();
                grade = ED4.getText().toString();

                Student std = new Student(name, course, grade);
                mDatabase.child("Students").child(id).setValue(std);
            }
        });

        B2.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                id = ED5.getText().toString();
                mDatabase.child("Students").child(id).addValueEventListener(new ValueEventListener() {
                    @Override
                    public void onData(@NonNull DataSnapshot snapshot) {
                        // Extract the values from snapshot at this location.
                        Student post = snapshot.getValue(Student.class);
                        if (post != null) {
                            id1 = id;
                            name = post.name;
                            course = post.course;
                            grade = post.grade;
                        }
                    }

                    @Override
                    public void onCancelled(@NonNull DatabaseError databaseError) {
                        // This method is required in case there is an error reading from the database.
                    }
                });
            }
        });
    }
}
```

# Listing 2.3. The Student.java file

```java
package com.example.student_records;

public class Student {
  public String name;
  public String course;
  public String grade;

  public Student(){
}
  public Student(String name, String course, String grade){
    this.name=name;
    this.course=course;
    this.grade=grade;
  }
}
```
