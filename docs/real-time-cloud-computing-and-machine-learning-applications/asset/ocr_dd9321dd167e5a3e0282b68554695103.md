# Listing 7.10. The fetchForDate Method

```
public Object[][] fetchForDate(String date) {
double[] result = null;
String[] timestamps = null;
try {
  //Creates a Statement for sending SQL statements to the database.
  Statement stmt = conn.createStatement();
  //Executes the SQL query, returns a single ResultSet object.
  ResultSet resultSet = stmt.executeQuery("SELECT FROM weather_params WHERE Timestamp LIKE "" + date + "%%%");
  //iterate through the result_set to find the number of records
  int N = 0;
  while (resultSet.next()) {
    N++;
  }
  //Execute the query once we have the size, so that we initialize the array
  resultSet = stmt.executeQuery("SELECT FROM weather_params WHERE TIMESTAMP LIKE "" + date + "%%%");
  result = new double[N][7];
  timestamps = new String[N];
  //Iter through the ResultSet and fill the data and timestamps array.
  int j = 0;
  while (resultSet.next()) {
    timestamps[j] = resultSet.getString(2);
    // fetches the weather parameters starting from column 3 and store it
    for (int i = 0; i < 7; i++) {
      result[j][i] = resultSet.getDouble(i + 3);
    }
    j++;
  }
} catch (SQLException ex) {
  Logger.getLogger(SQLConnector.class.getName()).log(Level.SEVERE, null, ex);
}
//Creates an object array to return multiple types of arrays into a single array
return new Object[][][]{timestamps, result];
}
```
