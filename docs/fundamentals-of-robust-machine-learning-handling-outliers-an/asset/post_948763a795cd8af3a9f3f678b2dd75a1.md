if (LC):
    v = p*(1.-p)*np.tanh(a*(y-p))
    v = v.T
    v_stacked = np.tile(v[:, np.newaxis],
        (1, X.shape[1]) )
    X_multiplied = -X * v_stacked
    dw = np.sum(X_multiplied, axis=0)
    db = np.sum(p*(1.-p)*np.tanh(a*(y-p)) * (-1))
else:
    v = y-p
    v_stacked = np.tile(v[:, np.newaxis],
        (1, X.shape[1]) )
    X_multiplied = -X * v_stacked
    dw = np.sum(X_multiplied, axis=0)
    db = np.sum((y-p)*(-1))
return dw, db
