Next we need to update the Gallery type's Images field, along with the ImagesSplit method.

```
type Gallery struct {
	gorm.Model
	UserID uint `gorm:"not null;index"`
	Title string `gorm:"not null"`
	Images []Image `gorm:"-\`
}
```

```
func (g *Gallery) ImagesSplitN(n int) ([][]Image {
	ret := make([][]Image, n)
	for i := 0; i < n; i++ {
		ret[i] = make([]Image, 0)
	}
	for i, img := range g.Images {
		bucket := i % n
		ret[bucket] = append(ret[bucket], img)
	}
	return ret
}
```
