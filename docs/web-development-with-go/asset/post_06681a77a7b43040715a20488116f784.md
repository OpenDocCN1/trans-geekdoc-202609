func WithGorm(dialect, connectionInfo string) ServicesConfig {
  return func(s *Services) error {
    db, err := gorm.Open(dialect, connectionInfo)
    if err != nil {
      return err
    }
    s.db = db
    return nil
  }
}
func WithLogMode(mode bool) ServicesConfig {
  return func(s *Services) error {
    s.db.LogMode(mode)
    return nil
  }
}
