```
userId := 1
if i > 3 {
    userId = 2
}

amount := 1000 * i
description := fmt.Sprintf("USB-C Adapter x%d", i)

err = db.QueryRow(
    INSERT INTO orders (user_id, amount, description)
    VALUES ($1, $2, $3)
    RETURNING id
).Scan(&id, &amount, &description)
if err != nil {
    panic(err)
}
fmt.Printf("Created an order with the ID: %s\n", id)

db.Close()
```

After running this code you should be able to run SELECT * FROM orders; in psql and get something like Listing 8.19 where we have several orders, with a few associated to the user with the id 1, and a few associated with the user with the id 2.
