## 12.1.1 规则表示

第一条规则移除三枚硬币，第二条移除两枚硬币，第三条移除一枚硬币。由于我们有四种可能的路径，第三条规则将并行执行两次。

```python
def rules_depth(
    qc = QuantumCircuit(8)
    #First Rule
    # 根据路径描述符设置标记 5
    qc.ccx(6,7,5)
    # 移动
    qc.cx(5,0)
    qc.cx(5,1)
    qc.cx(5,2)
    # 重置标记
    qc.ccx(6,7,5)

    # 第二规则
    # 根据路径描述符设置标记 5
    qc.x(6)
    qc.ccx(6,7,5)
    # 移动
    qc.cx(5,0)
    qc.cx(5,1)
    # 重置标记
    qc.ccx(6,7,5)
    qc.x(6)

    # 第三规则
    # 根据路径描述符设置标记 5
    qc.x(7)
    qc.ccx(6,7,5)
    # 移动
    qc.cx(5,0)
    # 重置标记
    qc.ccx(6,7,5)
    qc.x(7)

    # 第四=第三规则
    # 根据路径描述符设置标记 5
    qc.x(6)
```

def Grover():
    qc = QuantumCircuit(13)
    qc.h([6, 7, 11, 12])
    gate = MCXGate(3)
    qc.append(gate, [7, 11, 12, 6])
    qc.h([6, 7, 11, 12])
    qc.name="G"
return qc
