$= \sqrt[q]{\frac{\prod_{i=1}^n\left(1+(\gamma-1)(\mu_{\tilde{Q}_i})^q\right)^{\omega_i} - \prod_{i=1}^n\left(1-(\mathcal{U}_{\tilde{Q}_i})^q\right)^{\omega_i}}{\prod_{i=1}^n\left(1+(\gamma-1)(\mu_{\tilde{Q}_i})^q\right)^{\omega_i} + (\gamma-1)\prod_{i=1}^n\left(1-(\mu_{\tilde{Q}_i})^q\right)^{\omega_i}}}$,

where $\gamma>0$, $q\ge0$.
