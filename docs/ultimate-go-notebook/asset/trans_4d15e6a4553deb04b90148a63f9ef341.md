# 列表 8.18

```go
func numCPU(n []int, lvl int) []int {
  ..

  // 我们需要的 goroutine 数量不需要超过逻辑处理器数量。
  if lvl <= maxLevel {
    lvl++

    // 每次拆分我们会有 2 个 goroutine。
    var wg sync.WaitGroup
    wg.Add(2)

    // 并发对左侧排序。
    go func() {
      l = numCPU(n[:i], lvl)
      wg.Done()
    }()

    // 并发对右侧排序。
    go func() {
      r = numCPU(n[i:], lvl)
      wg.Done()
    }()

    // 等待拆分结束。
    wg.Wait()

    // 将元素排序并合并有序列表。
    return merge(l, r)
  }

  // 在当前 goroutine 中对左侧和右侧排序。
  l = numCPU(n[:i], lvl)
  r = numCPU(n[i:], lvl)

  // 将元素排序并合并列表。
  return merge(l, r)
}
```
