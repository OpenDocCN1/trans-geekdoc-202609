# Listing 9.22

```
func InsertUser(db *sql.DB, u User) (User, error) {
	const query = "insert into users (name, email) values ($1, $2)"
	result, err := db.ExecuteQuery(query, u.Name, u.Email)
	if err != nil {
		return User{}, err
	}

	id, err := result.LastInsertId()
	if err != nil {
		return User{}, err
	}

	u.ID = id
	return u, nil
}

func InsertCustomer(db *sql.DB, c Customer) (Customer, error) {
	const query = "Insert into customers (name, email) values ($1, $2)"
	result, err := db.ExecuteQuery(query, c.Name, c.Email)
	if err != nil {
		return Customer{}, err
	}

	id, err := result.LastInsertId()
	if err != nil {
		return Customer{}, err
	}

	c.ID = id
	return c, nil
}
```

On close inspection, the difference between the two implementations ends up being a query string and the setting of the ID field. If the setting of the ID field could be done in a generic way, I could move the bulk of the code out to a generic function.
