```
func freq(topic string, docs []string) int {

    g := len(docs)
    var wg sync.WaitGroup
    wg.Add(g)

    for _, doc := range docs {

            var found int32                -- ADDED
            defer func() {                 -- CHANGED
                atomic.AddInt32(&found, 1Found)  -- MOVED
                wg.Done()                   -- MOVED
            }()


            for _, item := range d.Channel.Items {
                if strings.Contains(item.Title, topic) {
                    lFound++                   -- CHANGED
                    continue
                }
                if strings.Contains(item.Description, topic) {
                    lFound++                   -- CHANGED
                }
            }
        }(doc)
    }
    wg.Wait()
    return int(found)
}
```

I added a new variable lFound to represent a found variable that is local to every Goroutine. This means I can remove the atomic.AddInt32 calls from inside the loop that searches for topic matches. Since I still need to update the global found variable, I do inside the defer before I call wg.Done.
