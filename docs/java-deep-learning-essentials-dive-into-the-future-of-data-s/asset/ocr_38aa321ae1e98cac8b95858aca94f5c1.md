public Integer[] predict(double[] x) {

    double[] y = output(x); // activate input data through learned
networks
    Integer[] t = new Integer[nOut]; // output is the probability, so
cast it to label

    int argmax = -1;
    double max = 0.;

    for (int i = 0; i < nOut; i++) {
        if (max < y[i]) {
            max = y[i];
            argmax = i;
        }
    }

    for (int i = 0; i < nOut; i++) {
        if (i == argmax) {
            t[i] = 1;
        } else {
            t[i] = 0;
        }
    }

    return t;
}
