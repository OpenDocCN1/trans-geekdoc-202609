```
import statsmodels.api as sm

# Explanatory variables
x = np.array([[1, 2], [3, 4], [5, 6], [7, 8]])
# Objective variables
y = np.array([1, 2, 3, 4])

model = sm.OLS(y, x)
results = model.fit()

# Find regression coefficients
print(results.summary())
```
