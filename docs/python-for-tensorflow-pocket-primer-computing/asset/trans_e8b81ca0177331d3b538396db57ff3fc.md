# 使用分隔符读取数据文件

c stuff

  1 2:6.5

# 列表 3.16: multiple-delims.py

```python
import pandas as pd

df = pd.read_csv('multidelim.dat', skiprows=3,

                 engine='python')

print("dataframe")
print(df)
print(df.head())
```
