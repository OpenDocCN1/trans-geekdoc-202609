$$\begin{aligned}
&- \eta_i \beta \mathbb{E}_{t-1} \left[ \| \phi_i^t - \phi_i^t \|^2 \right] - \frac{2}{t} \mathbb{E}_{t-1} \left[ b^T (\phi_i^t - v) \right] \\
&+ \eta_i \left[ \frac{\alpha_3 \| b \|^2}{t^{3/2-\varepsilon}} + \frac{\alpha_2(1+4v+\tau) \| b \|^2}{t^2} \right].
\end{aligned}
(1.30)$$

To demonstrate the desired result, it suffices to multiply both sides of (1.30) by $\pi_i$ and then aggregate these multiplied expressions by summing them for all $i$ from 1 to $n$. Then, we obtain

$$\begin{aligned}
\sum_{i=1}^n \pi_i \mathbb{E}_{t-1} \left[ \| p_i^t - v \|^2 \right] \\
\le (1 + \frac{\alpha_3}{t^{3/2-\varepsilon}}) \sum_{i=1}^n \pi_i \mathbb{E}_{t-1} \left[ \sum_{j=1}^n W_{ij}^t \| p_j^{t-1} - v \|^2 \right] \\
- \eta_{\min} \beta \sum_{i=1}^n \mathbb{E}_{t-1} \left[ \pi_i \| \phi_i^t - \phi_i^t \|^2 \right] \\
- \frac{2}{t} \sum_{i=1}^n \mathbb{E}_{t-1} \left[ \pi_i b^T (\phi_i^t - v) \right] \\
+ \left[ \frac{\alpha_3}{t^{3/2-\varepsilon}} + \frac{\alpha_2(1+4v+\tau)}{t^2} \right] \| b \|^2,
\end{aligned}
(1.31)$$

where we use $\eta_{\min} \le \eta_i \le 1$ with $\eta_{\min} = \min_{i \in \mathcal{V}} \{\eta_i\}$. By utilizing $\pi^T \bar{W} = \pi^T$, it yields

$$\begin{aligned}
\sum_{i=1}^n \pi_i \mathbb{E}_{t-1} \left[ \sum_{j=1}^n W_{ij}^t \| p_j^{t-1} - v \|^2 \right] \\
= \sum_{j=1}^n \| p_j^{t-1} - v \|^2 \sum_{i=1}^n \pi_i \mathbb{E}_{t-1} \left[ W_{ij}^t \right] \\
= \sum_{j=1}^n \| p_j^{t-1} - v \|^2 \sum_{i=1}^n \pi_i \bar{W}_{ij} = \sum_{i=1}^n \pi_i \| p_i^{t-1} - v \|^2.
\end{aligned}
(1.32)$$

Define $\bar{\phi}^t = \pi^T \phi^t = \sum_{i=1}^n \pi_i \phi_i^t$. Then, it is straightforward to show that

$$\sum_{i=1}^n \pi_i b^T (\phi_i^t - v) = b^T (\bar{\phi}^t - v).$$

Therefore by combining (1.31) and (1.32), we can deduce the conclusion stated in Lemma 1.5, thus completing the proof. $\square$
