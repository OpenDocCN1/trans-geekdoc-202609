as \( n \to \infty \). To this end, let \( L > 0 \) be such that \( \psi \in \text{PL}_2(r, L) \), and for each \( y \in \mathbb{R} \), define \( \psi_y, \tilde{\psi}_y : \mathbb{R} \to \mathbb{R} \) by \( \psi_y(z) := \psi(\tau_1 z, y) \) and \( \tilde{\psi}_y(z) := \psi_y(z) - \mathbb{E}(\psi_y(Z)) \), where \( Z \sim N(0, 1) \). Then by Lemma B.23(a), there exists \( K_0 > 0 \), depending only on \( \tau_1 \) and \( r \), such that \( \psi_y \in \text{PL}_1(r, K_0 L_y) \) with \( L_y := L(1 \lor |y|^{r-1}) \). For fixed \( n \in \mathbb{N} \) and \( y_1, \dots, y_n \in \mathbb{R} \), define \( \check{L} \equiv \check{L}(y_1, \dots, y_n) := (L_{y_1}, \dots, L_{y_n}) \). Let \( r' := r/(r-1) \in (1, 2] \) be the Hölder conjugate of \( r \), so that \( 1/r + 1/r' = 1 \), and note that since \( \|\cdot\|_{p'} \le \|\cdot\|_p \) for \( 1 \le p \le p' \le \infty \), we have

\(\begin{aligned} \frac{\|\check{L}\|_\infty}{n^{1/r'}} \le \frac{\|\check{L}\|_2}{n^{1/r'}} \le \frac{\|\check{L}\|_{r'}}{n^{1/r'}} = \|\check{L}\|_{n, r'} \\ = \left( \frac{1}{n} \sum_{i=1}^n |L_{y_i}|^{r'} \right)^{1/r'} \le L \left( 1 + \frac{1}{n} \sum_{i=1}^n |y_i|^r \right)^{1/r'} \\ = L(1 + \|y\|_{n, r'}^r)^{1/r'}. \end\text{(A.26)}\)

By Lemma B.12, there exists a universal constant \( C > 0 \) such that if \( Z_1, \dots, Z_n \stackrel{\text{iid}}{\sim} N(0, 1) \), then

\(\begin{aligned} P(n, t, y_1, \dots, y_n) \\ := \mathbb{P}\left( \left| \frac{1}{n} \sum_{i=1}^n \bar{\psi}_{y_i}(Z_i) \right| \ge t \right) \\ \le \exp\left( 1 - \min \left\{ \left( \frac{nt}{(Cr)^r K_0 \|\check{L}\|_2} \right)^2, \left( \frac{nt}{(Cr)^r K_0 \|\check{L}\|_\infty} \right)^{2/r} \right\} \right) \\ \le \exp\left( 1 - \min \left\{ \left( \frac{n^{1/r} t}{(Cr)^r K_0 \|\check{L}\|_{n, r'}} \right)^2, \left( \frac{n^{1/r} t}{(Cr)^r K_0 \|\check{L}\|_{n, r'}} \right)^{2/r} \right\} \right) \\ =: E_r(n, t, K_0 \|\check{L}\|_{n, r'}) \equiv E_r(n, t, K_0 \|\check{L}(y_1, \dots, y_n)\|_{n, r'}) \end\text{(A.27)}\)

for every \( t \ge 0 \). Returning to (A.25), we see that

\(\psi(\tilde{h}_i^1, \gamma_i) - \mathbb{E}(\psi(\tilde{h}_i^1, \gamma_i) | \mathcal{S}_0) = \psi_{\gamma_i}(\bar{Z}'_i) - \mathbb{E}(\psi_{\gamma_i}(\bar{Z}'_i) | \mathcal{S}_0) = \bar{\psi}_{\gamma_i}(\bar{Z}'_i)\)

for all \( 1 \le i \le n \), where the final equality follows from Lemma B.7 and the fact that \( \bar{Z}' \equiv \bar{Z}'(n) \) is independent of \( \mathcal{S}_0 \equiv \mathcal{S}_0(n) = \sigma(\gamma, m^0) \). We deduce from this and (A.27) that

\(\begin{aligned} \mathbb{P}(|T'_{n1}| > \varepsilon | \mathcal{S}_0) = \mathbb{P}\left( \left| \frac{1}{n} \sum_{i=1}^n \bar{\psi}_{\gamma_i}(\bar{Z}'_i) \right| \ge \varepsilon \middle| \mathcal{S}_0 \right) \\ = P(n, \varepsilon, \gamma_1, \dots, \gamma_n) \le E_r(n, \varepsilon, K_0 \check{L}_0(n)) \end\text{(A.28)}\)
