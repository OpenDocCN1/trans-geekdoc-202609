```
def evaluation_configuration():
    eval_config = tfma.EvalConfig(
        model_specs=[tfma.ModelSpec(signature_name='eval')],
        metrics_specs=tfma.MetricsSpec(metrics=
            [tfma.MetricConfig(class_name='ExampleCount')],

                0.5}),
```
