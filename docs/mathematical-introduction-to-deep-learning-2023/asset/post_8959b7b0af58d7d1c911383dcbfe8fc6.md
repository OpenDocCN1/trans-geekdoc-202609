
(i) it holds for all $x \in E$ that $\mathbb{E}[|Z_x|] < \infty$,
(ii) it holds that $\Omega \supseteq \omega \to \sup_{x \in E}[Z_x(\omega) - \mathbb{E}[Z_x]] \in [0, \infty)$ is $\mathcal{F}/\mathcal{B}([0, \infty))$-measurable, and
(iii) it holds for all $p \in [2, \infty)$, $r \in (0, \infty)$ that

\[
\begin{gather*}
\left( \mathbb{E}\left[ \sup_{x \in E} |Z_x - \mathbb{E}[Z_x]|^p \right] \right)^{1/p} \\
\leq 2 \big(\mathcal{C}^{(E, \delta, r)}\big)^{1/p} \left[ Lr + \frac{\sqrt{p - 1}}{\sqrt{M}} \left( \sup_{x \in E} \max_{n \in \{1, \dots, M\}} \left( \mathbb{E}\left[ |Y_{x, m} - \mathbb{E}[Y_{x, m}]|^p \right] \right)^{1/p} \right) \right] 
\end{gather*} \tag{13.29}
\]

(cf. Definition 4.3.2).

Proof of Lemma 13.2.4. Observe that the assumption that for all $x \in E$, $m \in \{1, 2, \dots, M\}$ it holds that $\mathbb{E}[|Y_{x, m}]| < \infty$ proves that for all $x \in E$ it holds that

\[
\mathbb{E}[|Z_x|] = \mathbb{E}\left[\frac{1}{M}\sum_{m = 1}^{M}Y_{x, m}\right] \leq \frac{1}{M}\sum_{m = 1}^{M}\mathbb{E}[|Y_{x, m}]| \leq \max_{m \in \{1, 2, \dots, M\}}\mathbb{E}[|Y_{x, m}]| < \infty. \tag{13.30}
\]

This establishes item (i). Note that (13.27) demonstrates that for all $x$, $y \in E$ it holds that

\[
|Z_x - Z_y| = \frac{1}{M}\left|\left[\sum_{m = 1}^{M}Y_{x, m}\right] - \left[\sum_{m = 1}^{M}Y_{y, m}\right]\right| \leq \frac{1}{M}\left[\sum_{m = 1}^{M}|Y_{x, m} - Y_{y, m}|\right] \leq L \delta(x, y). \tag{13.31}
\]

Item (i) and Lemma 12.3.5 therefore prove item (ii). It thus remains to show item (iii). For this observe that item (i), (13.31), and Lemma 13.2.3 imply that for all $p \in [1, \infty)$, $r \in (0, \infty)$ it holds that

\[
\left( \mathbb{E}\left[ \sup_{x \in E}|Z_x - \mathbb{E}[Z_x]|^p \right] \right)^{1/p} \leq \big(\mathcal{C}^{(E, \delta, r)}\big)^{1/p}\left[ 2Lr + \sup_{x \in E}\left( \mathbb{E}\left[ |Z_x - \mathbb{E}[Z_x]|^p \right] \right)^{1/p} \right] \tag{13.32}
\]

(cf. Definition 4.3.2). Furthermore, note that (13.30) and Corollary 13.1.6 (applied to $(d \frown 1, (X_m)_{m \in \{1, \dots, M\}}) \frown (Y_{x, m})_{m \in \{1, \dots, M\}}$ for $x \in E$ in the notation of Corollary 13.1.6) ensure that for all $x \in E$, $p \in [2, \infty)$, $r \in (0, \infty)$ it holds that

\[
\begin{gather*}
\left( \mathbb{E}\left[ |Z_x - \mathbb{E}[Z_x]|^p \right] \right)^{1/p} = \left( \mathbb{E}\left[\left|\frac{1}{M}\sum_{m = 1}^{M}Y_{x, m} - \mathbb{E}\left[\frac{1}{M}\sum_{m = 1}^{M}Y_{x, m}\right]\right|^p\right]\right)^{1/p} \\
\leq \frac{2\sqrt{p - 1}}{\sqrt{M}}\left[ \max_{m \in \{1, \dots, M\}}\left( \mathbb{E}\left[ |Y_{x, m} - \mathbb{E}[Y_{x, m}]|^p \right] \right)^{1/p} \right]^{1/p}. \tag{13.33}
\end{gather*}
\]

Combining this with (13.32) shows that for all $p \in [2, \infty)$, $r \in (0, \infty)$ it holds that

\[
\begin{gather*}
\left( \mathbb{E}\left[ \sup_{x \in E}|Z_x - \mathbb{E}[Z_x]|^p \right] \right)^{1/p} \\
\leq \big(\mathcal{C}^{(E, \delta, r)}\big)^{1/p}\left[ 2Lr + \frac{2\sqrt{p - 1}}{\sqrt{M}}\left( \sup_{x \in E}\max_{m \in \{1, \dots, M\}}\left( \mathbb{E}\left[ |Y_{x, m} - \mathbb{E}[Y_{x, m}]|^p \right] \right)^{1/p} \right) \right] \\
= 2\big(\mathcal{C}^{(E, \delta, r)}\big)^{1/p}\left[ Lr + \frac{\sqrt{p - 1}}{\sqrt{M}}\left( \sup_{x \in E}\max_{m \in \{1, \dots, M\}}\left( \mathbb{E}\left[ |Y_{x, m} - \mathbb{E}[Y_{x, m}]|^p \right] \right)^{1/p} \right) \right]. \tag{13.34}
\end{gather*}
\]

The proof of Lemma 13.2.4 is thus complete. \square

### Corollary 13.2.5.

Let $(E, \delta)$ be a non-empty separable metric space, let $(\Omega, \mathcal{F}, \mathbb{P})$ be a probability space, let $M \in \mathbb{N}$, for every $x \in E$ let $Y_{x, m}: \Omega \rightarrow \mathbb{R}$, $m \in \{1, 2, \ldots, M\}$, be independent random variables with $\mathbb{E}\left[|Y_{x, 1}| + |Y_{x, 2}| + \cdots + |Y_{x, m}|\right] < \infty$, let $L \in (0, \infty)$ satisfy for all $x, y \in E$, $m \in \{1, 2, \ldots, M\}$ that $|Y_{x, m} - Y_{y, m}| \leq L \delta(x, y)$, and for every $x \in E$ let $Z_{x}: \Omega \rightarrow \mathbb{R}$ satisfy

\[
Z_x = \frac{1}{M} \left[ \sum_{m=1}^{M} Y_{x,m} \right]. \tag{13.35}
\]

Then

- (i) it holds for all $x \in E$ that $\mathbb{E}[|Z_x|] < \infty$.
- (ii) it holds that $\Omega \ni \omega \supset \sup_{x \in E}|Z_x(\omega) - \mathbb{E}[Z_x]| \in [0, \infty)$ is $\mathcal{F}/\mathcal{B}([0, \infty))$-measurable, and
- (iii) it holds for all $p \in [2, \infty)$, $c \in (0, \infty)$ that

\[
\begin{aligned}
&\left(\mathbb{E}\left[\sup_{x \in E}|Z_x - \mathbb{E}[Z_x]|^{p}\right]\right)^{1 / p} \\
& \leq \frac{2 \sqrt{p-1}}{\sqrt{M}} \left(\mathcal{C}^{(E, \delta), \frac{c \sqrt{p-1}}{L \sqrt{M}}} / \frac{c \sqrt{p-1}}{L \sqrt{M}}\right)^{1 / p} \left[c + \sup_{x \in E} \max_{m \in \{1, 2, \ldots, M\}}\left(\mathbb{E}\left[|Y_{x, m} - \mathbb{E}[Y_{x, m}]|^{p}\right]\right)^{1 / p}\right] \tag{13.36}
\end{aligned}
\]

(cf. Definition 4.3.2).

Proof of Corollary 13.2.5. Observe that Lemma 13.2.4 establishes items (i) and (ii). Note that Lemma 13.2.4 (applied with $r \frown \frac{c \sqrt{p-1}}{(L \sqrt{M})}$ for $c \in (0, \infty)$ in the notation of Lemma 13.2.4) demonstrates that for all $p \in [2, \infty)$, $c \in (0, \infty)$ it holds that

\[
\begin{aligned}
&\left(\mathbb{E}\left[\sup_{x \in E}|Z_x - \mathbb{E}[Z_x]|^{p}\right]\right)^{1 / p} \\
& \leq 2 \left(\mathcal{C}^{(E, \delta), \frac{c \sqrt{p-1}}{L \sqrt{M}}} / \frac{c \sqrt{p-1}}{L \sqrt{M}}\right)^{1 / p}\left[\frac{L c \sqrt{p-1}}{L \sqrt{M}} \right. \\
& \qquad \left. + \frac{\sqrt{p-1}}{\sqrt{M}}\left(\sup_{x \in E} \max_{m \in \{1, 2, \ldots, M\}}\left(\mathbb{E}\left[|Y_{x, m} - \mathbb{E}[Y_{x, m}]|^{p}\right]\right)^{1 / p}\right)\right] \\
& = \frac{2 \sqrt{p-1}}{\sqrt{M}} \left(\mathcal{C}^{(E, \delta), \frac{c \sqrt{p-1}}{L \sqrt{M}}} / \frac{c \sqrt{p-1}}{L \sqrt{M}}\right)^{1 / p}\left[c + \sup_{x \in E} \max_{m \in \{1, 2, \ldots, M\}}\left(\mathbb{E}\left[|Y_{x, m} - \mathbb{E}[Y_{x, m}]|^{p}\right]\right)^{1 / p}\right] \tag{13.37}
\end{aligned}
\]

(cf. Definition 4.3.2). This proves item (iii). The proof of Corollary 13.2.5 is thus complete.

## 13.3 Strong convergence rates for the generalisation error

### Lemma 13.3.1.

Let $(E, \delta)$ be a separable metric space, assume $E \neq \emptyset$, let $(\Omega, \mathcal{F}, \mathbb{P})$ be a probability space, let $M \in \mathbb{N}$, let $X_{x, m}: \Omega \rightarrow \mathbb{R}$, $m \in \{1, 2, \ldots, M\}$, $x \in E$, and $Y_{m}: \Omega \rightarrow \mathbb{R}$, $m \in \{1, 2, \ldots, M\}$, be functions, assume for all $x \in E$ that $(X_{x,m}, Y_m)$, $m \in \{1, 2, \ldots, M\}$, are i.i.d. random variables, let $L, b \in (0, \infty)$ satisfy for all $x, y \in E$, $m \in \{1, 2, \ldots, M\}$ that

\[
|X_{x,m} - Y_m| \leq b \quad \text{ and } \quad |X_{x,m} - X_{y,m}| \leq L \delta(x,y), \tag{13.38}
\]

and let $\mathbf{R}: E \to [0, \infty)$ and $\mathfrak{R}: E \times \Omega \to [0, \infty)$ satisfy for all $x \in E$, $\omega \in \Omega$ that

\[
\mathbf{R}(x) = \mathbb{E}\left[|X_{x,1} - Y_1|^2\right] \quad \text{ and } \quad \mathfrak{R}(x, \omega) = \frac{1}{M} \left[\sum_{m=1}^M |X_{x,m}(\omega) - Y_m(\omega)|^2\right]. \tag{13.39}
\]

Then

- it holds that $\Omega \ni \omega \mapsto \sup_{x \in E}|\mathfrak{R}(x, \omega) - \mathbf{R}(x)| \in [0, \infty)$ is $\mathcal{F}/\mathcal{B}([0, \infty))$-measurable and
- it holds for all $p \in [2, \infty)$, $c \in (0, \infty)$ that

\[
\left(\mathbb{E}\left[\sup_{x \in E}|\mathfrak{R}(x) - \mathbf{R}(x)|^p\right]\right)^{1/p} \leq \left(c^{(E, \delta, \frac{cb\sqrt{p-1}}{2L\sqrt{M}})}\right)^{1/p} \left[\frac{2(c+1)b^2\sqrt{p-1}}{\sqrt{M}}\right]. \tag{13.40}
\]

(cf. Definition 4.3.2).

Proof of Lemma 13.3.1. Throughout this proof, for every $x \in E$, $m \in \{1, 2, \ldots, M\}$ let $\mathcal{Y}_{x,m}: \Omega \to \mathbb{R}$ satisfy $\mathcal{Y}_{x,m} = |X_{x,m} - Y_m|^2$. Observe that the assumption that for all $x \in E$ it holds that $(X_{x,m}, Y_m)$, $m \in \{1, 2, \ldots, M\}$, are i.i.d. random variables implies that for all $x \in E$ it holds that

\[
\mathbb{E}[\mathfrak{R}(x)] = \frac{1}{M} \left[\sum_{m=1}^M \mathbb{E}\left[|X_{x,m} - Y_m|^2\right]\right] = \frac{M \mathbb{E}\left[|X_{x,1} - Y_1|^2\right]}{M} = \mathbf{R}(x). \tag{13.41}
\]

Furthermore, note that the assumption that for all $x \in E$, $m \in \{1, 2, \ldots, M\}$ it holds that $|X_{x,m} - Y_m| \leq b$ shows that for all $x \in E$, $m \in \{1, 2, \ldots, M\}$ it holds that

\[
\begin{align*}
\mathbb{E}\left[|\mathcal{Y}_{x,m}|\right] &= \mathbb{E}\left[|X_{x,m} - Y_m|^2\right] \leq b^2 < \infty, \tag{13.42} \\
\mathcal{Y}_{x,m} - \mathbb{E}[\mathcal{Y}_{x,m}] &= |X_{x,m} - Y_m|^2 - \mathbb{E}\left[|X_{x,m} - Y_m|^2\right] \leq |X_{x,m} - Y_m|^2 \leq b^2, \tag{13.43}
\end{align*}
\]

and

\[
\mathbb{E}[\mathcal{Y}_{x,m}] - \mathcal{Y}_{x,m} = \mathbb{E}\left[|X_{x,m} - Y_m|^2\right] - |X_{x,m} - Y_m|^2 \leq \mathbb{E}\left[|X_{x,m} - Y_m|^2\right] \leq b^2. \tag{13.44}
\]

Observe that (13.42), (13.43), and (13.44) ensure for all $x \in E$, $m \in \{1, 2, \ldots, M\}$, $p \in (0, \infty)$ that

\[
\left(\mathbb{E}\left[|\mathcal{Y}_{x,m} - \mathbb{E}[\mathcal{Y}_{x,m}]|^p\right]\right)^{1/p} \leq \left(\mathbb{E}\left[b^{2p}\right]\right)^{1/p} = b^2. \tag{13.45}
\]
