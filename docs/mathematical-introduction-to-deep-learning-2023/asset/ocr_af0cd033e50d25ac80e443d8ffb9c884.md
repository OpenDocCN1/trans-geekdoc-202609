(cf. Definition 2.2.8. Items (i) and (ii) in Proposition 2.2.3 therefore imply

+   (A) that
\[
\mathcal{R}_{a}^{\mathbf{N}}\big(\mathbf{P}_{n}(\mathcal{E}_{L, l_{1}}(\Phi_{1}), \mathcal{E}_{L, l_{2}}(\Phi_{2}), \ldots, \mathcal{E}_{L, l_{n}}(\Phi_{n}))\big) \in C\big(\mathbb{R}^{[\sum_{j=1}^{n} \mathcal{I}(\Phi_{j})]}, \mathbb{R}^{[\sum_{j=1}^{n} \mathcal{O}(\Phi_{j})]}\big)
\tag{2.110}
\]
and

+   (B) that for all $x_{1} \in \mathbb{R}^{\mathcal{I}(\Phi_{1})}, x_{2} \in \mathbb{R}^{\mathcal{I}(\Phi_{2})}, \ldots, x_{n} \in \mathbb{R}^{\mathcal{I}(\Phi_{n})}$ it holds that
\[
\begin{aligned}
& \left(\mathcal{R}_{a}^{\mathbf{N}}\left(\mathbf{P}_{n}\left(\mathcal{E}_{L, l_{1}}\left(\Phi_{1}\right), \mathcal{E}_{L, l_{2}}\left(\Phi_{2}\right), \ldots, \mathcal{E}_{L, l_{n}}\left(\Phi_{n}\right)\right)\right)\right)\left(x_{1}, x_{2}, \ldots, x_{n}\right) \\
& \quad=\left(\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\mathcal{E}_{L, l_{1}}\left(\Phi_{1}\right)\right)\right)\left(x_{1}\right),\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\mathcal{E}_{L, l_{2}}\left(\Phi_{2}\right)\right)\right)\left(x_{2}\right), \ldots,\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\mathcal{E}_{L, l_{n}}\left(\Phi_{n}\right)\right)\right)\left(x_{n}\right)\right) \\
& \quad=\left(\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\Phi_{1}\right)\right)\left(x_{1}\right),\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\Phi_{2}\right)\right)\left(x_{2}\right), \ldots,\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\Phi_{n}\right)\right)\left(x_{n}\right)\right)
\end{aligned}\tag{2.111}
\]
(cf. Definition 2.2.1). Combining this with (2.106) and the fact that $L=\max _{j \in\{1, \ldots, n\}} \mathcal{L}(\Phi_{j})$ ensures

+   (C) that
\[
\mathcal{R}_{a}^{\mathbf{N}}\left(\mathbf{P}_{n, \mathbb{I}}(\Phi)\right) \in C\big(\mathbb{R}^{[\sum_{j=1}^{n} \mathcal{I}(\Phi_{j})]}, \mathbb{R}^{[\sum_{j=1}^{n} \mathcal{O}(\Phi_{j})]}\big)\tag{2.112}
\]
and

+   (D) that for all $x_{1} \in \mathbb{R}^{\mathcal{I}(\Phi_{1})}, x_{2} \in \mathbb{R}^{\mathcal{I}(\Phi_{2})}, \ldots, x_{n} \in \mathbb{R}^{\mathcal{I}(\Phi_{n})}$ it holds that
\[
\begin{aligned}
& \left(\mathcal{R}_{a}^{\mathbf{N}}\left(\mathbf{P}_{n, \mathbb{I}}(\Phi)\right)\right)\left(x_{1}, x_{2}, \ldots, x_{n}\right) \\
& \quad=\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\mathbf{P}_{n}\left(\mathcal{E}_{L, l_{1}}\left(\Phi_{1}\right), \mathcal{E}_{L, l_{2}}\left(\Phi_{2}\right), \ldots, \mathcal{E}_{L, l_{n}}\left(\Phi_{n}\right)\right)\right)\right)\left(x_{1}, x_{2}, \ldots, x_{n}\right) \\
& \quad=\left(\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\Phi_{1}\right)\right)\left(x_{1}\right),\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\Phi_{2}\right)\right)\left(x_{2}\right), \ldots,\left(\mathcal{R}_{a}^{\mathbf{N}}\left(\Phi_{n}\right)\right)\left(x_{n}\right)\right).
\end{aligned}\tag{2.113}
\]

This establishes items items (i) and (ii). The proof of Lemma 2.2.13 is thus complete.
