
This proves item (ii). Moreover, note that (6.335), (6.336), (6.341), and (6.342) demonstrate that for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) it holds that

\begin{equation}
\begin{split}
\Theta_n^{(i)} &= \Theta_{n-1}^{(i)} - \gamma_n \left[ \varepsilon + \mathbb{M}_{n}^{(i)} \right]^{-1/2} \mathcal{G}_i(\Theta_{n-1}) \\
&= \Theta_{n-1}^{(i)} - \gamma_n \left[ \varepsilon + \sum_{k=0}^{n-1} b_{n,k} [\mathcal{G}_i(\Theta_k)]^2 \right]^{-1/2} \mathcal{G}_i(\Theta_{n-1}).
\end{split}
\tag{6.343}
\end{equation}

This establishes item (iii). The proof of Lemma 6.6.2 is thus complete.

### 6.6.2 Bias-adjusted root mean square propagation \(\mathbf{GD}\) optimization

**Definition 6.6.3 (Bias-adjusted \(\mathbf{RMSprop}\) \(\mathbf{GD}\) optimization method).** Let \(\mathfrak{d} \in \mathbb{N}\), \((\gamma_n)_{n \in \mathbb{N}} \subseteq [0, \infty)\), \((\beta_n)_{n \in \mathbb{N}} \subseteq [0, 1]\), \(\varepsilon \in (0, \infty)\), \(\xi \in \mathbb{R}^\mathfrak{d}\) satisfy

\begin{equation}
\beta_1 < 1
\tag{6.344}
\end{equation}

and let \(\mathcal{L} : \mathbb{R}^\mathfrak{d} \to \mathbb{R}\) and \(\mathcal{G} = (\mathcal{G}_1, \ldots, \mathcal{G}_\mathfrak{d}) : \mathbb{R}^\mathfrak{d} \to \mathbb{R}^\mathfrak{d}\) satisfy for all \(U \in \{V \subseteq \mathbb{R}^\mathfrak{d} : V \text{ is open}\}\), \(\theta \in U\) with \(\mathcal{L}|_U \in C^1(U, \mathbb{R}^\mathfrak{d})\) that

\begin{equation}
\mathcal{G}(\theta) = (\nabla \mathcal{L})(\theta).
\tag{6.345}
\end{equation}

Then we say that \(\Theta\) is the bias-adjusted \(\mathbf{RMSprop}\) \(\mathbf{GD}\) process for the objective function \(\mathcal{L}\) with generalized gradient \(\mathcal{G}\), learning rates \((\gamma_n)_{n \in \mathbb{N}}\), second moment decay factors \((\beta_n)_{n \in \mathbb{N}}\), regularizing factor \(\varepsilon\), and initial value \(\xi\) (we say that \(\Theta\) is the bias-adjusted \(\mathbf{RMSprop}\) \(\mathbf{GD}\) process for the objective function \(\mathcal{L}\) with learning rates \((\gamma_n)_{n \in \mathbb{N}}\), second moment decay factors \((\beta_n)_{n \in \mathbb{N}}\), regularizing factor \(\varepsilon\), and initial value \(\xi\)) if and only if there holds that \(\Theta = (\Theta^{(1)}, \ldots, \Theta^{(\mathfrak{d})}) : \mathbb{N}_0 \to \mathbb{R}^\mathfrak{d}\) is the function from \(\mathbb{N}_0\) to \(\mathbb{R}^\mathfrak{d}\) which satisfies that there exists \(\mathbb{M} = (\mathbb{M}^{(1)}, \ldots, \mathbb{M}^{(\mathfrak{d})}) : \mathbb{N}_0 \to \mathbb{R}^\mathfrak{d}\) such that for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) it holds that

\begin{equation}
\Theta_0 = \xi, \quad \mathbb{M}_0 = 0, \quad \mathbb{M}_n^{(i)} = \beta_n \mathbb{M}_{n-1}^{(i)} + (1 - \beta_n) [\mathcal{G}_i(\Theta_{n-1})]^2,
\tag{6.346}
\end{equation}

and

\begin{equation}
\Theta_n^{(i)} = \Theta_{n-1}^{(i)} - \gamma_n \left[ \varepsilon + \frac{\mathbb{M}_n^{(i)}}{1 - \prod_{k=1}^{n} \beta_k} \right]^{1/2}^{-1} \mathcal{G}_i(\Theta_{n-1}).
\tag{6.347}
\end{equation}

**Lemma 6.6.4 (On a representation of the second order terms in bias-adjusted \(\mathbf{RMSprop}\).** Let \(\mathfrak{d} \in \mathbb{N}\), \((\gamma_n)_{n \in \mathbb{N}} \subseteq [0, \infty)\), \((\beta_n)_{n \in \mathbb{N}} \subseteq [0, 1]\), \((b_{n,k})_{(n,k) \in (\mathbb{N}_0)^2} \subseteq \mathbb{R}\), \(\varepsilon \in (0, \infty)\), \(\xi \in \mathbb{R}^\mathfrak{d}\) satisfy for all \(n \in \mathbb{N}\), \(k \in \{0, 1, \ldots, n - 1\}\) that

\begin{equation}
b_{n,k} = \frac{(1 - \beta_{k+1}) \prod_{\ell = k+2}^{n} \beta_\ell}{1 - \prod_{k=1}^{n} \beta_k},
\tag{6.348}
\end{equation}

let \(\mathcal{L}: \mathbb{R}^\mathfrak{d} \to \mathbb{R}\) and \(\mathcal{G} = (\mathcal{G}_1, \ldots, \mathcal{G}_\mathfrak{d}): \mathbb{R}^\mathfrak{d} \to \mathbb{R}^\mathfrak{d}\) satisfy for all \(U \in \{V \subseteq \mathbb{R}^\mathfrak{d}: V \text{ is open}\}\), \(\theta \in U\) with \(\mathcal{L}|_U \in C^1(U, \mathbb{R}^\mathfrak{d})\) that

\begin{equation}
\mathcal{G}(\theta) = (\nabla \mathcal{L})(\theta), \tag{6.349}
\end{equation}

and let \(\Theta = (\Theta^{(1)}, \ldots, \Theta^{(\mathfrak{d})}): \mathbb{N}_0 \to \mathbb{R}^\mathfrak{d}\) be the bias-adjusted RMSprop GD process for the objective function \(\mathcal{L}\) with generalized gradient \(\mathcal{G}\), learning rates \((\gamma_n)_{n \in \mathbb{N}}\), second moment decay factors \((\beta_n)_{n \in \mathbb{N}}\), regularizing factor \(\varepsilon\), and initial value \(\xi\) (cf. Definition 6.6.3). Then

- \((i)\) it holds for all \(n \in \mathbb{N}\), \(k \in \{0, 1, \ldots, n-1\}\) that \(0 \le b_{n,k} \le 1\),
- \((ii)\) it holds for all \(n \in \mathbb{N}\) that

\begin{equation}
\sum_{k=0}^{n-1} b_{n,k} = 1, \tag{6.350}
\end{equation}

and

- \((iii)\) it holds for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) that

\begin{equation}
\Theta^{(i)}_n = \Theta^{(i)}_{n-1} - \gamma_n \left[ \varepsilon + \left[\sum_{k=0}^{n-1} b_{n,k} |\mathcal{G}_i(\Theta_k)|^2 \right]^{1/2} \right]^{-1} \mathcal{G}_i(\Theta_{n-1}). \tag{6.351}
\end{equation}

*Proof of Lemma 6.6.4.* Throughout this proof, let \(\mathbb{M} = (\mathbb{M}^{(1)}, \ldots, \mathbb{M}^{(\mathfrak{d})}): \mathbb{N}_0 \to \mathbb{R}^\mathfrak{d}\) satisfy for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) that \(\mathbb{M}^{(i)}_0 = 0\) and

\begin{equation}
\mathbb{M}^{(i)}_n = \beta_n \mathbb{M}^{(i)}_{n-1} + (1-\beta_n) |\mathcal{G}_i(\Theta_{n-1})|^2 \tag{6.352}
\end{equation}

and let \((B_{n,k})_{(n,k) \in (\mathbb{N}_0)^2} \subseteq \mathbb{R}\) satisfy for all \(n \in \mathbb{N}\), \(k \in \{0, 1, \ldots, n-1\}\) that

\begin{equation}
B_{n,k} = (1-\beta_{k+1}) \left[\prod_{l=k+2}^{n} \beta_l \right]. \tag{6.353}
\end{equation}

Observe that (6.348) implies item \((i)\). Note that \((6.348)\), \((6.352)\), \((6.353)\), and Lemma 6.3.3 assure that for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) it holds that

\begin{equation}
\mathbb{M}^{(i)}_n = \sum_{k=0}^{n-1} B_{n,k} |\mathcal{G}_i(\Theta_k)|^2 \quad \text{and} \quad \sum_{k=0}^{n-1} b_{n,k} = \frac{\sum_{k=0}^{n-1} B_{n,k}}{1-\prod_{k=1}^{n} \beta_k} = \frac{1-\prod_{k=1}^{n} \beta_k}{1-\prod_{k=1}^{n} \beta_k} = 1. \tag{6.354}
\end{equation}

This proves item \((ii)\). Observe that \((6.346)\), \((6.347)\), \((6.352)\), and \((6.354)\) demonstrate that for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) it holds that

\begin{equation}
\begin{aligned}
\Theta^{(i)}_n &= \Theta^{(i)}_{n-1} - \gamma_n \left[ \varepsilon + \left[\frac{\mathbb{M}^{(i)}_n}{1-\prod_{k=1}^{n} \beta_k} \right]^{1/2} \right]^{-1} \mathcal{G}_i(\Theta_{n-1}) \tag{6.355} \\
&= \Theta^{(i)}_{n-1} - \gamma_n \left[ \varepsilon + \left[\sum_{k=0}^{n-1} b_{n,k} |\mathcal{G}_i(\Theta_k)|^2 \right]^{1/2} \right]^{-1} \mathcal{G}_i(\Theta_{n-1}).
\end{aligned}
\end{equation}

This establishes item \((iii)\). The proof of Lemma 6.6.4 is thus complete.

### 6.7 AdaDelta GD optimization

The AdaDelta GD optimization method reviewed in this section is an extension of the RMSprop GD optimization method. Like the RMSprop GD optimization method, the AdaDelta GD optimization method adapts the learning rates for every component of the optimization process separately. To do this, the AdaDelta GD optimization method uses two exponentially decaying averages: one over the squares of the past partial derivatives of the objective function as does the RMSprop GD optimization method (cf. (6.358) below) and another one over the squares of the past increments (cf. (6.360) below). As in the case of Adagrad and RMSprop optimization, AdaDelta optimization was introduced in a stochastic setting (see Zeiler [429]), but for pedagogical purposes we present in this section a deterministic version of AdaDelta optimization. We refer to Section 7.8 below for the original stochastic version of AdaDelta optimization.

**Definition 6.7.1 (AdaDelta GD optimization method).** Let \(\mathfrak{d} \in \mathbb{N}\), \((\beta_n)_{n \in \mathbb{N}} \subseteq [0, 1]\), \((\delta_n)_{n \in \mathbb{N}} \subseteq [0, 1]\), \(\varepsilon \in (0, \infty)\), \(\xi \in \mathbb{R}^\mathfrak{d}\) and let \(\mathcal{L}: \mathbb{R}^\mathfrak{d} \rightarrow \mathbb{R}\) and \(\mathcal{G} = (\mathcal{G}_1, \ldots, \mathcal{G}_\mathfrak{d}): \mathbb{R}^\mathfrak{d} \rightarrow \mathbb{R}^\mathfrak{d}\) satisfy for all \(U \in \{V \subseteq \mathbb{R}^\mathfrak{d}: V \text{ is open}\}\), \(\theta \in U\) with \(\mathcal{L}|_U \in C^1(U, \mathbb{R}^\mathfrak{d})\) that

\begin{equation}
\mathcal{G}(\theta) = (\nabla\mathcal{L})(\theta). \tag{6.356}
\end{equation}

Then we say that \(\Theta\) is the AdaDelta GD process for the objective function \(\mathcal{L}\) with generalized gradient \(\mathcal{G}\), second moment decay factors \((\beta_n)_{n \in \mathbb{N}}\), delta decay factors \((\delta_n)_{n \in \mathbb{N}}\), regularizing factor \(\varepsilon\), and initial value \(\xi\) (we say that \(\Theta\) is the AdaDelta GD process for the objective function \(\mathcal{L}\) with second moment decay factors \((\beta_n)_{n \in \mathbb{N}}\), delta decay factors \((\delta_n)_{n \in \mathbb{N}}\), regularizing factor \(\varepsilon\), and initial value \(\xi\)) if and only if it holds that \(\Theta = (\Theta^{(1)}, \ldots, \Theta^{(\mathfrak{d})}): \mathbb{N}_0 \rightarrow \mathbb{R}^\mathfrak{d}\) is the function from \(\mathbb{N}_0\) to \(\mathbb{R}^\mathfrak{d}\) which satisfies that there exist \(\mathbb{M} = (\mathbb{M}^{(1)}, \ldots, \mathbb{M}^{(\mathfrak{d})}): \mathbb{N}_0 \rightarrow \mathbb{R}^\mathfrak{d}\) and \(\Delta = (\Delta^{(1)}, \ldots, \Delta^{(\mathfrak{d})}): \mathbb{N}_0 \rightarrow \mathbb{R}^\mathfrak{d}\) such that for all \(n \in \mathbb{N}\), \(i \in \{1, 2, \ldots, \mathfrak{d}\}\) it holds that

\begin{equation}
\Theta_0 = \xi, \quad \mathbb{M}_0 = 0, \quad \Delta_0 = 0, \tag{6.357}
\end{equation}

\begin{equation}
\mathbb{M}^{(i)}_n = \beta_n \mathbb{M}^{(i)}_{n-1} + (1 - \beta_n) |\mathcal{G}_i(\Theta_{n-1})|^2, \tag{6.358}
\end{equation}

\begin{equation}
\Theta^{(i)}_n = \Theta^{(i)}_{n-1} - \left[\frac{\varepsilon + \Delta^{(i)}_{n-1}}{\varepsilon + \mathbb{M}^{(i)}_n}\right]^{1/2} \mathcal{G}_i(\Theta_{n-1}), \tag{6.359}
\end{equation}

and

\begin{equation}
\Delta^{(i)}_n = \delta_n \Delta^{(i)}_{n-1} + (1 - \delta_n) |\Theta^{(i)}_n - \Theta^{(i)}_{n-1}|^2. \tag{6.360}
\end{equation}
