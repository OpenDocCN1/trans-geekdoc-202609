
Furthermore, observe that (8.1) and the fact that $D_{L+1} = \mathbf{I}_{L}$ ensure that

$$
\begin{align*}
\left(\frac{\partial \mathbf{f}_1}{\partial \boldsymbol{\theta}_L}\right)(\vartheta, \mathbf{z}_0) &= (\mathbb{R}^{n_{L}} \ni \theta_{L} \mapsto F_{L}(\theta_{L}, \mathbf{f}_{L-1})) \in \mathbb{R}^{l_{1}}{}^{\prime}(\vartheta_{L}) \\
&= \left[\frac{\partial F_{L}}{\partial \theta_{L}}\right](\vartheta_{L}, \mathbf{f}_{L-1}) \tag{8.13} \\
&= D_{L+1}\left[\frac{\partial F_{L}}{\partial \theta_{L}}(\vartheta_{L}, \mathbf{f}_{L-1})\right].
\end{align*}
$$

Combining this and (8.12) establishes item (iii). The proof of Proposition 8.1.1 is thus complete.

## Corollary 8.1.2 (Backpropagation for parametric functions with loss).

Let $L \in \mathbb{N}$, $l_0, l_1, \ldots, l_L, \mathbf{d}_1, \mathbf{d}_2, \ldots, \mathbf{d}_L \in \mathbb{N}$, $\vartheta = (\vartheta_1, \vartheta_2, \ldots, \vartheta_L) \in \mathbb{R}^{\mathbf{d}_1} \times \mathbb{R}^{\mathbf{d}_2} \times \cdots \times \mathbb{R}^{\mathbf{d}_L}$, $\mathbf{z}_0 \in \mathbb{R}^{l_0}$, $\mathbf{z}_1 \in \mathbb{R}^{l_1}$, $\ldots$, $\mathbf{z}_L \in \mathbb{R}^{l_L}$, $\mathbf{y} \in \mathbb{R}^{l_L}$, let $\mathfrak{C} = (\mathfrak{C}(x, y))_{(x, y) \in \mathbb{R}^{l_L} \times \mathbb{R}^{l_L}} : \mathbb{R}^{l_L} \times \mathbb{R}^{l_L} \rightarrow \mathbb{R}$ be differentiable, for every $k \in \{1, 2, \ldots, L\}$ let $F_k = F_k(\theta_k, \mathbf{x}_{k-1})_{(\theta_k, \mathbf{x}_{k-1}) \in \mathbb{R}^{\mathbf{d}_k} \times \mathbb{R}^{l_{k-1}}} : \mathbb{R}^{\mathbf{d}_k} \times \mathbb{R}^{l_{k-1}} \rightarrow \mathbb{R}^{l_k}$ be differentiable, let $\mathcal{L} = (\mathcal{L}(\theta_1, \theta_2, \ldots, \theta_L))_{(\theta_1, \theta_2, \ldots, \theta_L) \in \mathbb{R}^{\mathbf{d}_1} \times \mathbb{R}^{\mathbf{d}_2} \times \cdots \times \mathbb{R}^{\mathbf{d}_L}} : \mathbb{R}^{\mathbf{d}_1} \times \mathbb{R}^{\mathbf{d}_2} \times \cdots \times \mathbb{R}^{\mathbf{d}_L} \rightarrow \mathbb{R}$ satisfy for all $\theta = (\theta_1, \theta_2, \ldots, \theta_L) \in \mathbb{R}^{\mathbf{d}_1} \times \mathbb{R}^{\mathbf{d}_2} \times \cdots \times \mathbb{R}^{\mathbf{d}_L}$ that

$$
\mathcal{L}(\theta) = (\mathfrak{C}(:, \mathbf{y}) \circ F_L(\theta_L, \cdot) \circ F_{L-1}(\theta_{L-1}, \cdot) \circ \cdots \circ F_1(\theta_1, \cdot))(\mathbf{z}_0), \tag{8.14}
$$

assume for all $k \in \{1, 2, \ldots, L\}$ that

$$
\mathbf{f}_k = F_k(\vartheta_k, \mathbf{f}_{k-1}), \tag{8.15}
$$

and let $D_k \in \mathbb{R}^{l_{k-1}}$, $k \in \{1, 2, \ldots, L+1\}$, satisfy for all $k \in \{1, 2, \ldots, L\}$ that

$$
D_{L+1} = (\nabla_{\mathbf{x}}\mathfrak{C})(\mathbf{f}_L, \mathbf{y}) \quad \text{and} \quad D_k = \left[\left(\frac{\partial F_k}{\partial \mathbf{x}_{k-1}}\right)(\vartheta_k, \mathbf{f}_{k-1})\right]^{*} D_{k+1}. \tag{8.16}
$$

Then

- (i) it holds that $\mathcal{L} : \mathbb{R}^{\mathbf{d}_1} \times \mathbb{R}^{\mathbf{d}_2} \times \cdots \times \mathbb{R}^{\mathbf{d}_L} \rightarrow \mathbb{R}$ is differentiable and
- (ii) it holds for all $k \in \{1, 2, \ldots, L\}$ that

$$
(\nabla_{\vartheta_k}\mathcal{L})(\vartheta) = \left[\left(\frac{\partial F_k}{\partial \theta_k}\right)(\vartheta_k, \mathbf{f}_{k-1})\right]^{*} D_{k+1}. \tag{8.17}
$$

Proof of Corollary 8.1.2. Throughout this proof, let $\mathbf{D}_k \in \mathbb{R}^{l_L \times l_{k-1}}$, $k \in \{1, 2, \ldots, L+1\}$, satisfy for all $k \in \{1, 2, \ldots, L\}$ that $\mathbf{D}_{L+1} = \mathbf{I}_L$ and

$$
\mathbf{D}_k = \mathbf{D}_{k+1}\left[\left(\frac{\partial F_k}{\partial \mathbf{x}_{k-1}}\right)(\vartheta_k, \mathbf{f}_{k-1})\right] \tag{8.18}
$$

and let $f = (f(\theta_1, \theta_2, \ldots, \theta_L))_{(\theta_1, \theta_2, \ldots, \theta_L) \in \mathbb{R}^{\ell_1} \times \mathbb{R}^{\ell_2} \times \cdots \times \mathbb{R}^{\ell_L}} : \mathbb{R}^{\ell_1} \times \mathbb{R}^{\ell_2} \times \cdots \times \mathbb{R}^{\ell_L} \rightarrow \mathbb{R}^{\ell_L}$ satisfy for all $\theta = (\theta_1, \theta_2, \ldots, \theta_L) \in \mathbb{R}^{\ell_1} \times \mathbb{R}^{\ell_2} \times \cdots \times \mathbb{R}^{\ell_L}$ that

$$
f(\theta) = \left(F_L(\theta_L, \cdot) \circ F_{L-1}(\theta_{L-1}, \cdot) \circ \cdots \circ F_1(\theta_1, \cdot)\right)(\mathfrak{f}_0)
$$

(cf. Definition 1.5.5). Note that item (**i**) in Proposition 8.1.1 ensures that $f : \mathbb{R}^{\ell_1} \times \mathbb{R}^{\ell_2} \times \cdots \times \mathbb{R}^{\ell_L} \rightarrow \mathbb{R}^{\ell_L}$ is differentiable. Thus, the assumption that $\mathfrak{C} : \mathbb{R}^{\ell_L} \times \mathbb{R}^{\ell_L} \rightarrow \mathbb{R}$ is differentiable, and the fact that $\mathcal{L} = \mathfrak{C}(\cdot, \mathfrak{f}) \circ f$ ensure that $\mathcal{L} : \mathbb{R}^{\ell_1} \times \mathbb{R}^{\ell_2} \times \cdots \times \mathbb{R}^{\ell_L} \rightarrow \mathbb{R}$ is differentiable. This establishes item (**i**). Next we claim that for all $k \in \{1, 2, \ldots, L+1\}$ it holds that

$$
[D_k]^* = \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbf{D}_k. \tag{8.19}
$$

We now prove (8.19) by induction on $k \in \{L+1, L, \ldots, 1\}$. For the base case $k = L+1$ note that (8.16) and (8.18) assure that

$$
\begin{align*}
[D_{L+1}]^* &= [(\nabla_x \mathfrak{C})(\mathfrak{f}_L, \mathfrak{f})]^* = \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix} \\
&= \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbb{I}_{\ell_L} = \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbf{D}_{L+1}.
\end{align*}
$$

This establishes (8.19) in the base case $k = L+1$. For the induction step observe (8.16) and (8.18) demonstrate that for all $k \in \{L, L-1, \ldots, 1\}$ with $[D_{k+1}]^* = \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbf{D}_{k+1}$ it holds that

$$
\begin{align*}
[D_k]^* &= [D_{k+1}]^* \begin{bmatrix} \left(\frac{\partial F_k}{\partial x_{k-1}}\right)(\vartheta_k, \mathfrak{f}_{k-1}) \end{bmatrix} \\
&= \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbf{D}_{k+1} \begin{bmatrix} \left(\frac{\partial F_k}{\partial x_{k-1}}\right)(\vartheta_k, \mathfrak{f}_{k-1}) \end{bmatrix} = \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbf{D}_k.
\end{align*}
$$

Induction thus establishes (8.19). Furthermore, note that item (**iii**) in Proposition 8.1.1 assures that for all $k \in \{1, 2, \ldots, L\}$ it holds that

$$
\left(\frac{\partial f}{\partial \theta_k}\right)(\vartheta) = \mathbf{D}_{k+1} \begin{bmatrix} \left(\frac{\partial F_k}{\partial \theta_k}\right)(\vartheta_k, \mathfrak{f}_{k-1}) \end{bmatrix}. \tag{8.20}
$$

Combining this with chain rule, the fact that $\mathcal{L} = \mathfrak{C}(\cdot, \mathfrak{f}) \circ f$, and (8.19) ensures that for all $k \in \{1, 2, \ldots, L\}$ it holds that

$$
\begin{align*}
\left(\frac{\partial \mathcal{L}}{\partial \theta_k}\right)(\vartheta) &= \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(f(\vartheta), \mathfrak{f}) \end{bmatrix} \begin{bmatrix} \left(\frac{\partial f}{\partial \theta_k}\right)(\vartheta) \end{bmatrix} \\
&= \begin{bmatrix} \frac{\partial \mathfrak{C}}{\partial x}(\mathfrak{f}_L, \mathfrak{f}) \end{bmatrix}\mathbf{D}_{k+1} \begin{bmatrix} \left(\frac{\partial F_k}{\partial \theta_k}\right)(\vartheta_k, \mathfrak{f}_{k-1}) \end{bmatrix} \\
&= [D_{k+1}]^* \begin{bmatrix} \left(\frac{\partial F_k}{\partial \theta_k}\right)(\vartheta_k, \mathfrak{f}_{k-1}) \end{bmatrix}.
\end{align*}
$$

Hence, we obtain that for all $k \in \{1, 2, \dots, L\}$ it holds that

$$
\big(\nabla_{\mathfrak{b}_k} \mathcal{L}\big)(\vartheta)=\biggl[\biggl(\frac{\partial \mathcal{L}}{\partial \mathfrak{b}_k}\biggr)(\vartheta)\biggr]^* = \biggl[\biggl(\frac{\partial F_k}{\partial \mathfrak{b}_k}\biggr)(\vartheta_k, \mathfrak{f}_{k-1})\biggr]^* D_{k+1}.
$$

This establishes item (ii). The proof of Corollary 8.1.2 is thus complete.
