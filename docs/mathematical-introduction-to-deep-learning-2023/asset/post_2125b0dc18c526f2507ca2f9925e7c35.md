
Moreover, note that (13.38) and the fact that for all $x_1, x_2, y \in \mathbb{R}$ it holds that $(x_1 - y)^2 - (x_2 - y)^2 = (x_1 - x_2)((x_1 - y) + (x_2 - y))$ show that for all $x, y \in E$, $m \in \{1, 2, \ldots, M\}$ it holds that

$$
\begin{align*}
|\mathcal{Y}_{x,m} - \mathcal{Y}_{y,m}| &= |(X_{x,m} - Y_m)^2 - (X_{y,m} - Y_m)^2| \\
&\leq |X_{x,m} - X_{y,m}|(|X_{x,m} - Y_m| + |X_{y,m} - Y_m|) \\
&\leq 2b|X_{x,m} - X_{y,m}| \leq 2b\delta d(x, y).
\end{align*}
$$

The fact that for all $x \in E$ it holds that $\mathcal{Y}_{x,m}$, $m \in \{1, 2, \ldots, M\}$, are independent random variables, (13.42), and Corollary 13.2.5 (applied with $(\tilde{\mathcal{Y}}_{x,m})_{x \in E, m \in \{1, 2, \ldots, M\}}$ and $(\mathcal{Y}_{x,m})_{x \in E, m \in \{1, 2, \ldots, M\}}$, $L \lor 2b\breve{L}$, $(Z_{z})_{z \in E} \lor (\Omega \ni \omega \mapsto \mathscr{R}(x, \omega) \in \mathbb{R})_{x \in E}$ in the notation of Corollary 13.2.5) hence establish that

- (I) it holds that $\Omega \ni \omega \mapsto \sup_{x \in E}|\mathscr{R}(x, \omega) - \mathbf{R}(x)| \in [0, \infty]$ is $\mathcal{F}/\mathcal{B}([0, \infty])$-measurable and
- (II) it holds for all $p \in [2, \infty)$, $c \in (0, \infty)$ that

$$
\left( \mathbb{E}\left[ \sup_{x \in E}|\mathscr{R}(x) - \mathbb{E}[\mathscr{R}(x)]|^p \right] \right)^{1/p} \leq \frac{2\sqrt{p-1}}{\sqrt{M}} \left( c^{(E,\delta)} \cdot \frac{cb^2 \sqrt{p-1}}{2L\sqrt{M}} \right)^{1/p} \left[ cb^2 + \sup_{x \in E} \max_{m \in \{1, \ldots, M\}} \left( \mathbb{E}\left[ |\mathcal{Y}_{x,m} - \mathbb{E}[\mathcal{Y}_{x,m}]|^p \right] \right)^{1/p} \right].
$$

Observe that item (II), (13.41), (13.42), and (13.45) demonstrate that for all $p \in [2, \infty)$, $c \in (0, \infty)$ it holds that

$$
\left( \mathbb{E}\left[ \sup_{x \in E}|\mathscr{R}(x) - \mathbf{R}(x)|^p \right] \right)^{1/p} \leq \frac{2\sqrt{p-1}}{\sqrt{M}} \left( c^{(E,\delta)} \cdot \frac{cb^2 \sqrt{p-1}}{2L\sqrt{M}} \right)^{1/p} \left[ cb^2 + b^2 \right]
$$

$$
= \left( c^{(E,\delta)} \cdot \frac{cb^2 \sqrt{p-1}}{2L\sqrt{M}} \right)^{1/p} \frac{\left[ 2(c+1)b^2 \sqrt{p-1} - 1 \right]}{\sqrt{M}}.
$$

This and item (I) prove items (i) and (ii). The proof of Lemma 13.3.1 is thus complete.

### Proposition 13.3.2.

Let $d \in \mathbb{N}$, $D \subseteq \mathbb{R}^d$, let $(\Omega, \mathcal{F}, \mathbb{P})$ be a probability space, let $M \in \mathbb{N}$, let $\overline{\mathbf{X}}_m = (X_m, Y_m) : \Omega \to (D \times \mathbb{R})$, $m \in \{1, 2, \ldots, M\}$, be i.i.d. random variables, let $\alpha \in \mathbb{R}$, $\beta \in (\alpha, \infty)$, $\mathbf{d} \in \mathbb{N}$, let $f = (f_\theta)_{\theta \in [\alpha, \beta]} : [\alpha, \beta] \to C(D, \mathbb{R})$, let $L, b \in (0, \infty)$ satisfy for all $\theta, \vartheta \in [\alpha, \beta]^{\mathbf{d}}$, $m \in \{1, 2, \ldots, M\}$, $x \in D$ that

$$
|f_\theta(X_m) - Y_m| \leq b \quad \text{ and } \quad |f_\theta(x) - f_\vartheta(x)| \leq L|\theta - \vartheta|_\infty,
$$

and let $\mathbf{R} : [\alpha, \beta]^{\mathbf{d}} \to [0, \infty)$ and $\mathscr{R} : [\alpha, \beta]^{\mathbf{d}} \times \Omega \to [0, \infty)$ satisfy for all $\theta \in [\alpha, \beta]^{\mathbf{d}}$, $\omega \in \Omega$ that

$$
\mathbf{R}(\theta) = \mathbb{E}\left[ |f_\theta(X_1) - Y_1|^2 \right] \quad \text{ and } \quad \mathscr{R}(\theta, \omega) = \frac{1}{M} \left[ \sum_{m=1}^{M} |f_\theta(X_m(\omega)) - Y_m(\omega)|^2 \right]
$$

(cf. Definition 3.3.4). Then

(i) it holds that $\Omega \ni \omega \mapsto \sup_{\theta \in [\alpha, \beta]^d} |\mathcal{R}(\theta, \omega) - \mathbf{R}(\theta)| \in [0, \infty]$ is $\mathcal{F}/\mathcal{B}([0, \infty])$-measurable and

(ii) it holds for all $p \in (0, \infty)$ that

$$
\begin{align}
\mathbb{E}\left[\left(\sup_{\theta \in [\alpha, \beta]^d} |\mathcal{R}(\theta) - \mathbf{R}(\theta)|^p\right)^{1/p}\right] &\nonumber \\
\le \inf_{c, \varepsilon \in (0, \infty)} \left[ \frac{2(c+1)b^2 \max\{1, [2\sqrt{M} L (\beta - \alpha)(cb)^{-1}\varepsilon] \varepsilon \} \max\{1, p, \frac{A}{\varepsilon}\}}{\sqrt{M}} \right]^{1/p} \tag{13.51} \\
&\le \inf_{c \in (0, \infty)} \left[ \frac{2(c+1)b^2 \sqrt{e} \max\{1, p, \mathrm{d}\ln(4M L^2 (\beta - \alpha)^2 (cb)^{-2})\}}{\sqrt{M}} \right].
\end{align}
$$

#### Proof of Proposition 13.3.2.

Throughout this proof, let $(\kappa_c)_{c \in (0, \infty)} \subseteq (0, \infty)$ satisfy for all $c \in (0, \infty)$ that

$$
\kappa_c = \frac{2\sqrt{M} L (\beta - \alpha)}{cb}, \quad (13.52)
$$

let $\mathcal{X}_{\theta,m}\colon \Omega \to \mathbb{R}$, $m \in \{1, 2, \dots, M\}$, $\theta \in [\alpha, \beta]^d$, satisfy for all $\theta \in [\alpha, \beta]^d$, $m \in \{1, 2, \dots, M\}$ that

$$
\mathcal{X}_{\theta,m} = f_\theta(X_m), \quad (13.53)
$$

and let $\delta\colon [\alpha, \beta]^{\mathrm{d}} \times [\alpha, \beta]^{\mathrm{d}} \to [0, \infty)$ satisfy for all $\theta, \vartheta \in [\alpha, \beta]^{\mathrm{d}}$ that

$$
\delta(\theta, \vartheta) = \lVert \theta - \vartheta \rVert_\infty. \quad (13.54)
$$

First, note that the assumption that for all $\theta \in [\alpha, \beta]^d$, $m \in \{1, 2, \dots, M\}$ it holds that $|f_\theta(X_m) - Y_m| \le b$ implies for all $\theta \in [\alpha, \beta]^d$, $m \in \{1, 2, \dots, M\}$ that

$$
|\mathcal{X}_{\theta,m} - Y_m| = |f_\theta(X_m) - Y_m| \le b. \quad (13.55)
$$

Furthermore, observe that the assumption that for all $\theta, \vartheta \in [\alpha, \beta]^d$, $x \in D$ it holds that $|f_\theta(x) - f_\vartheta(x)| \le L \lVert \theta - \vartheta \rVert_\infty$ ensures for all $\theta, \vartheta \in [\alpha, \beta]^d$, $m \in \{1, 2, \dots, M\}$ that

$$
\begin{align}
|\mathcal{X}_{\theta,m} - \mathcal{X}_{\vartheta,m}| = |f_\theta(X_m) - f_\vartheta(X_m)| & \le \sup_{x \in D} |f_\theta(x) - f_\vartheta(x)| \le L \lVert \theta - \vartheta \rVert_\infty = L \delta(\theta, \vartheta). \tag{13.56}
\end{align}
$$

The fact that for all $\theta \in [\alpha, \beta]^d$ it holds that $(\mathcal{X}_{\theta,m}, Y_m)$, $m \in \{1, 2, \dots, M\}$, are i.i.d. random variables, (13.55), and Lemma 13.3.1 (applied with $p \circlearrowright q$, $C \circlearrowright C$, $(E, \delta) \circlearrowright ([\alpha, \beta]^d, \delta)$, $(X_{x,m})_{x \in E, m \in \{1, \dots, M\}} \circlearrowright (\mathcal{X}_{\theta,m})_{\theta \in [\alpha, \beta]^d, m \in \{1, \dots, M\}}$ for $p \in [2, \infty)$, $C \in (0, \infty)$) in the notation of Lemma 13.3.1) therefore ensure that for all $p \in [2, \infty)$, $c \in (0, \infty)$ it holds that $\Omega \ni \omega \mapsto \sup_{\theta \in [\alpha, \beta]^d} |\mathcal{R}(\theta, \omega) - \mathbf{R}(\theta)| \in [0, \infty]$ is $\mathcal{F}/\mathcal{B}([0, \infty])$-measurable and

$$
\mathbb{E}\left[\left(\sup_{\theta \in [\alpha, \beta]^d} |\mathcal{R}(\theta) - \mathbf{R}(\theta)|^p\right)^{1/p}\right] \le \left(C^{([\alpha, \beta]^d, \delta), \frac{cb\sqrt{p-1}}{2L\sqrt{M}}} \right)^{1/p} \left[\frac{2(c+1)b^2\sqrt{p-1}}{\sqrt{M}}\right] \tag{13.57}
$$

(cf. Definition 4.3.2). This establishes item (i). Note that Proposition 12.2.24 (applied with $d \sim \bm{d}$, $a \sim \alpha$, $b \sim \beta$, $r \sim r$ for $r \in (0, \infty)$ in the notation of Proposition 12.2.24) shows that for all $r \in (0, \infty)$ it holds that

$$
\begin{align}
\mathcal{C}^{([\alpha, \beta], \delta), r} & \leq \mathbb{I}_{[0, r]}\big(\frac{\beta - \alpha}{2}\big) + \big(\frac{\beta - \alpha}{r}\big)^{\bm{d}} \mathbb{I}_{(r, \infty)}\big(\frac{\beta - \alpha}{2}\big) \nonumber \\
& \leq \max \big\{ \mathbb{I}, \big(\frac{\beta - \alpha}{r}\big)^{\bm{d}} \big\} \big( \mathbb{I}_{[0, r]}\big(\frac{\beta - \alpha}{2}\big) + \mathbb{I}_{(r, \infty)}\big(\frac{\beta - \alpha}{2}\big) \big\} \nonumber \\
& = \max \big\{ \mathbb{I}, \big(\frac{\beta - \alpha}{r}\big)^{\bm{d}} \big\}.
\end{align}
$$

Hence, we obtain for all $c \in (0, \infty)$, $p \in [2, \infty)$ that

$$
\begin{align}
\bigg( \mathcal{C}^{([\alpha, \beta], \bm{d}, c), \frac{\bm{c}_p \vee \bm{1}}{2L \vee M}} \bigg)^{1/p} & \leq \max \bigg\{ \mathbb{I}, \big(\frac{2(\beta - \alpha)L \vee \sqrt{M}}{c\beta \vee \mathbb{1}}\big)^{\bm{d}/p} \bigg\} \nonumber \\
& \leq \max \bigg\{ \mathbb{I}, \big(\frac{2(\beta - \alpha)L \vee \sqrt{M}}{c\beta} \big)^{\bm{d}/p} \bigg\} = \max \big\{ \mathbb{I}, (\kappa_c)^{\bm{d}/p} \big\}.
\end{align}
$$

This, (13.57), and Jensen’s inequality demonstrate that for all $c, \varepsilon, p \in (0, \infty)$ it holds that

$$
\begin{align}
& \big( \mathbb{E}\big[ \sup_{\bm{\theta} \in [\alpha, \beta]^{\bm{d}}} | \mathfrak{R}(\bm{\theta}) - \bm{R}(\bm{\theta}) |^{p} \big] \big)^{1/p} \nonumber \\
& \leq \big( \mathbb{E}\big[ \sup_{\bm{\theta} \in [\alpha, \beta]^{\bm{d}}} | \mathfrak{R}(\bm{\theta}) - \bm{R}(\bm{\theta}) |^{\max \{2p, \bm{d}/\varepsilon\}} \big] \big)^{\frac{1}{\max \{2p, \bm{d}/\varepsilon\}}} \nonumber \\
& \leq \max \bigg\{ \mathbb{I}, (\kappa_c)^{\frac{\bm{d}}{\max \{2p, \bm{d}/\varepsilon\}}} \bigg\} \frac{2(c+1)b^{2} \sqrt{\max \{2, p, \frac{\bm{d}}{\varepsilon}\} - 1}}{\sqrt{M}} \nonumber \\
& = \max \big\{ \mathbb{I}, (\kappa_c)^{\min \{4/2, \bm{d}/p, \bm{d}/\varepsilon\}} \big\} \frac{2(c+1)b^{2} \sqrt{\max \{\mathbb{I}, p - 1, \bm{d}/\varepsilon - 1\}}}{\sqrt{M}} \nonumber \\
& \leq \frac{2(c+1)b^{2} \max \{\mathbb{I}, (\kappa_c)^{\varepsilon}\} \sqrt{\max \{\mathbb{I}, p, \bm{d}/\varepsilon\}}}{\sqrt{M}}.
\end{align}
$$

Moreover, observe that the fact that for all $a \in (1, \infty)$ it holds that $a^{1/(2 \ln(a))} = e^{\ln(a)/(2 \ln(a))} = e^{1/2} = \sqrt{e} \geq 1$ proves that for all $c, p \in (0, \infty)$ with $\kappa_c > 1$ it holds that

$$
\begin{align}
& \inf_{\varepsilon \in (0, \infty)} \bigg[ \frac{2(c+1)b^{2} \max \{\mathbb{I}, (\kappa_c)^{\varepsilon}\} \sqrt{\max \{\mathbb{I}, p, \frac{\bm{d}}{\varepsilon}\}}}{\sqrt{M}} \bigg] \nonumber \\
& \leq \frac{2(c+1)b^{2} \max \{ \mathbb{I}, (\kappa_c)^{1/(2 \ln(\kappa_c))} \} \sqrt{\max \{ \mathbb{I}, p, 2 \bm{d} \ln(\kappa_c) \}}}{\sqrt{M}} \nonumber \\
& = \frac{2(c+1)b^{2} \sqrt{e} \max \{ \mathbb{I}, p, \bm{d} \ln([\kappa_c]^2) \}}{\sqrt{M}}.
\end{align}
$$
