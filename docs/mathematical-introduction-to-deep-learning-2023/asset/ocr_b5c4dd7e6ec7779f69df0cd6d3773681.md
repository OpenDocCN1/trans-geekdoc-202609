and

```
$\begin{align*} \mathbb{E}\Big[\left\langle X_{k_{1}}-\mathbb{E}[X_{k_{1}}], X_{k_{2}}-\mathbb{E}[X_{k_{2}}]\right\rangle\Big] &=\left\langle\mathbb{E}\left[X_{k_{1}}-\mathbb{E}[X_{k_{1}}]\right],\mathbb{E}\left[X_{k_{2}}-\mathbb{E}[X_{k_{2}}]\right]\right\rangle \\ &=\left\langle\left(\mathbb{E}[X_{k_{1}}]-\mathbb{E}[X_{k_{1}}]\right),\mathbb{E}\left[X_{k_{2}}\right]-\mathbb{E}[X_{k_{2}}]\right\rangle=0. \end{align*} \quad (7.30)$
```

Therefore, we obtain that

```
$\begin{align*} \mathbb{E}\left[\left\|\sum_{k=1}^{n}\left(X_{k}-\mathbb{E}\left[X_{k}\right]\right)\right\|^{2}\right] &=\mathbb{E}\left[\left\langle\sum_{k_{1}=1}^{n}\left(X_{k_{1}}-\mathbb{E}[X_{k_{1}}]\right),\sum_{k_{2}=1}^{n}\left(X_{k_{2}}-\mathbb{E}[X_{k_{2}}]\right)\right\rangle\right] \\ &=\mathbb{E}\left[\sum_{k_{1},k_{2}=1}^{n}\left\langle\left(X_{k_{1}}-\mathbb{E}[X_{k_{1}}]\right), X_{k_{2}}-\mathbb{E}[X_{k_{2}}]\right\rangle\right] \\ &=\mathbb{E}\left[\left(\sum_{k=1}^{n}\left\|X_{k}-\mathbb{E}\left[X_{k}\right]\right\|^{2}\right)+\left(\sum_{\begin{array}{c} k_{1},k_{2}\in\{1,2,\ldots,n\}, \\ k_{1}\neq k_{2}\end{array}}\left\langle X_{k_{1}}-\mathbb{E}\left[X_{k_{1}}\right], X_{k_{2}}-\mathbb{E}[X_{k_{2}}]\right\rangle\right)\right] \\ &=\left(\sum_{k=1}^{n}\mathbb{E}\left[\left\|X_{k}-\mathbb{E}\left[X_{k}\right]\right\|^{2}\right]\right)+\left(\sum_{\begin{array}{c} k_{1},k_{2}\in\{1,2,\ldots,n\}, \\ k_{1}\neq k_{2}\end{array}}\mathbb{E}\left[\left\langle X_{k_{1}}-\mathbb{E}\left[X_{k_{1}}\right], X_{k_{2}}-\mathbb{E}\left[X_{k_{2}}\right]\right\rangle\right]\right) \\ &=\sum_{k=1}^{n}\mathbb{E}\left[\left\|X_{k}-\mathbb{E}\left[X_{k}\right]\right\|^{2}\right]. \end{align*} \quad (7.31)$
```

The proof of Lemma 7.2.6 is thus complete. $\square$
