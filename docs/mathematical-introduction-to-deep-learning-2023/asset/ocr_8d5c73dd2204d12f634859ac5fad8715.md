$$
\begin{array}{rl}
\int_{\mathbb{R}^{d}}\left\langle D_{2}\left(s, x+\sqrt{2 p(t-s)} z\right), \frac{\rho z}{\sqrt{2 p(t-s)}}\right\rangle \gamma(z) \mathrm{d} z &

\end{array}
$$

Moreover, note that integration by parts proves that for all $t \in (0, T]$, $x \in \mathbb{R}^{d}$, $s \in [0, t)$, $i \in\{1, 2, \ldots, d\}$, $a \in \mathbb{R}$, $b \in (a, \infty)$ it holds that

$$
\begin{array}{rl}
\int_{a}^{b} D_{2, i}\left(s, x+\sqrt{2 p(t-s)}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right)\left(\frac{\partial \gamma}{\partial z_{i}}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right) \mathrm{d} z_{i} & 
\\
=\left[D_{2, i}\left(s, x+\sqrt{2 p(t-s)}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right) \gamma\left(z_{1}, z_{2}, \ldots, z_{d}\right)\right]_{z_{i}=a}^{z_{i}=b} & \text {(17.48)} 
\\
\quad-\int_{a}^{b} \sqrt{2 p(t-s)} H_{i, i}\left(s, x+\sqrt{2 p(t-s)}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right) \gamma\left(z_{1}, z_{2}, \ldots, z_{d}\right) \mathrm{d} z_{i}.
\end{array}
$$

The assumption that $u$ has at most polynomially growing derivatives hence establishes that for all $t \in (0, T]$, $x \in \mathbb{R}^{d}$, $s \in [0, t)$, $i \in\{1, 2, \ldots, d\}$ it holds that

$$
\begin{array}{rl}
\int_{\mathbb{R}} D_{2, i}\left(s, x+\sqrt{2 p(t-s)}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right)\left(\frac{\partial \gamma}{\partial z_{i}}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right) \mathrm{d} z_{i} & 
\\
=-\sqrt{2 p(t-s)} \int_{\mathbb{R}} H_{i, i}\left(s, x+\sqrt{2 p(t-s)}\right)\left(z_{1}, z_{2}, \ldots, z_{d}\right) \gamma\left(z_{1}, z_{2}, \ldots, z_{d}\right) \mathrm{d} z_{i}. & \text {(17.49)}
\end{array}
$$

Combining this with (17.47) and Fubini's theorem ensures that for all $t \in (0, T]$, $x \in \mathbb{R}^{d}$, $s \in [0, t)$ it holds that

$$
\begin{array}{l}
\int_{\mathbb{R}^{d}}\left\langle D_{2}\left(s, x+\sqrt{2 p(t-s)} z\right), \frac{\rho z}{\sqrt{2 p(t-s)}}\right\rangle \gamma(z) \mathrm{d} z & 
\\
=-\rho \sum_{i=1}^{d} \int_{\mathbb{R}^{d}} H_{i, i}\left(s, x+\sqrt{2 p(t-s)}\right)(z) \gamma(z) \mathrm{d} z & \text {(17.50)} 
\\
=-\int_{\mathbb{R}^{d}} \rho \operatorname{Trace}\left(H\left(s, x+\sqrt{2 p(t-s)}(z)\right)\right) \gamma(z) \mathrm{d} z.
\end{array}
