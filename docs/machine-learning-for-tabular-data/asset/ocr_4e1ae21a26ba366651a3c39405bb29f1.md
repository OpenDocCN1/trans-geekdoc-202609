```
def extract_xgb_node_info(tree):
    return: [tree['split'], tree['origin'], tree['gain'],
            tree['cover'], tree['split_condition']] # Function extracting various information from a split node in the tree structure
```

```
def extract_xgb_leaf_info(tree):
    return {
        tree['nodeid'],
        tree['origin'],
        tree['cover']
    } # Function extracting information from a leaf in the tree structure
```

```
def traverse_xgb_tree(tree):
    if not 'origin' in tree:
        tree['origin'] = ""
    if not 'children' in tree:
        return [extract_xgb_leaf_info(tree)]
    left_branch = tree['children'][0]
    right_branch = tree['children'][1]
    left_branch['origin'] = '<'
    right_branch['origin'] = '>'
    left_paths = traverse_xgb_tree(left_branch)
    node_info = [Split(extract_xgb_node_info(tree))]
    return [node_info + path for path in left_paths + right_paths] # Function traversing the tree structure recursively to extract paths
```
