def label_smiles(line, MAX_SMI_LEN, smi_idx):
    X = np.zeros(MAX_SMI_LEN)
    for i, ch in enumerate(line[:MAX_SMI_LEN]):
        if ch!='\n':
            X[i] = smi_idx[ch]
    return X
