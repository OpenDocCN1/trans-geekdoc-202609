```
controls.infoBut.onRelease = function() {
  if (_root.infovis == false) {
    info_.visible = true;
    infovis = true;
  } else {
    info_.visible = false;
    _root.infovis = false;
  }
}
info_.closeBut.onRelease = function() {
  info_.visible = false;
  _root.infovis = false;
};
```
