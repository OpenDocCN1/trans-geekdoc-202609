```
# Import PyTorch

import torch

# Create a tensor

tensor = torch.tensor([1, 2, 3])

print("Original tensor:")

print(tensor)

# Check the current device of the tensor

print("\nTensor is on: ", tensor.device)

# Move the tensor to GPU if available

if torch.cuda.is_available():

    tensor = tensor.to('cuda')

print("\nMoved tensor:")
```
