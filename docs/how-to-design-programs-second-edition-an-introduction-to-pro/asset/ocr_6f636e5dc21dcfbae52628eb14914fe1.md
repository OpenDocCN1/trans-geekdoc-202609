; version 3
(define (f*ldl f e l)

          (cond
            [(empty? l) a]
            [else (invert/a (rest l)
                            (cons (first l) a))]))
         (define (foldr l)
           (cond
             [(empty? l) e]
             [else
               (f (first l) (foldr (rest l)))]))
