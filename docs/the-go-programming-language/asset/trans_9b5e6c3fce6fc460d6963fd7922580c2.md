
结构体被编码为字段绑定的列表，每个字段绑定都是一个包含两个元素的列表，其第一个元素（一个符号）是字段名，第二个元素是字段值。映射也被编码为键值对的列表，每个键值对对应映射中的一个条目。传统上，S 表达式使用单个 *cons* 单元 `(key . value)` 来表示每个键值对，而不是两个元素的列表，但为了简化解码，我们将忽略点对列表表示法。

编码由单个递归函数 `encode` 完成，如下所示。其结构本质上与上一节中的 `Display` 相同：

[点击此处查看代码图像](#ch12_images.html#id339pro01_r)

`gopl.io/ch12/sexpr`

```
func encode(buf *bytes.Buffer, v reflect.Value) error {
    switch v.Kind() {
    case reflect.Invalid:
        buf.WriteString("nil")

    case reflect.Int, reflect.Int8, reflect.Int16,
        reflect.Int32, reflect.Int64:
        fmt.Fprintf(buf, "%d", v.Int())

    case reflect.Uint, reflect.Uint8, reflect.Uint16,
        reflect.Uint32, reflect.Uint64, reflect.Uintptr:
        fmt.Fprintf(buf, "%d", v.Uint())

    case reflect.String:
        fmt.Fprintf(buf, "%q", v.String())

    case reflect.Ptr:
        return encode(buf, v.Elem())

    case reflect.Array, reflect.Slice: // (value ...)
        buf.WriteByte('(')
        for i := 0; i < v.Len(); i++ {
            if i > 0 {
                buf.WriteByte(' ')
            }
            if err := encode(buf, v.Index(i)); err != nil {
                return err
            }
        }
        buf.WriteByte(')')

    case reflect.Struct: // ((name value) ...)
        buf.WriteByte('(')
        for i := 0; i < v.NumField(); i++ {
            if i > 0 {
                buf.WriteByte(' ')
            }
            fmt.Fprintf(buf, "(%s ", v.Type().Field(i).Name)
            if err := encode(buf, v.Field(i)); err != nil {
                return err
            }
            buf.WriteByte(')')
        }
        buf.WriteByte(')')

    case reflect.Map: // ((key value) ...)
        buf.WriteByte('(')
        for i, key := range v.MapKeys() {
            if i > 0 {
                buf.WriteByte(' ')
            }
            buf.WriteByte('(')
            if err := encode(buf, key); err != nil {
                return err
            }
            buf.WriteByte(' ')
            if err := encode(buf, v.MapIndex(key)); err != nil {
                return err
            }
            buf.WriteByte(')')
        }
        buf.WriteByte(')')

    default: // float, complex, bool, chan, func, interface
        return fmt.Errorf("unsupported type: %s", v.Type())
    }
    return nil
}
```

func Marshal(v interface{}) ([]byte, error) {
    var buf bytes.Buffer
    if err := encode(&buf, reflect.ValueOf(v)); err != nil {
        return nil, err
    }
    return buf.Bytes(), nil
}

## 12.5 使用 `reflect.Value` 设置变量

到目前为止，反射只是以各种方式*解读*我们程序中的值。然而，本节的重点是*修改*它们。

回想一下，某些 Go 表达式（如 `x`、`x.f[1]` 和 `*p`）表示变量，但其他表达式（如 `x + 1` 和 `f(2)`）则不是。变量是一个包含值的*可寻址*存储位置，其值可以通过该地址进行更新。

类似的区别也适用于 `reflect.Value`。有些是可寻址的；有些则不是。请看以下声明：

[点击此处查看代码图像](#ch12_images.html#id342pro01_r)

```
x := 2                     // value   type   variable?
a := reflect.ValueOf(2)    // 2       int    no
b := reflect.ValueOf(x)    // 2       int    no
c := reflect.ValueOf(&x)   // &x      *int   no
d := c.Elem()              // 2       int    yes (x)
```

`a` 中的值不可寻址。它只是整数 2 的一个副本。`b` 也是如此。`c` 中的值也是不可寻址的，它是指针值 `&x` 的一个副本。事实上，`reflect.ValueOf(x)` 返回的任何 `reflect.Value` 都不可寻址。但 `d` 通过解引用其中的指针从 `c` 导出，引用了一个变量，因此是可寻址的。我们可以使用这种方法，调用 `reflect.ValueOf(&x).Elem()`，为任意变量 `x` 获取一个可寻址的 `Value`。

我们可以通过其 `CanAddr` 方法询问 `reflect.Value` 是否可寻址：

[点击此处查看代码图像](#ch12_images.html#id342pro02_r)

```
fmt.Println(a.CanAddr()) // "false"
fmt.Println(b.CanAddr()) // "false"
fmt.Println(c.CanAddr()) // "false"
fmt.Println(d.CanAddr()) // "true"
```
