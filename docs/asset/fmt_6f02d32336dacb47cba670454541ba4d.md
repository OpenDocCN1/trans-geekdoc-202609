void idea_enc(idea_ctx *c, unsigned char *data, int blocks) {
    int i;
    unsigned char *d = data;

        ideaCipher(d, d, c->ek);

void idea_dec(idea_ctx *c, unsigned char *data, int blocks) {
    int i;
    unsigned char *d = data;
    for(i=0; i<blocks; i++) {
        ideaCipher(d, d, c->dk);
        d+=8;
    }
}

static word32 gost(gost_ctx *ctx, word32 x) {
    x ^= ctx->k8[(x >> 24) & 0xFF];
    x ^= ctx->k7[(x >> 16) & 0xFF];
    x ^= ctx->k6[(x >> 8) & 0xFF];
    x ^= ctx->k5[x & 0xFF];
    x ^= ctx->k4[(x >> 24) & 0xFF];
    x ^= ctx->k3[(x >> 16) & 0xFF];
    x ^= ctx->k2[(x >> 8) & 0xFF];
    x ^= ctx->k1[x & 0xFF];
    return x << 11; /* 左旋转 11 位 */
}

void gostcrypt(gost_ctx *ctx, word32 *d) {
