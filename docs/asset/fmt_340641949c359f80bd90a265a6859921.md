          set assignments w/= i <- j;
        }

      mutable sum = new Double[Length(data[0])];
      mutable count = 0;
      for (i in 0..Length(data)-1) {
        if (assignments[i] == j) {
          sum = sum + data[i];
          count += 1;
        }
      }
      set centroids w/= j <- sum / count;
    }
  } until (ConvergenceCheck(centroids, assignments));
  return centroids;
}

### 量子近似优化算法 (QAOA)

优化最有前景的量子算法之一是量子近似优化算法（`QAOA`）。`QAOA`旨在通过近似最优解来解决组合优化问题。该算法通过交替在量子态上施于特定问题的哈密顿量和混合哈密顿量来运行。

#### QAOA：理论基础

`QAOA`的核心思想是将优化问题编码到哈密顿量`H_C`中，并使用混合哈密顿量`H_M`来探索解空间。量子态的演化如下：

`|ψ⟩ = e^{-i γ_p H_C} e^{-i β_p H_M} ... e^{-i γ_1 H_C} e^{-i β_1 H_M |s⟩`

其中`γ_i`和`β_i`是控制演化的参数，`|s⟩`是初始态。

#### 在 Q 中实现 QAOA

让我们探索 `QAOA` 在 Q 中的实现。我们将开始定义`problemHamiltonian`和`mixingHamiltonian`，然后是 `QAOA` 的量子电路。

#### Q 中的 QAOA 实现

```q
operation QAOA(parameters : Double[], problemHamiltonian : (Qubit[] => Unit), mixingHamiltonian : (Qubit[] => Unit), qubits : Qubit[]) : Unit {
  body {
    let p = Length(parameters) / 2;
    for (i in 0 .. p-1) {
      let gamma = parameters[2*i];
      let beta = parameters[2*i+1];
      // Apply 参数 gamma 应用问题哈密顿量
      problemHamiltonian(qubits);
      Rz(gamma, qubits);
      // Apply 参数 beta 应用混合哈密顿量
      mixingHamiltonian(qubits);
      Rx(beta, qubits);
    }
  }
}
```
