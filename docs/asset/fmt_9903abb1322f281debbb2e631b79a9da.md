下面的函数会加载相应 CNN 架构提取的特征，并使用 SMOTE 算法为相对于其他占主导地位类别较少的类别生成合成数据点。

例如，“其他”类有 400 张图像，它是数据集中某一类别图像数量最多的。因此，将为其他所有类别创建合成图像，使每个类别的图像数量达到 400。例如，“PayPal”类有 121 张图像，因此将为“PayPal”类生成 279 张合成图像。其余类别也将按此方式处理。

```python
def getSMOTE(model):
    X_SMOTE = np.load(data_dir + model + "_train_images.npy")
    y_train = np.load(data_dir + '/train_labels.npy')

    # SMOTE
    from imblearn.over_sampling import SMOTE
    oversample = SMOTE()
    X_train_SMOTE, y_train_SMOTE = oversample.fit_resample(X_SMOTE, y_train)

    np.save(out_dir + model + "_X_train_SMOTE", X_train_SMOTE)
    np.save(out_dir + model + "_y_train_SMOTE", y_train_SMOTE)
```

下面的循环会遍历在之前创建的列表中的所有 CNN 架构：

```python
for arch in list_of_architectures:
    getSMOTE(arch)
```

循环结束后，所有使用不同 CNN 架构提取的特征的数据点数量都将被增强。

def getClassificationMetrics(test_labels, prediction):
    from sklearn.metrics import confusion_matrix
    test_label_list = list(np.unique(test_labels))
    cnf_matrix = confusion_matrix(test_labels, prediction, labels=test_label_list)
    np.set_printoptions(precision=2)

    # Plot non-normalized confusion matrix
    plt.figure(figsize=(8, 8))
    plot_confusion_matrix(cnf_matrix, classes=test_label_list,
                          title='Confusion matrix, without normalization')

    from sklearn import metrics

    accuracy = round(metrics.accuracy_score(test_labels, prediction), 3)
    precision = round(metrics.precision_score(test_labels, prediction, average='weighted'), 3)
    recall = round(metrics.recall_score(test_labels, prediction, average='weighted'), 3)
    f1_score = round(metrics.f1_score(test_labels, prediction, average='weighted'), 3)

    FP = cnf_matrix.sum(axis=0) - np.diag(cnf_matrix)
    FN = cnf_matrix.sum(axis=1) - np.diag(cnf_matrix)
    TP = np.diag(cnf_matrix)
    TN = cnf_matrix.sum() - (FP + FN + TP)

    # Sensitivity, hit rate, recall, or true positive rate
    TPR = round((TP / (TP + FN)).mean(), 3)
    # Specificity or true negative rate
    TNR = round((TN / (TN + FP)).mean(), 3)
    # Precision or positive predictive value
    PPV = round((TP / (TP + FP)).mean(), 3)
    # Negative predictive value
    NPV = round((TN / (TN + FN)).mean(), 3)
    # Fall out or false positive rate
    FPR = round((FP / (FP + TN)).mean(), 3)
    # False negative rate
    FNR = round((FN / (TP + FN)).mean(), 3)
    # False discovery rate
    FDR = round((FP / (TP + FP)).mean(), 3)

    # Balanced metrics
    balanced_accuracy = round(metrics.balanced_accuracy_score(test_labels, prediction), 3)
    balanced_precision = PPV
    balanced_recall = TPR
    balanced_f1 = round((2 * (PPV * TPR) / (PPV + TPR)), 3)

    print("Acc = ", accuracy)
    print("Pre = ", precision)
    print("Rec = ", recall)
    print("F1 = ", f1_score)
    print("")

    print("Bal_Acc = ", balanced_accuracy)
    print("Bal_Pre = ", balanced_precision)
    print("Bal_Rec = ", balanced_recall)
    print("Bal_F1 = ", balanced_f1)
    print("")

    from sklearn.metrics import classification_report, confusion_matrix
    print(classification_report(test_labels, prediction))

    return accuracy, precision, recall, f1_score, balanced_accuracy, balanced_precision, balanced_recall, balanced_f1

def classify(model_name, classifier, image_size):
    X_train, y_train, X_test, y_test = getData(model_name, image_size)

    if classifier == "RF":
        from sklearn.ensemble import RandomForestClassifier
        model = RandomForestClassifier(n_estimators=100, random_state=42)
    elif classifier == "SVM":
        from sklearn.svm import SVC
        model = SVC(kernel='linear', degree=3, C=1, decision_function_shape='ovo')

    # Train the model on training data
    model.fit(X_train, y_train)  # For sklearn no one hot encoding

    # Now predict using the trained RF model.
    prediction = model.predict(X_test)

    # Get the encoder from the pickle file
    from sklearn import preprocessing
    le = preprocessing.LabelEncoder()
    le.classes_ = getLabelEncoder(layer, image_size)

    # Inverse le transform to get original label back.
    prediction = le.inverse_transform(prediction)
    y_test = le.inverse_transform(y_test)

    accuracy, precision, recall, f1_score, balanced_accuracy, balanced_precision, balanced_recall, balanced_f1 = getClassificationMetrics(y_test, prediction)
    return accuracy, precision, recall, f1_score, balanced_accuracy, balanced_precision, balanced_recall, balanced_f1
