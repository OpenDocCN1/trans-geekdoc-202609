```python
class CustomCallback(Callback):
    """用于打印 epoch 信息的自定义 Keras 回调。"""
    def on_epoch_end(self, epoch, logs=None):
        """每 10 个 epoch 打印一次 epoch 信息。"""
        if (epoch + 1) % 10 == 0:
            logger.info(
                f"Epoch [{epoch + 1}/{self.params['epochs']}], "
                f"Loss: {logs['loss']:.4f}")
```

```python
class DiabetesProcessor:
    """用于预处理糖尿病数据的类。"""
    @staticmethod
    def preprocess_data(
            spark,
            data_file_path,
            train_parquet_path,
            test_parquet_path
    ):
        """使用 PySpark 预处理糖尿病数据。"""
        try:
            diabetes_df = spark.read.csv(
                data_file_path,
                header=True,
                inferSchema=True
            )
            diabetes_df = diabetes_df.filter(
                (col("Glucose") != 0)
                & (col("BloodPressure") != 0)
                & (col("BMI") != 0)
            )
            feature_cols = [
                "Pregnancies",
                "Glucose",
                "BloodPressure",
                "BMI",
                "DiabetesPedigreeFunction",
                "Age"
            ]
            assembler = VectorAssembler(
                inputCols=feature_cols,
                outputCol="features")
            diabetes_df = assembler.transform(diabetes_df)
            scaler = StandardScaler(
                inputCol="features",
                outputCol="scaled_features"
            )
            scaler_model = scaler.fit(diabetes_df)
            diabetes_df = scaler_model.transform(diabetes_df)
            train_df, test_df = diabetes_df.randomSplit(
                [0.8, 0.2],
                seed=42
            )
            train_df.write.parquet(
                train_parquet_path,
                mode="overwrite"
            )
            test_df.write.parquet(
                test_parquet_path,
                mode="overwrite"
            )
            logger.info("数据预处理已完成。")
        except Exception as e:
            logger.error(
                f"Error occurred during data preprocessing: "
                f"{str(e)}"
            )
```

                    max_value=32,
                    step=8
                )
            )
        )
        model.add(ReLU())
        model.add(Dense(1, activation='sigmoid'))
        optimizer = SGD(
            learning_rate=hp.Float(
                'lr',

            )
        )
        model.compile(loss='binary_crossentropy', optimizer=optimizer, metrics=['accuracy'])
        return model

```python
class DiabetesModelTrainer:
    """用于训练 TensorFlow (Keras) 模型的类。"""
    @staticmethod
    def train_tensorflow_model(
        X_train,
        y_train,
        X_test,
        y_test,
        epochs=50
    ):
        """使用 keras_tuner 训练 TensorFlow (Keras) 模型。"""
        try:
            tuner = RandomSearch(
                DiabetesHyperModel(),
                objective='val_accuracy',
                max_trials=5,
                executions_per_trial=3,
                directory='tuner_dir',
                project_name='diabetes_tuning'
            )
            tuner.search(
                X_train,
                y_train,
                epochs=epochs,
                validation_data=(X_test, y_test),
                verbose=1
            )
            best_model = tuner.get_best_models(num_models=1)[0]
            logger.info("最佳模型训练完成。")
            return best_model
        except Exception as e:
            logger.error(
                f"模型训练过程中发生错误: "
                f"{str(e)}"
            )
```

```python
class DiabetesModelEvaluator:
    """用于评估 TensorFlow (Keras) 模型的类。"""
    @staticmethod
    def evaluate_model(model, X_test, y_test):
        """评估 TensorFlow (Keras) 模型。"""
        try:
            y_pred = (model.predict(X_test) > 0.5).astype(int)
            accuracy = accuracy_score(y_test, y_pred)
            precision = precision_score(y_test, y_pred)
            recall = recall_score(y_test, y_pred)
            f1 = f1_score(y_test, y_pred)
            logger.info(f"准确率: {accuracy:.4f}")
            logger.info(f"精确率: {precision:.4f}")
            logger.info(f"召回率: {recall:.4f}")
            logger.info(f"F1 分数: {f1:.4f}")
        except Exception as e:
            logger.error(
                f"模型评估过程中发生错误: "
                f"{str(e)}"
            )
```
