Algorithm 4.3 SlidingWindowParser(T, Q, W_K)
Input: Tree T = {T_l}_{l=0}^{L-1} where T_l \in R^{H_l \times W_l \times d_{model}}, query vectors Q \in R^{N_D \times d_{model}}, weight matrix W_K \in R^{d_{model} \times d_{model}}
Output: List of selected node embeddings S
S \leftarrow \emptyset
\bar{q} \leftarrow \frac{1}{N} \sum_{i=1}^{N_D} Q_i^T \quad \triangleright \bar{q} \in R^{d_{model}}
s_{prev} \leftarrow -\infty
for l = L - 1 to 0 do
    if l = L - 1 then
N \leftarrow T_l
    else
N \leftarrow Children(T_l, (i_{max}, j_{max})) \quad \triangleright N \in R^{K_h \times K_w \times d_{model}}
    end if
K \leftarrow N W_K \quad \triangleright Project node embeddings
s \leftarrow SimilarityScore(\bar{q}, K) \quad \triangleright s \in R^{K_h \times K_w}
s_{max} \leftarrow max s
if s_{max} > s_{prev} then
    if l = L - 1 then
S \leftarrow T_l
    else
\tilde{N} \leftarrow AreaOfInterest(T_l, (i_{max}, j_{max})) \quad \triangleright \tilde{N} \in R^{3 K_h \times 3 K_w \times d_{model}}
\tilde{K} \leftarrow \tilde{N} W_K
\tilde{s} \leftarrow SimilarityScore(\bar{q}, \tilde{K})
S \leftarrow SelectNodes(\tilde{N}, \tilde{s})
    end if
    s_{prev} \leftarrow s_{max}
    (i_{max}, j_{max}) \leftarrow arg max(s)
else
    break
end if
end for
return S
