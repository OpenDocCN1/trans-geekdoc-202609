def get_mask(
    dim_data:list[int],
    dim_window:list[int],
    ) -> torch.Tensor:

    d = dim_data
    w = dim_window
    n = len(d)
    k = 1
    mask = torch.zeros(*d)
    for i in range(n):
        mask = torch.transpose(mask, 0, i)
        mask[d[i] - w[i] : d[i]] += k
        mask[d[i]//w[i]//2 : d[i]] += k
        mask = torch.transpose(mask, 0, i)
        k *= 3

    part = [
        j
        for i in range(n)
        for j in (d[i]//w[i], w[i])
    ]
    perm = (
        list(range(0, 2 * n, 2))
        + list(range(1, 2 * n, 2))
    )
    wins = torch.prod(torch.tensor(w)).item()

    mask = torch.reshape(mask, part)
    mask = torch.permute(mask, perm)
    mask = torch.reshape(mask, (-1, wins))
    mask = mask.unsqueeze(1) - mask.unsqueeze(2)
    mask[mask != 0] = -100.

    return mask
