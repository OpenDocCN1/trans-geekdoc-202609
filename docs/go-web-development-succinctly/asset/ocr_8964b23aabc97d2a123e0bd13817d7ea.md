# Tidying the output

All we have now is some rather ugly text dumped out to the browser. Let's make it a little bit more pleasant to look at by displaying the query results in an HTML table. We do this simply by rewriting the code in our handler, as in Code Listing 9.

Code Listing 9: Using HTML to Format the Query Results

```
func showCity(w http.ResponseWriter, r *http.Request) {
	city := City()
	queryParam := r.URL.Path[1:] + "%"
	rows, err := database.Query("SELECT Name, CountryCode, Population FROM city WHERE Name LIKE ?", queryParam)
	if err != nil {
		log.Fatal(err)
	}
	defer rows.Close()

	hml := "<html><head><title>CitySearch</title></head><h1>Search for " + queryParam + "</h1><table border='1'><tr><th>City</th><th>Country</th><th>Population</th></tr>"

	for rows.Next() {
		err := rows.Scan(&city.Name, &city.CountryCode, &city.Population)
	
		if err != nil {
			log.Fatal(err)
		}
			hml +=
		fmt.Sprintf("<tr><td>%s</td><td>%s</td><td>%d</td></tr>", city.City, city.CountryCode, city.Population)
	}
	err := rows.Err()
	if err != nil {
		log.Fatal(err)
	} else {
		hml += "<table></body></body></html>"
		fmt.Fprintln(w, hml)
	}
```
