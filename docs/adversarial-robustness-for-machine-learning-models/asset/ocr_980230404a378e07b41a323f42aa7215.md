\(\begin{aligned} & = \| \boldsymbol{v} \|_2^2 + \mathbb{E} \left[ \left\| \frac{1}{q} \sum_{i=1}^q \left( \mathbf{g}^{(i)} - \boldsymbol{v} \right) \right\|_2^2 \right] \\ & = \| \boldsymbol{v} \|_2^2 + \frac{1}{q} \mathbb{E} \left[ \| \boldsymbol{g}^{(1)} - \boldsymbol{v} \|_2^2 \right] \tag{3.20} \\ & = \| \boldsymbol{v} \|_2^2 + \frac{1}{q} \mathbb{E} \left[ \| \boldsymbol{g}^{(1)} \|_2^2 \right] - \frac{1}{q} \| \boldsymbol{v} \|_2^2, \tag{3.21} \end{aligned}\)

where we have used the fact that \(\mathbb{E}[\mathbf{g}^{(i)}] = \mathbb{E}[\mathbf{g}^{(1)}] = \boldsymbol{v}\) for all \(i\). The definition of \(\boldsymbol{v}\) in (3.18) yields

\(\begin{aligned} \| \boldsymbol{v} \|_2^2 \le 2 \frac{b^2}{d^2} \| \nabla f(\mathbf{x}) \|_2^2 + 2 \| \boldsymbol{y} \|_2^2 \\ \le 2 \frac{b^2}{d^2} \| \nabla f(\mathbf{x}) \|_2^2 + \frac{1}{2} b^2 \beta^2 L^2 . \end{aligned} \tag{3.22}\)

From (3.17) we also obtain that for all \(i\),

\(\mathbb{E} \left[ \| \mathbf{g}^{(i)} \|_2^2 \right] \le \frac{b^2 L^2 \beta^2}{2} + \frac{2 b^2}{d} \| \nabla f(\mathbf{x}) \|_2^2 . \tag{3.23}\)

Substituting (3.22) and (3.23) into (3.21), we obtain

\(\begin{aligned} \mathbb{E} \left[ \| \overline{\mathbf{g}} \|_2^2 \right] \le \| \boldsymbol{v} \|_2^2 + \frac{1}{q} \mathbb{E} \left[ \| \boldsymbol{g}^{(1)} \|_2^2 \right] \tag{3.24} \\ \le 2 \left( \frac{b^2}{d^2} + \frac{b^2}{dq} \right) \| \nabla f(\mathbf{x}) \|_2^2 + \frac{q+1}{2q} b^2 L^2 \beta^2 . \tag{3.25} \end{aligned}\)

Finally, we bound the mean squared estimation error as

\(\begin{aligned} \mathbb{E} \left[ \| \overline{\mathbf{g}} - \nabla f(\mathbf{x}) \|_2^2 \right] \le 2 \mathbb{E} \left[ \| \overline{\mathbf{g}} - \boldsymbol{v} \|_2^2 \right] + 2 \| \boldsymbol{v} - \nabla f(\mathbf{x}) \|_2^2 \\ \le 2 \mathbb{E} \left[ \| \overline{\mathbf{g}} \|_2^2 \right] + 2 \| \frac{b}{d} \nabla f(\mathbf{x}) + \boldsymbol{y} - \nabla f(\mathbf{x}) \|_2^2 \\ \le 4 \left( \frac{b^2}{d^2} + \frac{b^2}{dq} + \frac{(b-d)^2}{d^2} \right) \| \nabla f(\mathbf{x}) \|_2^2 + \frac{2q+1}{q} b^2 L^2 \beta^2 , \end{aligned} \tag{3.26}\)

which completes the proof.
