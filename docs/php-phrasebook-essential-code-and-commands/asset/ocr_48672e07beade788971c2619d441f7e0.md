<?php
// ...

$data = array(
    array('April 2005', '19,720,597', '1,310,181'),
    array('January 2005', '18,455,683',
        '1,317,871'),
    array('December 2004', '17,826,404',
        '1,318,739')
);

if ($fp = @fopen('usage.csv', 'w')) {
    foreach ($data as $line) {
        fputcsv($fp, $line);
    }
    fclose($fp);
    echo 'CSV written.';
} else {
    echo 'Cannot open file.';
}
?>
