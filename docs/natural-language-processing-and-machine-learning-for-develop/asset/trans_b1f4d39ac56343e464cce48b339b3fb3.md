```python
from sklearn.datasets import load_digits
from matplotlib import pyplot as plt
digits = load_digits()  # set interpolation='none'
fig = plt.figure(figsize=(3, 3))
plt.imshow(digits['images'][66], cmap="gray", interpolation='none')
plt.show()
```
