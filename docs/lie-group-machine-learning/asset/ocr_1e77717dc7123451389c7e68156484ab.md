\end{array}$$

where $K = (x_{n}^{\text{max}} - x_{n}^{\text{min}})$, and $x_{n}^{\text{max}}$ and $x_{n}^{\text{min}}$ are the upper and lower bounds of $x_{n}$, respectively. Further, $x_{n}$ satisfies the condition that for any $x_{n}\notin [x_{n}^{\text{min}}, x_{n}^{\text{max}}]$, $f(x_{1}, x_{2}, \ldots, x_{n})g(u_{1} - x_{1}, u_{2} - x_{2}, \ldots, u_{n} - x_{n}) = 0$. $$\begin{array}{l}
f(x_{1}, x_{2}, \ldots, x_{n}) * g(x_{1}, x_{2}, \ldots, x_{n})(u_{1}, u_{2}, \ldots, u_{n}) \\
\quad=K\cdot [F(x_{1}, x_{2}, \ldots, x_{n-1})*G(x_{1}, x_{2}, \ldots, x_{n-1})]
(u_{1}+\lambda u_{n}\cos\alpha, \\
\quad u_{2}+\lambda u_{n}\sin\alpha, u_{3}, \ldots, u_{n})
\end{array}$$

where $K$ is a variable determined by $f$ and $g$,

$$\begin{array}{l}
F(x_{1}, x_{2}, \ldots, x_{n-1}) = f(x_{1}-\lambda X_{n}(x_{1}, x_{2})\cos\alpha, \\
\quad x_{2}-\lambda X_{n}(x_{1}, x_{2})\sin\alpha, x_{3}, \ldots, x_{n-1}, X_{n}(x_{1}, x_{2})) \\

$$X_{n}(x_{1}, x_{2}) = \frac{\sqrt{\left(x_{1}-\frac{j}{i}\right)^{2}+\left(x_{2}-\frac{m}{n}\right)^{2}}}{\lambda}$$
