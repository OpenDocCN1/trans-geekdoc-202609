### EX 8.3. The following code gives a function to compute the marginal likelihood. Note that we optimise the log of the parameters to avoid having to use a constrained optimisation:

```matlab
function ML = marg_like(hyp,y,x)
% Function to compute the marginal likelihood
% hyp contains log of the hyperparameters
N = length(x);
% Compute covariance function
CSS = exp(-exp(hyp(1))*(repmat(x,1,N) - repmat(x',N,1)).^2);
CSS = CSS + exp(hyp(2))*eye(N);
% Compute marginal likelihood
ML = -(N/2)*log(2*pi) - 0.5*log(det(CSS)) - 0.5*y'*inv(CSS)*y;
% Return negative as optimiser minimises
ML = -ML;
```

### EX 8.4. The function value at the nth data point is f_n. This function is then exponentiated to give the rate of a Poisson λ_n = exp(f_n). The integer count sampled from the Poisson is z_n. The gradient of the log posterior is given by:

$$
\frac{\delta}{\delta\mathbf{f}} = \mathbf{z} - \exp(\mathbf{f}) - \mathbf{C}^{-1}\mathbf{f}
$$

$$
\frac{\delta^2}{\delta\mathbf{f}\delta\mathbf{f}^T} = -\mathbf{C}^{-1} - \mathbf{E}
$$

where E is a matrix with zeros everywhere except the diagonal where the nth diagonal value is exp(f_n). This is demonstrated with the following code:

```matlab
% Generate some data
x = [0:0.01:1];
gam = 10.0;
N = length(x);
C = zeros(N);
% Create the kernel matrix
for n = 1:N
    for m = 1:N
        C(n,m) = exp(-gam*(x(n)-x(m))^2);
    end
end
C = C + 1e-6*eye(N);
% Generate a sample
f_samps = gausssamp(repmat(0,N,1),C,1)
% Plot the function
figure(1);
hold off
plot(x,f_samps);
% Generate the counts
rate = exp(f_samps);
n = poissrnd(rate)';
hold on
plot(x,n,'ro')
% Use Newton-Raphson to optimise f
f = zeros(N,1);
iK = inv(C);
oldf = f;
for it = 1:100
    gr = n - exp(f) - iK*f;
    he = -iK - diag(exp(f));
    f = f - inv(he)*gr;
    ch = sum((f - oldf).^2);
    oldf = f;
    if ch < 1e-6
        break
    end
end
plot(x,f,'g')
```

### EX 8.5. The following code generates some data, uses M-H to sample from the posterior and uses the posterior samples to compute the predictions on a grid, in order to show the predictive contours.

```matlab
% GP classification using M-H
x = [randn(20,2)-repmat(4,20,2);randn(20,2)+repmat(4,20,2)];
t = [repmat(0,20,1);repmat(1,20,1)];
N = size(x,1);
% Compute the covariance matrix
gam = 1;alp = 5;
C = zeros(N);
for n = 1:N
    for m = 1:N
        C(n,m) = alp*exp(-gam*(sum((x(n,:)-x(m,:)).^2)));
    end
end
iC = inv(C);
nSamps = 10000;
pos0 = find(t==0);
pos1 = find(t==1);
% Initialise with f being -3 for the -ve class and +3 for the positive
% Could use anything, but this gets us to a good place a bit faster!
f = [repmat(-3,20,1);repmat(3,20,1)];
oldProb = -0.5*f'*iC*f - sum(t.*log(1+exp(-f)));
oldProb = oldProb + sum((1-t).*(-f - log(1+exp(-f))));
% Make a grid of prediction points and compute the
% test covariance matrix
[X,Y] = meshgrid(min(x(:,1))-0.5:0.1:max(x(:,1)),...
    min(x(:,2))-0.5:0.1:max(x(:,2)));
testData = [X(:),Y(:)];
n_test = size(testData,1);
testC = zeros(n_test,size(x,1));
for n = 1:n_test
    for m = 1:size(x,1)
        testC(n,m) = alp*exp(-gam*(sum((testData(n,:)-x(m,:)).^2)));
    end
end
% This term is used for predictions and doesn't change so we can just
% compute it once
testCovTerm = diag(testC*iC*testC');
% Run the sampler
testProbs = zeros(n_test,1);
allf = zeros(N,nSamps/100);
for s = 1:nSamps
    % Propose a new sample for each observation
    % Note that we could do this for all samples at once, but
    % it becomes very hard to get a sample accepted
    order = randperm(N);
    for n = 1:N
        newf = f;
        newf(order(n)) = f(order(n)) + randn(1,1)*0.1;
        newProb = -0.5*newf'*iC*newf - sum(t.*log(1+exp(-newf)));
        newProb = newProb + sum((1-t).*(-newf - log(1+exp(-newf))));
        u = rand;
        if u <= exp(newProb - oldProb)
            f = newf;
            oldProb = newProb;
        end
    end
    % Do the predictions
    testVar = alp - testCovTerm;
    testMu = testC*iC*f;
    f_samp = randn(n_test,1).*sqrt(testVar) + testMu;
    testProbs = testProbs + 1./(1+exp(-f_samp));
end
% Plot the output
figure(4)
hold off
pos = find(t==0);
plot(x(pos,1),x(pos,2),'ko');
pos = find(t==1);
hold on
plot(x(pos,1),x(pos,2),'ko','markerfacecolor','k')
testContour = reshape(testProbs./nSamps,size(X));
contour(X,Y,testContour,5);
```

### EX 9.1. The following code does both methods of sampling and then plots the autocorrelation:

```matlab
% Generate some points data in 1-D
x = [0:0.1:1];
alp = 5;gam = 1;
N = length(x);
C = zeros(N);
for n = 1:N
    for m = 1:N
        C(n,m) = alp*exp(-gam*(x(n)-x(m))^2);
    end
end
C = C + 1e-6*eye(N);
nSamps = 10000;
% Sample 1000 values from the Gaussian
f_full = gausssamp(zeros(N,1),C,nSamps);
% Samples using Gibbs sampling
f = zeros(N,1);
all_f = zeros(nSamps,N);
for s = 1:nSamps
    for n = 1:N
        % Compute the conditional mean and covariance
        subC = C;
        subC(n,:) = [];
        subC(:,n) = [];
        subC = C(n,:);
        subC(n) = [];
        subf = f;
        subf(n) = [];
        % Note that the mean is zero so these
        % expressions are a bit simpler than in
        % the book
        % Also -- the following expression is
        % much more accurate than subC*inv(subC)*subC'
        co = C(n,n) - subC/subC*subC';
        mu = subC*inv(subC)*subf;
        f(n) = randn.*sqrt(co) + mu;
    end
    all_f(s,:) = f';
end
% Compute the autocorrelation of the nth point
n = 5;
k_vals = [0:1:200];
ac = zeros(length(k_vals),2);
N = nSamps;
full_vals = f_full(:,n);
full_vals = full_vals - mean(full_vals);
all_vals = all_f(:,n);
all_vals = all_vals - mean(all_vals);
ss = [var(full_vals), var(all_vals)];
for k = 1:length(k_vals)
    ac(k,1) = sum(full_vals(1:end-k_vals(k)).*full_vals(1+k_vals(k):end));
    ac(k,1) = ac(k,1)/(ss(1)*(N-k_vals(k)));
    ac(k,2) = sum(all_vals(1:end-k_vals(k)).*all_vals(1+k_vals(k):end));
    ac(k,2) = ac(k,2)/(ss(2)*(N-k_vals(k)));
end
figure(1)
plot(ac)
xlabel('Lag');
ylabel('Autocorrelation');
```

### EX 9.5. The following code will sample a regression dataset and then obtain an approximate posterior via ABC:

```matlab
% ABC for GP regression
x = [0:0.1:1];
alp = 5;
gam = 1;
N = length(x);
C = zeros(N);
for n = 1:N
    for m = 1:N
        C(n,m) = alp*exp(-gam*(x(n)-x(m))^2);
    end
end
C = C + 1e-6*eye(N);
noise_var = 0.1;
t = gausssamp(repmat(0,N,1),C,1)' + randn(N,1).*sqrt(noise_var);
% ABC
nSamps = 5000;
accepted_samps = [];
err_thresh = 5.0;
figure(1);hold off
plot(x,t,'k','linewidth',2);hold on
for s = 1:nSamps
    % Propose a value
    proposal = gausssamp(repmat(0,N,1),C,1)' + randn(N,1).*sqrt(noise_var);
    err = sum((proposal - t).^2);
    if err < err_thresh
        accepted_samps = [accepted_samps proposal];
        plot(x,proposal,'k','color',[0.6 0.6 0.6]);drawnow
    end
end
plot(x,t,'k','linewidth',2);
%% Compare with the true posterior
postCov = inv((eye(N)*(1/noise_var) + inv(C)));
postMu = (1/noise_var) * postCov * t;
% Generate samples
samp_f = gausssamp(postMu,postCov,1000);
figure(2);hold off
plot(x,samp_f,'k','color',[0.6 0.6 0.6]);
hold on
plot(x,t,'k','linewidth',2);
```
