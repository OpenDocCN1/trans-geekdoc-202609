```
CREATE TABLE blocked_domains (
  domain varchar(30) NOT NULL,
  block_by_varchar(10) NOT NULL,
  block_dt datetime DEFAULT '0000-00-00 00:00:00' NOT NULL,
  release_dt datetime,
  notes text,
  modify_dt timestamp(14)
);
```
