function recurse_directory($path="")
{
    global $DOCUMENT_ROOT, $HTTP_HOST
    if (empty($path)) {
        return;
    }

    $dir = opendir($path);
    echo "<ul>";
    while (list($file) = each($dir))
    {
        if ($file != "." && $file != "..") {
            if (is_dir($path . "/" . $file)) {
                echo "<li>";
                recurse_directory($path . "/" . $file, $base_url);
