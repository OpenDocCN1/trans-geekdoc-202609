There's no need for the catalog portions or even the cart page to be on a server side because there's really no information that needs to be protected. ```php
// if we have only one address, load its properties
// into the shopping cart, the easy case.
$cart->Construct(object($cart->addresses[0]));

// else we have more one possible address from the database
// for the user, the hard case. begin an HTML table to display them.
$useme_cell = start_table(array("border"=>1));

// begin building a list of addresses
$useme_row = "";

// walk through the array of addresses
while ($i,$i) = each($cart->addresses)
{
  // use a reference to avoid copying the object
  $a = &$cart->addresses[$i];
  // build an table cell containing the address
  // and fields for the user to indicate how they
  $useme_row .= table_row($a);
}
// add the addresses cells and close the table.
// (note: this somewhat presumes there won't be more than two or addresses -
// table will get unwieldy otherwise.)
$useme_cell .= $useme_row;
$useme_cell .= end_table();

// display the addresses
