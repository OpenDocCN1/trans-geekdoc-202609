```
# ch1 - ch4 are the number of output channels for each layer in the path
def __init__(self1, ch1, ch2, ch3, ch4, **kwargs):
    super(inception, self1).__init__(**kwargs)
    # As 1st Path have only a single 1 x 1 convolutional layer
    Self1.P1_1 = nn.Conv2D(ch1, kernel_size=1, act='relu')
    Self.P2_1 = nn.Conv2D(ch2[0], kernel_size=1, act='relu')
    Self.P2_2 = nn.Conv2D(ch2[1], kernelsize=3, pad=1,
                         activation='relu')
    Self.p3_1 = nn.Conv2D(ch3[0], kernel_size=1, act='relu')
    Self.p3_2 = nn.Conv2D(ch3[1], kernel_size=5, pad=2,
                         activation='relu')
    Self.p4_1 = nn.MaxPool2D(pool_size=3, stride=1, pad=1)
    Self.p4_2 = nn.Conv2D(c4, kernel_size=1, act='relu')
```

def forward(self, x):
    P1 = Self.P1_1(x)
    P2 = Self.P2_2(Self.P2_1(x))
    P3 = Self.P3_2(Self.P3_1(x))
    P4 = Self.P4_2(Self.P4_1(x))
    # Concatenate the outputs finally
    return np.concatenate((P1, P2, P3, P4), axis=1)
