### 6. 感知器学习的Python代码

```python
import numpy as np
class Perceptron(object):
    def __init__(self, no_of_inputs, threshold=50, learning_rate=0.02):
        self.threshold = threshold
        self.learning_rate = learning_rate
        self.weights = np.zeros(no_of_inputs + 1)

    def predict(self, inputs):
        summation = np.dot(inputs, self.weights[1:]) + self.weights[0]
        if summation > 0:
            activation = 1
        else:
            activation = 0
        return activation

    def train(self, training_inputs, labels):
        for _ in range(self.threshold):
            for inputs, label in zip(training_inputs, labels):
                prediction = self.predict(inputs)
                self.weights[1:] += self.learning_rate * (label - prediction) * inputs
                self.weights[0] += self.learning_rate * (label - prediction)
```

```python
def sigmoid(a):
    return 1 / (1 + np.exp(-a))
```

```python
def forward(X, W1, b1, W2, b2):
    Z = sigmoid(X.dot(W1) + b1) # sigmoid
    # Z = np.tanh(X.dot(W1) + b1) # tanh
    # Z = np.maximum(X.dot(W1) + b1, 0) # relu
    A = Z.dot(W2) + b2
    expA = np.exp(A)
    Y = expA / expA.sum(axis=1, keepdims=True)
    return Y
```

```python
# determine the classification rate
# num correct / num total
def classification_rate(Y, P):
    n_correct = 0
    n_total = 0
    for i in range(len(Y)):
        n_total += 1
        if Y[i] == P[i]:
            n_correct += 1
    return float(n_correct) / n_total
```

```python
        y_pred = m_curr*x + b_curr

        print("f=", f)
        g= sum((y-y_pred)**2)
        print("g=", g)
        cost = (1/n)*sum((y-y_pred)**2)

        print("total=",sum(x*(y-y_pred)))
        md = -(2/n)*sum(x*(y-y_pred))

        b_curr = b_curr - learning_rate * bd
        print("m_curr = , b_curr=, iteration =, md =, bd =, cost=, cost1= ", m_curr, b_curr, i, md, bd, cost, cost1)
```
