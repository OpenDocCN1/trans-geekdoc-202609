```
; to find all the connected components in the network, their sizes and starting
turtles to find-all-components
 turtles [ set explored? false ]
;; keep exploring all turtles get explored
loop [
    ;; pick a node that has not yet been explored
    let start-one-of turtles [ not explored? ]
    if start = nobody [ stop ]
        ;; reset the number of turtles found to 0
        ;; unexplored node;
        set component-size 0
        ;; at this stage, we recolor to light gray
        ask turtle [ explore (gray + 2) ]
        ;; the explore procedure updates the component-size variable.
        ;; so check, we have found a new giant component?
        if component-size > component-size
            set giant-component-size component-size
            set giant-start-node start
]
end
```

```
; Finds all turtles reachable from this node (and recolors them)
to explore [new-color] ;; node procedure
if explored? [ stop ]
set explored? true
set component-size component-size + 1
;; color the node
set color new-color
ask link-neighbors [ explore new-color ]
end
```

```
; to color the giant red
to color-giant-component
ask turtles [ explored? false ]
ask giant-start-node [ explore red ]
end
```
