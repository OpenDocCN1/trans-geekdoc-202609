[In]: def load_data(self, file_path: str):
    """
    Load stock price data from a CSV file using
    SparkSession.
    """
    try:
        df = self.spark.read.csv(
            file_path,
            header=True,
            inferSchema=True
        )
        return df
    except Exception as e:
        print(f"Error loading data: {str(e)}")
        return None
