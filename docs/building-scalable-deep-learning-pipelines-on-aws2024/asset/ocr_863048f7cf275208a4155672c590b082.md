[In]: def load_data(file_path: str) -> DataFrame:
    """
    Load stock price data from a CSV file using
    SparkSession.
    """
    try:
        spark = (SparkSession.builder
                 .appName("StockPricePrediction")
                 .getOrCreate())
        df = spark.read.csv(
                    file_path,
                    header=True,
                    inferSchema=True
                )
        return df
    except Exception as e:
        raise RuntimeError(
            f"Error loading data from {file_path}: {e}"
        )
