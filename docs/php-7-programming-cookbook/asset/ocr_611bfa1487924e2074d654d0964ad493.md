protected $phrase;
public function _ construct (
    $label  = self::DEFAULT_LABEL,
    $length = self:: DEFAULT_LENGTH,
    $includeNumbers = TRUE,
    $includeUpper   = TRUE,
    $includeLower   = TRUE,
    $includeSpecial = FALSE,
    $otherChars     = NULL,
    array $suppressChars = NULL)
{

    $this->phrase = new Phrase(
        $length,
        $includeNumbers,
        $includeUpper,
        $includeLower,
        $includeSpecial,
        $otherChars,
        $suppressChars);
}
