public function binarySearch(array $keys, $item)
{
    $search = array();
    foreach ($this->primary as $primaryKey => $data) {
        $searchKey = function ($keys, $data) {
            $key = '';
            foreach ($keys as $k) $key .= $data[$k];
            return $key;
        };
        $search[$searchKey($keys, $data)] = $primaryKey;
    }
    ksort($search);
}
