public function setTransport($transport = NULL)
{
    if ($transport) {
        $this->transport = $transport;
    } else {
        if (substr($this->uri, 0, 5) === self::TRANSPORT_HTTPS) {
            $this->transport = self::TRANSPORT_HTTPS;
        } else {
            $this->transport = self::TRANSPORT_HTTP;
        }
    }
}
