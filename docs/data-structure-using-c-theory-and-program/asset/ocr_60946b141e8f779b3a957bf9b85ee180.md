# Algorithm

Steps

+   1. Start
2. If first=NULLthen
3. Print message “UNDERFLOW” and then
4. Stop
5. ptr<-leftptr (first)
6. currptr<-leftptr
7. leftptr<-currptr
8. rightptr (currptr) <-first
9. leftptr<-avail
10. rightptr<-avail
11. avail<-ptr
12. stop

struct node *deleteatend(struct node *start)
{
  struct node *tmp;
  tmp=start;
  while(tmp->next)!=start)
    tmp=tmp->next;
  tmp->prev->next=start;
  start->prev=tmp->prev;
  free(tmp);
  return start;
}
