MERGE-SORT (A, p, r)

+   1. start
2. if p < r
3. then q = (p + r) / 2
4. MERGE-SORT(A, p, q)
5. MERGE-SORT(A, q+1, r)
6. MERGE(A, p, q, r)

MERGE (A, p, q, r)

+   1. N1 = q - p + 1
2. N2 = r - q
3. create arrays L[1.......n1+1] and R[1.......n2+1]
4. for i=1 to n1
5. do L[i] = A[p + i - 1]
6. for j =1 to n2
7. do R[j] = A[q + j]
8. L[n1+1] = ∞
9. R[n2+1] = ∞
10. i = 1
11. j = 1
12. for k = p to r
13. do if L[i] ≤ R[j]
14. then A[K] = L[i]
15. i = i + 1
16. else A[k] = R[j]
17. j = j + 1
