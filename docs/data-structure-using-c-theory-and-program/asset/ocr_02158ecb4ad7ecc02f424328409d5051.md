```
if(node==NULL)
{
    printf("\n Empty");
    return;
}
printf("\n Value of Nodes in the List is Forward Direction");
while(node!=NULL)
{
    temp=node;
    printf("%d", node->info);
    node=node->next;
}
printf("\n Value of Nodes in the List is Reverse Direction");
node=temp;
while(node!=NULL)
{
    printf("%d", node->info);
    node=node->prev;
}
```
