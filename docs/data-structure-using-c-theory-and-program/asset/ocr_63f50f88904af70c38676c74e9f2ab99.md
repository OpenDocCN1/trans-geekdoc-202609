    var->data=value;

    {
        head=var;

        head->next=NULL;
    }
    else
    {
        temp=head;
        while(temp!=NULL && temp->data!=loc)
        {
            temp=temp->next;
        }
        if(temp==NULL)
        {
            printf("\n%d is not present in list \",loc);
        }
        else
        {
            temp1=temp->next;
            temp->previous=var;
            var->next=temp1;
            temp1->previous=var;
        }
    }
    last=head;
    while(last->next!=NULL)
    {
