```
#include<stdio.h>

void main()
{
    int a[5],flag=0,m;
    printf("Enter the number of Array elements \n");
    for(i=0;i<5;i++)
    {
        scanf("%d",&a[i]);
    }
    printf("The Array Elements are:-\n");
    for(i=0;i<5;i++)
    {
        printf("%d\t",a[i]);
    }
    printf("\n Enter the number to be searched: ");
    scanf("%d",&m);
    for(i=0;i<5;i++)
    {
        if(a[i]==m)
        {
            flag=1;
            break;
        }
    }
    if(flag==1)
        printf("Element Found at position %d ",i);
    else
        printf("Element Not Found");
}
```
