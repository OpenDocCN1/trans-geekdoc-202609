struct node *deleteatend(struct node *start)
{
  struct node *tmp;
  tmp=start;
  while(tmp->next)!=start)
    tmp=tmp->next;
  tmp->prev->next=start;
  start->prev=tmp->prev;
  free(tmp);
  return start;
}
