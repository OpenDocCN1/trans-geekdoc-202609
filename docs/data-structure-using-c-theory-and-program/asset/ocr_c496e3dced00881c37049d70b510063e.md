# Algorithm

Steps

+   1. start
2. if first=NULL then
3. print "UNDERFLOW" and then
4. stop
5. curr=curr->link=first
6. repeat step 7 while curr->link!=first
7. curr=curr->link
8. ptr=ptr->first
9. first=ptr->link
10. curr->link=first
11. ptr->link=avail
12. avail=avail->ptr
13. stop

```
void deletebeg()
{
  struct node *ptr, *curr;
  if (first==NULL)
  {
    printf("\n UNDERFLOW");
    return;
  }
  curr=first
  while(curr->link!=first)
    curr=curr->link
  ptr=first;
```
