# 3. 删除节点

删除操作用于从双链表中删除一个元素。
在执行删除之前，应该记住以下几点。

+   (a) 如果列表为空，则无法删除
- (b) 如果删除后列表剩下一个节点，则列表将为空
- (c) 如果未找到要删除的节点，显示消息“未找到节点”
- (d) 如果节点被删除，则该节点的内存空间将被释放。

# 算法

步骤

+   1. 开始
2. 持有第一个节点的地址
3. 设置 temp = start
4. 如果 start = NULL 则
5. 写入“UNDERFLOW”并
6. 退出
7. 设置 start = next [start]
8. 设置 prev [start] = NULL
9. 释放由 temp 分配的空间
10. 停止

# 算法

步骤

+   1. 开始
2. 持有第一个节点的地址
3. 设置 node = start
4. 设置 temp = start
5. 如果 node = NULL 则
6. 写入“UNDERFLOW”并
7. 退出
8. 重复步骤
9. 设置 temp = node
10. 设置 node = next [node]
11. 结束步骤 8
12. 设置 temp = NULL
13. 释放与节点关联的空间
14. 停止

# 程序：删除双链表的尾部节点

```c
#include<stdio.h>
#include<stdlib.h>
struct dllnode
{
 int info;
struct dllnode *prev;
struct dllnode *next;
};
struct dllnode *start;
void create(struct dllnode *);
void delete(struct dllnode *);
void display(struct dllnode *);
void main()
{
  struct dllnode *node;
  char ch='1';
  node=(struct dllnode*)malloc(sizeof(struct dllnode));
  if(node==NULL)
  {
    printf("\n 内存不足\n");
    exit(0);
  }
  start=node;
  create(node);
  while(ch=='3')
  {
    printf("\n 1. 删除尾部\n");
    printf("\n 2. 显示\n");
    printf("\n 3. 退出\n");
    printf("\n 输入您的选择:\n");
    ch=getchar();
    switch(ch)
    {
        case '1':
            delete(node);
            break;
        case '2':
            display(node);
            break;
        case '3':
            break;
        default:
            printf("选择错误\n");
    }
  }
}
void delete (struct dllnode *node)
{
    struct dllnode *temp;
    if(node->next==NULL)
    {
        temp=node;
        node=node->next;
        start=node;
        free(temp);
        return;
    }
    while(node->next!=NULL)
    {
        node=node->next;
        temp=node;
        node=node->prev;
    }
    node->next=NULL;
    free(temp);
}
void display(struct dllnode*node)
{
    struct dllnode *temp;
    if(node==NULL)
    {
        printf("\n 空列表");
        return;
    }
    printf("\n 列表中节点的值（正向方向）");
    while(node!=NULL)
    {
        temp=node;
        printf("%d", node->info);
        node=node->next;
    }
    printf("\n 列表中节点的值（反向方向）");
    node=temp;
    while(node!=NULL)
    {
        printf("%d", node->info);
        node=node->prev;
    }
}
```
