# Algorithm

Steps

+   1. start
2. if start=NULL then
3. print “OVERFLOW” and then
4. stop
5. ptr->ptr->start
6. start=start->link
7. read info
8. curr->curr->link
9. repeat step 10 while curr->link!=first
10. curr->curr->link
11. curr->link=ptr
12. ptr->link=first
13. stop

```
void insertatend()
{
  struct node *ptr, *curr;
  ptr=(struct node *) malloc(sizeof(struct node));
  if(ptr==NULL)
  {
    printf("OVERFLOW\n");
    return;
  }
  printf("Input New Node Information");
  scanf("%sd", &ptr->info);
  curr=first;
  while(curr->link!=first) {
    curr=curr->link;
  }
  curr->link=ptr;
  ptr->link=first;
}
```
