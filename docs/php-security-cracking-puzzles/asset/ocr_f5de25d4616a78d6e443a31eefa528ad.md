// Removing extra spaces
$POST['name'] = trim($POST['name']);
$POST['pass'] = trim($POST['pass']);
$POST['pass_again'] = trim($POST['pass_again']);
// Checking that the $ POST array is empty
if(empty($POST['name'])) exit();
// Checking that the required fields are properly filled
if(empty($POST['name'])) exit('The 'Name' field is not filled!');
if(empty($POST['pass'])) exit('Both 'Password' fields must be filled!');
if(empty($POST['pass_again'])) exit('Both 'Password' fields must be filled!');
if($POST['pass'] != $POST['pass_again']) exit('The passwords do not match!');
// If an email address is supplied, checking its format
if(empty($POST['email'])) {
    if(!preg_match('/^[0-9a-z._%+-]+@[0-9a-z._-]+\.[a-z]{2,6}$/i', $POST['email'])) {
        exit(\'The \'email\' address must be in the format somebody@somewhere.com.\');
    }
}
n// If magic quotes are not enabled on the
// server, processing the input data
// with the mysql_escape_string() function
if (!get_magic_quotes_gpc()) {
    $POST['name'] = mysql_escape_string($POST['name']);
    $POST['pass'] = mysql_escape_string($POST['pass']);
    $POST['email'] = mysql_escape_string($POST['email']);
