$filename = "text.txt";
// Checking whether the supplied name
// is already used
$arr = file($filename);
foreach($arr as $line) {
    // Breaking the line at the :: dividers
    $data = explode("::", $line);
    // Placing the names of already-registered users
    // into the $temp array
    $temp[] = $data[0];
}
// Checking whether the current name
// is in the $temp array
if(in_array($_POST['name'], $temp)) {
    exit("The user name is already used. Select another one.");
}
/////////////////////////////////////////////////
// The user registration block
/////////////////////////////////////////////////
// Placing the data into a text file
$fd = fopen($filename, "a");
if(!$fd) exit("Error opening the file!");
$str = $_POST['name']."::".
      $_POST['pass']."::".
      $_POST['email']."::".
      $_POST['url']."::".\r\n";
fwrite($fd,$str);
fclose($fd);
// Reloading the page to clear the POST data
echo "<HTML><HEAD>
<META HTTP-EQUIV = 'Refresh' = 0; URL = \$_SERVER['PHP_SELF']>
</HEAD></HTML>";?>
