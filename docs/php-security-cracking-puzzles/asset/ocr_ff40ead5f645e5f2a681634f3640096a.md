# Listing II.3.2. Editing remote files

```
<?php
// If the edited file contents are uploaded,
// opening the file and rewriting it
if(isset($_POST['content']))
{
    // Opening the file
    $fd = @fopen($_POST['filename'], "w");
    // If the file cannot be open,
    // displaying a warning to this effect
    if(!$fd) exit("No such file!");
    // Rewriting the file's contents
    fwrite($fd, stripslashes($_POST['content']));
    // Closing the file
    fclose($fd);
    // Placing the file's name into
    // the GET superglobal array
    $_GET['filename'] = $_POST['filename'];
}
?>
<form method="get">
    File name <input type="text" name="filename" 
    value="<?php echo $_GET['filename']; ?>">
    <input type="submit" value="Open">
</form>
```
