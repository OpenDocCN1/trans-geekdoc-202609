# Chapter II.8: Using Information from Other Sites

# Listing II.8.13. Extracting currency exchange rates

```
<?php
// Setting unlimited execution time for the script
set_time_limit(0);
// The page to search
$url = "http://www.ecb.eu/stats/eurofref/eurofref-daily.xml";
// Loading the contents of the page
$contents = file_get_contents($url);

// Extracting the date
$pattern = "|Cube time=([^\']+)'|is";
// Performing a search against the regular expression
preg_match($pattern, $contents, $out);
echo "The exchange rate for \.$out[1].<br><br";

// Extracting the currency exchange rates
$pattern = "|Cube currency=([^\']+)' rate='([^\']+)'|is";
// Performing a search against the regular expression
preg_match_all($pattern, $contents, $out, PREG_PATTERN_ORDER);
// Displaying the results of the search
for($i=0; $i < count($out[1]); $i++) {
    echo "1. \$out[1][$i]. - " . \$out[2][$i]." EUR<br";
}
?>
```
