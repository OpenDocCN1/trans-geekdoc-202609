# Listing I.7.2. Authorization using cookies

```
<?php
 // The form handler
if(!empty($_POST))
{
 // Setting the connection to the database
 require_once("config.php");
 // To protect yourself against SQL injection,
 // pass the supplied login name and password through the
 // mysql_escape_string function
 if (!get_magic_quotes(gpc()))
{
  $_POST['name'] = mysql_escape_string($_POST['name']);
  $_POST['password'] = mysql_escape_string($_POST['password']);
 // Executing the query that will return
 // the number of records matching
 // the password and the login
 $query = "SELECT COUNT(*) FROM userslist
 WHERE name = '$_POST[name]' AND pass = '$_POST[password]';
 $usr = mysql_query($query);
 if(!isset($usr)) exit("Error in authorization block");
 // Obtaining the number of records
 $Total = mysql_result($usr, 0);
 if($Total > 0)
{
 // Authorization was successful.
 // Setting the cookie for 24 hours (3600*24)
 setcookie("user", urlencode($_POST['name']), time() - 3600*24);
 // Reloading to reset POST data
 echo "<HTML><HEAD>
<META HTTP-EQUIV='Refresh' CONTENT=0; URL=$_SERVER['PHP_SELF']'>
</HEAD></html>"; ?>
```
