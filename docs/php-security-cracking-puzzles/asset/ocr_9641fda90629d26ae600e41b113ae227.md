// The password character enumeration function
// $pass - The password to guess
// $answer - The current answer (at the first call, an empty string)
function decrypt_md5($pass, $answer)
{
  $arr = array('a', 'b', 'c', 'd', 'e', 'f', 'g',
               'h', 'i', 'j', 'k', 'l', 'm', 'n',
               'o', 'p', 'q', 'r', 's', 't', 'u',
               'v', 'w', 'x', 'y', 'z');
  // Assuming that the password is
  // no longer than four characters
  $max_number = 3;
  if (strlen($answer) > $max_number) return;

  for($j = 0; $j < count($arr); $j++) {
    $temp = $answer.$arr[$j];
    if (md5($temp) == $pass) return $temp;
    // Recursively calling the function to increment
    // the length of the password being guessed
    $result = decrypt_md5($pass, $temp);
    // If the function returns a non-empty string,
    // // the password has been guessed and no further
    // search is necessary.
    if (strlen($result) > 0) return $result;
  }
}
