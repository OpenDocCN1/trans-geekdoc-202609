function scan_dir($dirname)
{
    // Opening the current directory
    $dir = opendir($dirname);
    // A loop to read the current directory
    while (($file = readdir($dir)) != false)
    {
        // Checking whether the $file variable equals
        // the current or the upper-level directory
        if($file != "." && $file != "..")
        {
            // If the current entry is a directory,
            // calling the scan_dir() function recursively
            if(is_dir($dirname."/"."$file))
            {
                echo $dirname."/"."$file." - ".";
                file_count($dirname."/"."$file)."<br>";
                scan_dir($dirname."/"."$file);
            }
        }
    }
    // Closing the directory
    closedir($dir);
}
