import pandas as pd
import matplotlib
import matplotlib.pyplot as plt

def azurerml_main(dataframe1):
    matplotlib.use("agg")
    size = len(dataframe1)
    df2 = pd.DataFrame()
    n = []
    d = []
    m = []
    p = []
    count = 1
    num = 1
    temp = 0
    for x in range(size):
        temp += dataframe1.at[x, 'AES']
        date = dataframe1.at[x, 'Date']
        month = dataframe1.at[x, 'Month']
        if count == 24:
            s = temp / 24
            p.append(s)
            n.append(num)
            d.append(date)
            m.append(month)
            num += 1
            count = 0
            temp = 0
        count += 1
    data = {'Num': n, 'Date': d, 'Month': m, 'AES': p}
    df2 = pd.DataFrame(data)
    # plotting a line graph
    fig = plt.figure()
    plt.plot(df2["Num"], df2["AES"])
    plt.xlabel("Days")
    plt.ylabel("Power, kw")
    plt.grid()
    fig.savefig("chart.png")
    # Return value of pandas.DataFrame
    return df2,
