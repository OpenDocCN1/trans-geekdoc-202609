import pandas as pd
import numpy as np

```python
# Data Preprocessing
dataset_train = pd.read_csv("Google_Stock_Price_Train.csv") # Notice train-set is now our main dataset
training_set = dataset_train.iloc[:1,2].values
```

```python
# feature scaling
from sklearn.preprocessing import MinMaxScaler
sc = MinMaxScaler(feature_range=(0, 1))
train_set_scaled = sc.fit_transform(training_set)
```

```python
print(f"Size of train set : {train_set_scaled.shape}")
print(f"No. of data-points : {train_set_scaled.shape[0]}")
```

```python
# creating a structure with 60 time-steps and 1 output.
data_size = train_set_scaled.shape[0]
X_train = []
y_train = []
```

```python
for i in range(60, data_size):
    X_train.append(train_set_scaled[i-60:i,:]) # append a list of 60 data-points in list (data_size-60) rows
    y_train.append(train_set_scaled[i,0]) # vector of (data_size-60) rows
```

```python
# Convert X_train and y_train to NumPy array.
X_train, y_train = np.array(X_train), np.array(y_train)
```

```python
# Reshaping
X_train = np.reshape(X_train, (X_train.shape[0], X_train.shape[1], 1))
```

```python
# Building RNN
from keras.models import Sequential
from keras.layers import Dense
from keras.layers import LSTM
from keras.layers import Dropout
```

```python
# Initialize the RNN
regressor_rnn = Sequential()
```

```python
# adding LSTM Layer with Dropout-Regularization
regressor_rnn.add(LSTM(units=50, return_sequences=True, input_shape=(X_train.shape[1], 1)))
regressor_rnn.add(Dropout(rate=0.2))
```

```python
# second LSTM with Dropout-Regularization
regressor_rnn.add(LSTM(units=50, return_sequences=True))
regressor_rnn.add(Dropout(rate=0.2))
```

```python
# third LSTM Layer with Dropout-Regularization
regressor_rnn.add(LSTM(units=50, return_sequences=True))
regressor_rnn.add(Dropout(0.2))
```

```python
# Fourth LSTM (last LSTM with Dropout-Regularization)
regressor_rnn.add(LSTM(units=50))
regressor_rnn.add(Dropout(0.2))
```

```python
# the output Layer
regressor_rnn.add(Dense(units=1))
```

```python
# Compiling the Model
regressor_rnn.compile(optimizer='adam', loss='mean_squared_error')
```

```python
# fit the model to training dataset (model Learn)
# The training will happen in this part.
regressor_rnn.fit(X_train, y_train, epochs=10, batch_size=32)
```
