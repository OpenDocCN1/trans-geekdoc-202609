# --------- Naive Bayes model
# Library
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

# Data Extract
dataset = pd.read_csv("Social_Network_Ads.csv")nX = dataset.iloc[:,[2,3]].values
y = dataset.iloc[:,3].values

# Feature-Scaling after Data Split
# Data Split
from sklearn.model_selection import train_test_split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.25, random_state=0)
# Feature-Scaling
from sklearn.preprocessing import StandardScaler
st = StandardScaler()
X_train = st.fit_transform(X_train)
X_test = st.transform(X_train)

# Fit model to Random Forest classifier: No parameter is needed
from sklearn.ensemble import RandomForestClassifier
classifier = RandomForestClassifier(estimator=100, criterion="entropy", random_state=0)
classifier.fit(X_train, y_train)
# Predict
y_pred = classifier.predict(X_test)
# Making the confusion matrix use the function confusion_matrix
from sklearn.metrics import confusion_matrix
cm = confusion_matrix(y_test, y_pred)
# parameters of cm: return=True, y_true: actual values, y_pred: predicted value

# Visualising the test set results
from matplotlib.colors import ListedColormap
X_set, y_set = X_test, y_test
X1, X2 = np.meshgrid(np.arange(X_set[:,0].min(), X_set[:,0].max() + 1, 0.01), np.arange(X_set[:,1].min(), X_set[:,1].max() + 1, 0.01))
plt.contour(X1, X2, classifier.predict(np.array([X1.ravel(), X2.ravel()]).T).reshape(X1.shape), alpha=0.25, cmap=ListedColormap(['red','green']))
plt.scatter(X_set[:,0], X_set[:,1], c=y_set, cmap=ListedColormap(['red','green']))
for i, j in enumerate(np.unique(y_set:
    plt.scatter(X_set[y_set == j,0], X_set[y_set == j,1], label=j)
plt.title("Random Forest (testing set")
plt.xlabel("Age")
plt.ylabel("Estimated Salary")
plt.legend()
plt.show()
