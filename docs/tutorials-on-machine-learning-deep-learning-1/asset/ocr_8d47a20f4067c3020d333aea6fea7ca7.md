def cell(self, input, hidden):
    # Cell that robustly activates inside if statements
    gate = sigmoid(self.W_gate @ input + self.U_gate @ hidden + self.b_gate)
    candidate = tanh(self.W_cand @ input + self.U_cand @ hidden + self.b_cand)
    output = sigmoid(self.W_output @ input + self.U_output @ hidden + self.b_output)
    state = gate * tanh(self.W_state @ input + self.U_state @ hidden + self.b_state)
    res = output * candidate
    return res, state
