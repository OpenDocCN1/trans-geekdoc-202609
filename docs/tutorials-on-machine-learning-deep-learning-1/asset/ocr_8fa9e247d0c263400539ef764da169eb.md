# Evaluating ANN : k-fold CV
from sklearn.wrappers import KerasClassifier
from sklearn.model_selection import cross_val_score

# make a function that creates ANN
from keras.models import Sequential
from keras.layers import Dense

```
# initialize the ANN model
def ann_classifier():
ann = Sequential())
ann.add(Dense(units=6, kernel_initializer='uniform', activation='relu', input_dim=11)) # "First Hidden-Layer"
ann.add(Dense(units=6, kernel_initializer='uniform', activation='relu')) # "Second Hidden-Layer"
ann.add(Dense(units=1, kernel_initializer='uniform', activation='sigmoid')) # "Output-Layer"
ann.compile(optimizer='adam', loss='binary_crossentropy', metrics=['Accuracy']) # Compile the ANN
return ann
```

```
acuracies = cross_val_score(estimator=ann_classifier(), x=x_train, y=y_train, cv=10, n_jobs=5) # Compile ANN classifier with 10-fold
```
