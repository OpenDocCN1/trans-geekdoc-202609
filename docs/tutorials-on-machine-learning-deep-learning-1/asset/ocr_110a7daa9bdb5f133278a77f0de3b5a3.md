class HDR(nn.Module):
    def __init__(self, nv, nh):
        self.w = torch.randn(nh, nv)
        self.a = torch.randn(1, nh)
        self.b = torch.randn(1, nh)
```

```
    def forward(self, x):
        xx = torch.matmul(x, self.w)
        activation = xx + self.a.expand_as(xx)
        p_given = torch.sigmoid(activation)
        return p_given, p_given.bernoulli(p_given)

    def sample(self, y):
