# Training the SAE model --
b_epoch = 200
for epoch in range(1, nb_epoch+1):
    train_loss = 0
    s = 0
    for user_id in range(nb_users):
        input = (train_set_tensor[user_id]).squeeze(0)
        target = input.clone()
        non_zero_ratings = torch.sum(target.data > 0)
        if non_zero_ratings > 0:
            output = sae(input)
            target.require_grad = False
            loss = criterion(output, target)
            mean_corrector = loss.mean() / nb_movies/float(non_zero_ratings + 1e-10)
            loss.backward()
            train_loss += np.sqrt(loss.data)*mean_corrector
            s += 1.0
    optimizer.step()
