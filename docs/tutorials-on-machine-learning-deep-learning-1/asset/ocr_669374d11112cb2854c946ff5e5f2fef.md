# # All code for k-fold-validation at once (practiced version)

# Library
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np

# Data Extract
dataset = pd.read_csv('Social_Network_Ads.csv')
X = dataset.iloc[:, [2,3]].values
y = dataset.iloc[:, 4].values

# Data Split
from sklearn.model_selection import train_test_split
X_train, X_test, y_train, y_test = train_split(X, y, test_size=0.25, random_state=0)

# Feature-Scaling
from sklearn.preprocessing import StandardScaler
sc = StandardScaler()
X_train = sc.fit_transform(X_train)
X_test = sc.transform(X_test)

# Fit set to Kernel-SVM classifier
from sklearn.svm import SVC
linear-data-points are non-separable, use "rbf" : Gaussian kernel gives better result.
kernel="rbf" instead of "linear"
clfier = SVC(kernel='rbf', random_state=0)
clfier.fit(X_train, y_train)

# Predict
y_pred = clfier.predict(X_test)
# Making the confusion matrix use the "confusion_matrix"
from sklearn.metrics import confusion_matrix
cm = confusion_matrix(y_test, y_pred, y_pred)
# parameters x=y_true: Real values, y_pred: Predicted value

# Applying K-fold cross-validation
from sklearn.model_selection import cross_val_score
accuracies = cross_val_score(clfier, X_train, y_train, cv=10)
mean_acc = accuracies.mean()
_acc = accuracies.std()

# Visualising the Training set
import matplotlib.pyplot as plt
plt.figure(figsize=(10,8))
plt.scatter(X_train[:,0], X_train[:,1], c=y_train, cmap='viridis')
X1, X2 = np.arange(X_train[:,0].min(), X_train[:,0].max()+1, 0.01)
Y1, Y2 = np.arange(X_train[:,1].min(), X_train[:,1].max()+1, 0.01)
X1, Y1 = np.meshgrid(X2, Y1)
Z = clfier.decision_function(np.c_[X1.ravel(), X1.ravel()]).reshape(X1.shape)
plt.contour(X1, Y1, Z, alpha=0.25, colors='red')
plt.title('Kernel-SVM (Training set)')
plt.xlabel('Age')
plt.ylabel('Estimated Salary')
plt.legend()
plt.show()

# Visualising the Test set
import matplotlib.pyplot as plt
plt.figure(figsize=(10,8))
plt.scatter(X_test[:,0], X_test[:,1], c=y_test, cmap='viridis')
X1, X2 = np.arange(X_test[:,0].min(), X_test[:,0].max()+1, 0.01)
Y1, Y2 = np.arange(X_test[:,1].min(), X_test[:,1].max()+1, 0.01)
X1, Y1 = np.meshgrid(X2, Y1)
Z = clfier.decision_function(np.c_[X1.ravel(), X1.ravel()]).reshape(X1.shape)
plt.contour(X1, Y1, Z, alpha=0.25, colors='red')
plt.title('Kernel-SVM (Test set)')
plt.xlabel('Age')
plt.ylabel('Estimated Salary')
plt.legend()
plt.show()
