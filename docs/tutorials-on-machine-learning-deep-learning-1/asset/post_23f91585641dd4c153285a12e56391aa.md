import pandas as pd

dataset = pd.read_csv("Position_Salaries.csv")

# Feature-Scaling
from sklearn.preprocessing import StandardScaler
sc_x = StandardScaler()

y_scaled = sc_y.fit_transform(y.reshape(-1, 1))

# Data Split : No need for this example
