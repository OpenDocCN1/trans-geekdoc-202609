This function returns the result set based on a user_id.

**get_answers()** This function returns an array of answers associated with a question, along with the total number of votes so far for each answer.

function get_answers($question_id=0)
{
	$question_id = (int)$question_id;
	$query = "select a.answer, a.answer_id, count(r.user_id) as
votes
		from answers a
			left join responses r on a.answer_id = r.answer_id
		where a.question_id = $question_id
		group by a.answer_id
		having votes > 0
		order by votes desc
	";
	$answers = array();
	$result = my_query($query);
	while ($row = mysql_fetch_assoc($result))
	{
		$answers[] = $row;
	}
	mysql_free_result($result);
	return $answers;
}
