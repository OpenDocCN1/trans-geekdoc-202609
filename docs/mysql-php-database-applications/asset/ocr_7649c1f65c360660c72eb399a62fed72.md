foreach (array('PHP_AUTH_USER','PHP_AUTH_PW') as $v)
{
    if (!isset($_SESSION[$v]))
    {
        if ($http_auth_OK)
        {
            $_SESSION[$v] = array_key_value($_SERVER,$v,'');
        }
        else
        {
            $_SESSION[$v] = array_key_value($_POST,$v,'');
        }
    }
}
