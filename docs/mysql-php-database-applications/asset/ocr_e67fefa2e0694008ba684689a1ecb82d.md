function mysql_session_read($id)
{
    $data = "";
    $query = "select sess_data from sessions
    where session_id = '$id'";
    $result= mysql_query($query) or die (mysql_error());
    if ($row = mysql_fetch_row($result) )
    {
        $data=session_decode($row[0]);
    }
    return $data;
}
