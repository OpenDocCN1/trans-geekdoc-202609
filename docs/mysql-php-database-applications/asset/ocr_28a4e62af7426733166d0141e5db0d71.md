                $result[$key] = $arg;
            }

            if ($key == NULL)
            {
                user_error("Argument '$arg' was passed with no

            }
            if (isset($result[$key]))
            {
                if (is_array($result[$key]))
                {
                    $result[$key] = array($result[$key]);
                }
                $result[$key][] = $arg;
            }
            else
            {
                $result[$key] = $arg;
            }
        }
    }
