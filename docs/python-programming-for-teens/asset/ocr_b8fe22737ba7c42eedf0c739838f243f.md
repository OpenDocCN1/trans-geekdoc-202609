Note that one callback function is passed to all the menu items when they are created. This time, the callback changes the color of the system turtle when you click a menu item.

```
```
"""
testmenuitem.py
A simple tester program for menu items.
"""

from turtle import *
from menuitem import MenuItem

def changePenColor(c):
    """Changes the system turtle's color to c."""
    color(c)

def createMenu(callBack):
    """Displays 6 menu items to respond to the given callback function."""
    x = - (window_width() / 2) + 30
    y = 100
    colors = ("red", "green", "blue", "yellow", "black", "purple")
    shape = "circle"
    for color in colors:
        MenuItem(x, y, shape, color, callBack)
        y -= 30
