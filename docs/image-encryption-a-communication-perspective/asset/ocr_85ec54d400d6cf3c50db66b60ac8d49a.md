%Convert 256 base number to decimal

for i = 1:4
    y = y+(x(i)*256^(i-1));
end

function y = LSB5(x)
%return the 5 least significant bits of x
y = dec2bin(x(1),8);
u = y(8:-1:3); u = u(end:-1:1);
y = bin2dec(u);
