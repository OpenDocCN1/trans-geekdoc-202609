function s = keygen(key,r)
%RC6密钥生成
p = [99 81 225 91];
q = [185 121 55 158];
s(1,:) = p;
for i = 2:2*r+4
    s(i,:) = add(s(i-1,:),q);
end;
i = 1;
a = zeros(1,4);
b = zeros(1,4);
v = 3*(2*r+4);
for h = 1:v
    s(i,:) = add(s(i,:),a);
    s(i,:) = add(s(i,:),b);
    s(i,:) = shifting(s(i,:),3);
    a = s(i,:);
    key = add(key,a);
    key = add(key,b);
    key = shifting(key,LSB5(add(a,b)));
    b = key;
    if i == 2*r+4
        i = 1;
    end
end

function y = RC6sub(x,w)
%z = a-almod 2^32
a = frombase256(x);
a1 = frombase256(w);
z = mod(a-a1,2^32);
y = tobase256(z);

function out = shifting (w,n)
% 将w移位n位
y = dec2bin(w,8);
y = rot90 (y); y = y(:);
y = circshift(y,n);
y = reshape(y,8,4)';
out = bin2dec(y(:,end:-1:1));

function y = shiftleft(key,round)
% RC6移位
if (round ==1||round ==2||round ==9||round ==16)
    y = circshift(key,1); else
    y = circshift(key,2); end

function y = tobase256(x)
%将十六进制转换为256位基数
for i = 4:-1:1
    y(i) = fix(x/256^(i-1)); x = x-(y(i)
        *256^(i-1));
end

function c = add(a,b)

% c = a+b mode 2^8
c = zeros(1,4);
for i = 1:4
    c(i) = mod(a(i)+
    b(i),2^8); if
    (i+1) ~= 5
        a(i+1) = a(i+1)+fix((a(i)+b(i))/256);
    end;
end;

function y = binvec2decA(x)
%将二进制转换为十进制
y = 0;
for i = 1:length(x)
    y = y+x(i)*2^(i-1);
end;

function cipherdes1 = cipher(plaindes1,r,keyi)
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%此文件用于将明文加密为
ciphertext using des
% clc
% clear all bin = zeros(8);
for i = 1:8
    bin(i,:) = dec2binvecA(plaindes1(i),8);
end;
plaindes = rot90(bin);
plaindes = plaindes(:);
ip = InitialPermutation(plaindes);
left = ip(1:32);
right = ip(33:64);
for round = 1:r
    expansion = exp1(right);
    xor_one = bitxor(expansion',keyi(round,:));
    substitution = DESsub(xor_one);
    permutation = p(substitution);
    xor_two = bitxor(left,permutation);
    if round ~= r
        left = right;
        right = xor_two;
    else
        left = xor_two;
    end;
end;
y(1:32) = left; y(33:64) = right;

function y = Convert256toHex(x)
%将256位基数转换为十六进制
a = length(x);
j = 1;
y = zeros(1,2*a);
for i = 1:a
    y(j) = fix(x(i)/16); y(j+1) = x(i)-y(j)*16; j = j+2;
end

function y = dec2binvecA(x,size)
%将十进制转换为二进制
y = zeros(1,size);
z = zeros(1,size+1);
z(1) = x;
for i = 1:size
    y(i) = mod(z(i),2^i)/2^(i-1);
    z(i+1) = z(i)-y(i)*2^(i-1);
end;
