function [Seq,ScaleS,ScaleD] = seqdual(Seq,ScaleS,ScaleD)
% Dual of a lifting sequence

L = size(Seq,1);

for k = 1: L
  % f'(z) = -f(z^-1)
  Seq{k,2} = -(Seq{k,2} - length(Seq{k,1}) + 1);
  Seq{k,1} = -fliplr(Seq{k,1});
end

if all(Seq{1,1} == 0)
  Seq = reshape({Seq{2: end,:}},L-1,2);
else
  [Seq{1: L+1,:}] = deal(0,Seq{1: L,1},0,Seq{1: L,2});
end

ScaleS = 1/ScaleS;
ScaleD = 1/ScaleD;
return;
