function ciphertextDES = DESOFB(plaintext1,m,n,CO,r,keyi)
%Encrypt data using DES in OFB mode

ciphertextDES = zeros(m*n/8,8);
for i = 1:m*n/8
plaintext = plaintext1(i,:);

if i == 1
cr = cipher(CO,r,keyi);

else
cr = cipher(cr,r,keyi);

end;
ciphertext = bitxor(plaintext,cr);
ciphertextDES(i,:) = ciphertext;

end;
