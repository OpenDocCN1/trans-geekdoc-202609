    temp = d; d = c; c = b; b = a;
    a = temp;
    u = multi(d,(2*d+1)); u = shifting(u,5);
    t = multi(b,(2*b+1)); t = shifting(t,5);
    c = RC6sub(c,s(2*i+2,:)); c = shifting(c,-LSB5(t)); c = bitxor(c,u);
    a = RC6sub(a,s(2*i+1,:)); a = shifting(a,-LSB5(u)); a = bitxor(a,t);
end d = RC6sub(d,s(2,:));
