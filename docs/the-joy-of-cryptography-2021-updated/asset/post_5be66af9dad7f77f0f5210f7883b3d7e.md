## One-time secrecy for public-key encryption (Definition 15.4):

L^\Sigma_{pk-ots-L}
(pk, sk) <- Sigma.KeyGen
count := 0
GETPK():
return pk
CHALLENGE(m_L, m_R \in Sigma.M):
count := count + 1
if count > 1: return null
return Sigma.Enc(pk, m_L)

L^\Sigma_{pk-ots-R}
(pk, sk) <- Sigma.KeyGen
count := 0
GETPK():
return pk
CHALLENGE(m_L, m_R \in Sigma.M):
count := count + 1
if count > 1: return null
return Sigma.Enc(pk, m_R)
