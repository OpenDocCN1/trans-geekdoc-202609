def __call__(self, input, hidden):
    return self.forward(input, hidden)

def evaluate(self, encoder_outputs, max_length):
    hidden = encoder_outputs[-1]
    hidden = hidden.reshape(1, hidden.shape[0], hidden.shape[1])
    input_T = self.encoder_outputs.shape[0]
    # encoder_outputs = np.pad(self.encoder_outputs, ((0, self.max_length - input_T), (0, 0), (0, 0)), 'constant')
    # input: (1, batch_size=1, input_size)
    input = np.array([SOS_token])
    decoded_words = []
    for t in range(max_length):
        output, hidden, _ = self.forward_step(input, hidden, encoder_outputs, False)
        output = np.argmax(output)
        if output == EOS_token:
            break
        else:
            decoded_words.append(output)
            input = np.array([output])
    return decoded_words

def backward(self, dZs):
    input_T = self.encoder_outputs.shape[0]
    d_encoder_outputs = np.zeros_like(self.encoder_outputs)
    T = len(dZs)
    dprev_hidden = np.zeros_like(self.h_0)
    for i in reversed(range(T)):
        self.out.x = self.out_x[i]
        dh = self.out.backward(dZs[i])
        dh += dprev_hidden[-1]
        dhs = np.expand_dims(dh, axis=0)
        self.gru.hs = self.gru_hs[i]
        self.gru.zs = self.gru_zs[i]
        relu_out, self.gru.h = self.gru_x[i]
        drelu_out, dprev_hidden = self.gru.backward(dhs, relu_out)
        drelu_out = drelu_out.reshape(drelu_out.shape[1], drelu_out.shape[2])
        self.relu.x = self.relu_x[i]
        d_relu_x = self.relu.backward(drelu_out)
        d_attn_combine_out = d_relu_x
        self.attn_combine.x = self.attn_combine_x[i]
        d_attn_combine_x = self.attn_combine.backward(d_attn_combine_out)
        d_embedded, d_attn_out = d_attn_combine_x[:, :self.hidden_size], d_attn_combine_x[:, self.hidden_size:]
        self.attn.alphas, self.attn.energies, self.attn.hidden, self.attn.encoder_outputs = self.attn_x[i]
        dprev_hidden_2, d_encoder_outputs_2 = self.attn.backward(d_attn_out)
        if self.use_dropout:
            self.dropout._mask = self.dropout_mask[i]
            d_embedding = self.dropout.backward(d_embedded)
        else:
            d_embedding = d_embedded
        self.embedding.x = self.embedded_x[i]  # recover the original x when do forward
        self.embedding.backward(d_embedding)
        dprev_hidden += dprev_hidden_2
        d_encoder_outputs += d_encoder_outputs_2  # [:input_T] #Every moment must be accumulated
    # d_encoder_outputs[input_T-1] += dprev_hidden[0]
    d_encoder_outputs[-1] += dprev_hidden[0]
    return dprev_hidden, d_encoder_outputs  # dhidden

        for layer in self.layers:
            for i, _ in enumerate(layer.params):
                self._params.append([layer.params[i], layer.grads[i]])
    return self._params

# 数据是机器学习和现代人工智能的基础

数据越多，机器学习算法的性能越好。正是因为海量数据的存在，大公司才能开发出搜索引擎、推荐系统、智能游戏等高性能人工智能产品。常言道：“谁拥有数据，谁就拥有未来。”大数据也是神经网络重现辉煌和深度学习发展的关键因素之一。

对于许多问题，人工获取数据（如医学影像数据）通常非常困难且昂贵。例如，要提升人脸识别等人脸相关算法的性能，需要大量人脸图像数据，而这些人脸图像的采集不仅需要用户授权，还需要支付一定代价。如果能自动生成难以与真实人脸区分的人脸图像，就能节省成本和费用，推动人脸相关研究与应用。另一个例子是，视频游戏影视作品中包含大量二维和三维场景。设计制作这些场景需要投入大量人力、物力和财力，导致拍摄电影和制作游戏的成本极高。如果能自动生成这些影视作品中的高质量场景，就能节省大量财力和人力，使产品开发者能更专注于创意工作。

机器学习中的**生成模型**专门研究如何利用计算机自动生成与真实数据相似的数据，即生成模型能自动生成难以与真实数据区分的假数据。第 7 章自然语言理解中的语言模型就是一种生成模型。优秀的语言模型能为机器翻译、聊天对话、文章生成等应用生成流畅的句子。通常，一旦训练完成，循环神经网络就可用于源源不断地生成序列数据。

因此，自动生成的数据可以解决许多研究问题的数据匮乏问题，不仅能提升相关问题上机器学习算法的性能，还能促进各类应用产品的发展。例如，自动人脸生成技术用于视频换脸（如 `DeepFake`）等各类人脸应用问题；自动图像生成可生成各种风格的图像；自动语音合成技术可自动合成各种类似真人的声音；以及自动作曲等。

本章主要讨论基于深度神经网络（深度学习）的两种最流行的生成模型技术：**变分自编码器**（Variational Auto-Encoders，**VAE**）、**生成对抗网络**（Generative Adversarial Network，**GAN**）。

## 8.1 生成模型

以生成人脸图像为例，如何自动生成与真实人脸相同的人脸？如果对图像的不同像素进行涂鸦式的任意着色，显然不可能生成看起来像真实图像的图像，更别提人脸了。

世界上每张人脸图像都不同，但无论差异多大，人们一眼就能分辨出一张图像是人脸，而不是猫、狗或植物。如果用形状相同的张量（如三维张量）表示所有人脸图像，即红、绿、蓝三色图像表示，例如用一个`3 × 1024 × 768`的张量表示一张人脸图像，包含`1024 × 768`个像素，每个像素由红、绿、蓝三色值表示，则一张人脸图像包含`3 × 1024 × 768`个变量值。设`x`代表该张量，则`x`是`3 × 1024 × 768`维线性空间中的数据点，或每张人脸图像对应的`x`对应该线性空间中的一个坐标点。该空间中所有真实人脸图像对应的`x`数据点并非随机分布，通常位于该空间的一个小子空间内，就像二维平面上直线上的点分布在该直线上一样。`x`是一个变化的随机变量。该大线性空间中所有真实人脸图像对应的`x`坐标点分布有其特定的概率分布规律或概率分布密度，但这种概率分布无法用解析数学表达式表示。

如果某个模型能自动生成难以与真实人脸图像区分的人脸图像，那么这些自动生成的图像必然服从真实人脸图像的底层概率分布。

因此，生成建模的核心在于生成与真实数据具有相同（或尽可能相似）概率分布的人工数据。例如，真实数据是平面上圆上的所有数据点，如果生成的数据点也位于这个圆上，则称生成的数据点满足圆的分布。如果真实数据是数轴上满足某种概率分布的实数，例如在区间`[0,1]`上均匀分布的实数，如果生成模型生成的实数也在`[0,1]`区间上均匀分布，即生成的实数与真实的实数具有相同的概率分布，则这两组实数难以区分。但通常我们只知道这些实数，而不知道这些实数背后的概率分布。如何生成与这些真实实数具有相同概率分布的实数？这就是生成模型要解决的问题。

生成模型通常是参数化模型，如参数化的神经网络函数。为了获得能生成服从真实数据分布的假数据的生成模型，需要基于真实数据学习参数化生成模型的参数。这与用真实数据学习回归模型参数是一样的。只要确定了参数化生成模型的参数，就能根据确定的生成模型自动生成服从真实数据分布的假数据，意味着这些假数据与真实数据难以区分。

当然，生成模型生成的数据分布不可能与真实数据分布完全相同。两个分布越接近，生成的数据就越难与真实数据区分。

如果有一组实数，即真实数据是实数，这些实数位于实数轴上，但它们在实数轴上的分布未知，如何生成难以与这些真实实数区分、或者说伪造数据服从这些真实实数分布的假实数？例如，这些实数是海南省人们的身高。如果生成的身高数据不服从这些身高数据的分布，就很容易被识别出来。

对于一组实数等低维数据，可以通过简单的统计计算用频率近似数据的概率分布。例如，下面是来自文件`real_values.npy`的一组实数，该集合`{x^(i)}`构成真实数据。

#### 分布分析方法

如何理解这组实数在实数空间 `R` 中的分布？或者说它们服从什么样的概率分布？可以将实数轴划分为许多小区间，统计落在每个小区间内的实数频数。只要数据量足够大，这个频数就能足够接近概率，从而了解这组实数在实数空间 `R` 中的概率分布。

下面的代码以直方图和曲线的形式展示了这组数据的近似概率频数分布：

```python
import numpy as np
from matplotlib import pyplot as plt

%matplotlib inline

def draw_hist(plt, x, bin_num=10):
    xmin, xmax = np.min(x), np.max(x)
    bins = np.linspace(xmin, xmax, bin_num)
    plt.hist(x, bins=bins, density=True, alpha=0.7)
    x2 = np.sort(x)  # 对实数进行排序
    p, _ = np.histogram(x2, bins, density=True)  # 计算 bins 中每个区间的频数 p
    p_x = np.linspace(xmin, xmax, len(p))
    plt.plot(p_x, p, 'b-', linewidth=2, label='real data')

draw_hist(plt, x, 26)
plt.show()
```
