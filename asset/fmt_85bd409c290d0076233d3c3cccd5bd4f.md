import numpy as np
import pandas as pd
import os
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split

# 定义 sigmoid 函数
def sigmoid(z):
    a = 1 / (1 + np.exp(-z))
    return a

# 初始化权重和偏置
def initialize(dim):
    w = np.zeros(dim).reshape(dim, 1)
    b = 0
    return w

# 计算损失
# 输入：numTraining, Y, A
# 输出：loss
def computeLoss(numTraining, Y, A):
    loss = -1/numTraining * np.sum(Y * np.log(A) + (1-Y) * (np.log(1-A)))
    return loss

# 执行前向传播
# 输入：w, b, X, Y
# 输出：gradients, loss (字典)
def forwardPropagation(w, b, X, Y):
    # 计算 Z
    Z = np.dot(w.T, X) + b

    # 确定训练样本数量
    numTraining = float(len(X))

    # 计算 sigmoid 激活函数的输出
    A = sigmoid(Z)

    # 计算损失
    loss = computeLoss(numTraining, Y, A)

    # 计算梯度 dZ, dw 和 db
    dZ = A - Y
    dw = 1/numTraining * np.dot(X, dZ.T)
    db = 1/numTraining * np.sum(dZ)

    # 以字典形式返回结果
    gradients = {"dw": dw, "db": db}
    loss = np.squeeze(loss)
    return gradients, loss

# 计算梯度下降
# 输入：w, b, X, Y, numIterations, learningRate
# 输出：params, grads, losses, idx
def gradientDescent(w, b, X, Y, numIterations, learningRate):
    losses = []
    idx = []

    # 迭代
    for i in range(numIterations):
        gradients, loss = forwardPropagation(w, b, X, Y)

        # 获取导数
        dw = gradients["dw"]
        db = gradients["db"]

        w = w - learningRate * dw
        b = b - learningRate * db

        # 存储损失
        if i % 100 == 0:
            idx.append(i)
            losses.append(loss)

    # 设置参数和梯度
    params = {"w": w, "b": b}
    grads = {"dw": dw, "db": db}
    return params, grads, losses, idx

# 预测训练集的输出
# 输入：w, b, X
# 输出：yPredicted
def predict(w, b, X):
    size = X.shape[1]
    yPredicted = np.zeros((1, size))
    Z = np.dot(w.T, X)

    # 计算 sigmoid
    A = sigmoid(Z)

    for i in range(A.shape[1]):
        # 如果值 > 0.5 则设为 1
        if A[0][i] > 0.5:
            yPredicted[0][i] = 1
        else:
            # 否则设为 0
            yPredicted[0][i] = 0
    return yPredicted

# 归一化数据
# 输入：x
# 输出：x (归一化后)
def normalize(x):
    x_norm = None
    x_norm = np.linalg.norm(x, axis=1, keepdims=True)
    x = x / x_norm
    return x

source("RFunctions-1.R")

# 定义 sigmoid 函数
sigmoid <- function(z) {
  a <- 1 / (1 + exp(-z))
  a
}

# 计算损失
# 输入：numTraining, Y, A
# 输出：loss
computeLoss <- function(numTraining, Y, A) {
  loss <- -1/numTraining * sum(Y * log(A) + (1-Y) * log(1-A))
  return(loss)
}

# 计算前向传播
# 输入：w, b, X, Y
# 输出：fwdProp (列表)
forwardPropagation <- function(w, b, X, Y) {
  # 计算 Z
  Z <- t(w) %*% X + b

  # 设置样本数量
  numTraining <- ncol(X)

  # 计算激活函数
  A <- sigmoid(Z)

  # 计算损失
  loss <- computeLoss(numTraining, Y, A)

  # 计算梯度 dZ, dw 和 db
  dZ <- A - Y
  dw <- 1/numTraining * X %*% t(dZ)
  db <- 1/numTraining * sum(dZ)

  fwdProp <- list("loss" = loss, "dw" = dw, "db" = db)
  return(fwdProp)
}

# 执行一个梯度下降周期
# 计算梯度下降
# 输入：w, b, X, Y, numIterations, learningRate
# 输出：gradDescnt (列表)
gradientDescent <- function(w, b, X, Y, numIterations, learningRate) {
  losses <- NULL
  idx <- NULL

  # 循环迭代次数
  for(i in 1:numIterations) {
    fwdProp <- forwardPropagation(w, b, X, Y)

    # 获取导数
    dw <- fwdProp$dw
    db <- fwdProp$db

    # 执行梯度下降
    w <- w - learningRate * dw
    b <- b - learningRate * db

    l <- fwdProp$loss

    # 存储损失
    if(i %% 100 == 0) {
      idx <- c(idx, i)
      losses <- c(losses, l)
    }
  }

  # 返回权重和损失
  gradDescnt <- list("w" = w, "b" = b, "dw" = dw, "db" = db, "losses" = losses, "idx" = idx)
  return(gradDescnt)
}

# 计算输入的预测值
# 预测训练集的输出
# 输入：w, b, X
# 输出：yPredicted
predict <- function(w, b, X) {
  m <- dim(X)[2]

  # 创建一个全 0 向量
  yPredicted <- matrix(rep(0, m), nrow = 1, ncol = m)

  Z <- t(w) %*% X + b

  # 计算 sigmoid
  A <- sigmoid(Z)

  for(i in 1:dim(A)[2]) {
    # 如果 A > 0.5 设值为 1
    if(A[1, i] > 0.5)
      yPredicted[1, i] <- 1
    else
      # 否则设为 0
      yPredicted[1, i] <- 0
  }
  return(yPredicted)
}

# 归一化矩阵
# 输入：x
# 输出：normalized
normalize <- function(x) {
  # 创建矩阵的范数。执行矩阵的 Frobenius 范数
  n <- as.matrix(sqrt(rowSums(x^2)))

  # 按行按范数扫描。注意函数中的 '1' 表示对每一行执行操作
  normalized <- sweep(x, 1, n, FUN = "/")
  return(normalized)
}
