public function newAction(Request $request) {
  $viewModel = [];
  $order = new Order();

  if ($request->getMethod() == 'POST') {
    $this->inputFilter
      ->setData($request->request->all());

    if ($this->inputFilter->isValid()) {
      $order = $this->hydrator->hydrate(
        $this->inputFilter->getValues(),
        $order

}
